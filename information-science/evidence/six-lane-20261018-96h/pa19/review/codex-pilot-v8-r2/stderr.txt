   333	class ProofSink:
   334	    """Owned FIFO reader writing the proof FILE with a HARD byte cap."""
   335	
   336	    def __init__(self, fifo_path, file_path, cap):
   337	        if os.path.lexists(fifo_path) or os.path.lexists(file_path):
   338	            raise RuntimeError("proof fifo/file path exists")
   339	        os.mkfifo(fifo_path, 0o600)
   340	        self.fifo_path, self.file_path, self.cap = fifo_path, file_path, cap
   341	        self.file = open(file_path, "xb")
   342	        self.sha = hashlib.sha256()
   343	        self.written = self.received = self.discarded = 0
   344	        self.over_cap = self.connected = self.eof = False
   345	        self.error = None
   346	        self.unblocked_by_runner = False
   347	        self.thread = threading.Thread(target=self._run, daemon=True)
   348	        self.thread.start()
   349	
   350	    def _run(self):
   351	        try:
   352	            fd = os.open(self.fifo_path, os.O_RDONLY)  # blocks until a writer opens the FIFO
   353	            self.connected = True
   354	            try:
   355	                while True:
   356	                    b = os.read(fd, 1 << 20)
   357	                    if not b:
   358	                        self.eof = True
   359	                        break
   360	                    self.received += len(b)
   361	                    if not self.over_cap and self.written + len(b) <= self.cap:
   362	                        self.file.write(b)
   363	                        self.sha.update(b)
   364	                        self.written += len(b)
   365	                    else:
   366	                        self.over_cap = True       # never write beyond the cap; drain so the producer can exit
   367	                        self.discarded += len(b)
   368	            finally:
   369	                os.close(fd)
   370	        except Exception as exc:
   371	            self.error = repr(exc)
   372	        finally:
   373	            try:
   374	                self.file.flush()
   375	                os.fsync(self.file.fileno())
   376	                self.file.close()
   377	            except Exception as exc:
   378	                self.error = (self.error or "") + " close:" + repr(exc)
   379	
   380	    def finish(self):
   381	        """After the producer is reaped: unblock a never-connected reader, then bounded join."""
   382	        deadline = time.time() + SINK_UNBLOCK
   383	        while self.thread.is_alive() and not self.connected and time.time() < deadline:
   384	            try:
   385	                fd = os.open(self.fifo_path, os.O_WRONLY | os.O_NONBLOCK)
   386	                self.unblocked_by_runner = True
   387	                os.close(fd)
   388	                break
   389	            except OSError as exc:
   390	                if exc.errno != errno.ENXIO:
   391	                    self.error = (self.error or "") + " unblock:" + repr(exc)
   392	                    break
   393	                time.sleep(0.05)
   394	        self.thread.join(STREAM_JOIN)
   395	        alive = self.thread.is_alive()
   396	        rec = {"fifo": self.fifo_path, "file": self.file_path, "cap": self.cap, "connected": self.connected,
   397	               "eof": self.eof, "received": self.received, "written": self.written, "discarded": self.discarded,
   398	               "over_cap": self.over_cap, "error": self.error, "unblocked_by_runner": self.unblocked_by_runner,
   399	               "thread_alive_after_join": alive, "sha256_of_written": self.sha.hexdigest()}
   400	        if alive:
   401	            LATCH.append("proof sink thread unsettled: " + self.fifo_path)
   402	        else:
   403	            try:
   404	                rec["file_rehash"] = list(sha_file(self.file_path))
   405	            except Exception as exc:
   406	                rec["file_rehash_error"] = repr(exc)
   407	        return rec

   148	def sha_file(path):
   149	    st = os.lstat(path)
   150	    if not stat.S_ISREG(st.st_mode):
   151	        raise RuntimeError("not a regular file: " + path)
   152	    h = hashlib.sha256()
   153	    n = 0
   154	    with open(path, "rb") as f:
   155	        for b in iter(lambda: f.read(1 << 20), b""):
   156	            h.update(b)
   157	            n += len(b)
   158	    return h.hexdigest(), n

   334	    """Owned FIFO reader writing the proof FILE with a HARD byte cap."""
   335	
   336	    def __init__(self, fifo_path, file_path, cap):
   337	        if os.path.lexists(fifo_path) or os.path.lexists(file_path):
   338	            raise RuntimeError("proof fifo/file path exists")
   339	        os.mkfifo(fifo_path, 0o600)

   341	        self.file = open(file_path, "xb")
   342	        self.sha = hashlib.sha256()
   343	        self.written = self.received = self.discarded = 0
   344	        self.over_cap = self.connected = self.eof = False
   345	        self.error = None
   346	        self.unblocked_by_runner = False
   347	        self.thread = threading.Thread(target=self._run, daemon=True)
   348	        self.thread.start()

   350	    def _run(self):
   351	        try:
   352	            fd = os.open(self.fifo_path, os.O_RDONLY)  # blocks until a writer opens the FIFO
   353	            self.connected = True
   354	            try:
   355	                while True:
   356	                    b = os.read(fd, 1 << 20)
   357	                    if not b:
   358	                        self.eof = True
   359	                        break
   360	                    self.received += len(b)
   361	                    if not self.over_cap and self.written + len(b) <= self.cap:
   362	                        self.file.write(b)
   363	                        self.sha.update(b)
   364	                        self.written += len(b)
   365	                    else:
   366	                        self.over_cap = True       # never write beyond the cap; drain so the producer can exit
   367	                        self.discarded += len(b)
   368	            finally:
   369	                os.close(fd)
   370	        except Exception as exc:
   371	            self.error = repr(exc)
   372	        finally:
   373	            try:
   374	                self.file.flush()
   375	                os.fsync(self.file.fileno())
   376	                self.file.close()
   377	            except Exception as exc:
   378	                self.error = (self.error or "") + " close:" + repr(exc)

   678	def verify_sat(cnf_path, kissat_lines, label):
   679	    """Two independent checks of a SAT model: clause satisfaction and both PA19 definitions."""
   680	    lits = set()
   681	    for s in kissat_lines:
   682	        if s.startswith("v "):
   683	            lits.update(int(x) for x in s[2:].split() if x != "0")
   684	    sat_all = True
   685	    with open(cnf_path, "rb") as f:
   686	        for ln in f:
   687	            if ln.startswith(b"p") or ln.startswith(b"c"):
   688	                continue
   689	            cl = [int(x) for x in ln.split()[:-1]]
   690	            if not any(l in lits for l in cl):
   691	                sat_all = False
   692	                break
   693	    A = [[None] * 8 for _ in range(19)]
   694	    cells_ok = True
   695	    for r in range(19):
   696	        for c in range(8):
   697	            ss = [s for s in range(6) if (1 + 48 * r + 6 * c + s) in lits]
   698	            if len(ss) != 1:
   699	                cells_ok = False
   700	            else:
   701	                A[r][c] = ss[0]
   702	    rowpair_ok = cells_ok and all(sum(A[r][c] == A[q][c] for c in range(8)) <= 1
   703	                                  for r in range(19) for q in range(r + 1, 19))
   704	    colpair_ok = cells_ok and all(len({(A[r][c], A[r][d]) for r in range(19)}) == 19
   705	                                  for c in range(8) for d in range(c + 1, 8))
   706	    cand = os.path.join(STATE["out"], label + ".SAT-CANDIDATE.json")
   707	    write_x(cand, {"host_clock": clock(), "array": A, "clauses_all_satisfied": sat_all,
   708	                   "cells_exactly_one": cells_ok, "row_pair_agreements_le_1": rowpair_ok,
   709	                   "column_pair_ordered_pairs_distinct": colpair_ok,
   710	                   "status": "CANDIDATE_UNVERIFIED - notify parent immediately"})
   711	    latch = os.path.join(LANE, "control", "SAT-CANDIDATE-" + label)
   712	    write_bytes_x(latch, cand.encode())
   713	    return {"candidate": cand, "latch": latch, "clauses_all_satisfied": sat_all, "cells": cells_ok,
   714	            "rowpair": rowpair_ok, "colpair": colpair_ok}

   717	def php_cnf(p, h):
   718	    """Pigeonhole: p pigeons into h holes; UNSAT iff p > h."""
   719	    var = lambda i, j: i * h + j + 1  # noqa: E731
   720	    cl = [[var(i, j) for j in range(h)] for i in range(p)]
   721	    for j in range(h):
   722	        for i in range(p):
   723	            for k in range(i + 1, p):
   724	                cl.append([-var(i, j), -var(k, j)])
   725	    return ("p cnf %d %d\n" % (p * h, len(cl)) + "".join(" ".join(map(str, c)) + " 0\n" for c in cl)).encode()
