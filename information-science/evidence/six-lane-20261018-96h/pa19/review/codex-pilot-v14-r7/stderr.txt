   617	    def reap_robust(self):
   618	        """r3#1/r4#1: exception-resistant reap state machine.  The whole retry path is protected; it
   619	        polls (never blocks) so a pending quarantine receipt keeps being retried; a SIGTERM is (re)tried
   620	        only while undelivered and only if termination was requested; it returns only after the retained
   621	        Popen is reaped."""
   622	        attempts = 0
   623	        while True:
   624	            try:
   625	                if self.stop is not None and self.term_sent_at is None and self.proc.returncode is None:
   626	                    ts = time.time()                       # r5#2
   627	                    self.proc.send_signal(signal.SIGTERM)
   628	                    self.term_sent_at = ts
   629	                if self.term_sent_at is not None and not self.quarantined and self.proc.poll() is None \
   630	                        and time.time() >= self.term_sent_at + TERM_WAIT:
   631	                    self.quarantined = True

   635	                    except BaseException:

   642	                                                       "no further signal; runner keeps waiting"})

   193	def bounded(x):
   194	    """r3#7: bound strings/lists recursively, replacing removed text by its SHA-256."""
   195	    if isinstance(x, str):
   196	        if len(x) <= STR_MAX:

   202	            y.append("...[%d more items]" % (len(x) - LIST_MAX))
   203	        return y
   204	    if isinstance(x, dict):
   205	        return {str(k)[:200]: bounded(v) for k, v in x.items()}

   233	def sha_file(path):
   234	    st = os.lstat(path)
   235	    if not stat.S_ISREG(st.st_mode):
   236	        raise RuntimeError("not a regular file: " + path)
   237	    h = hashlib.sha256()
   238	    n = 0
   239	    with open(path, "rb") as f:
   240	        for b in iter(lambda: f.read(1 << 20), b""):
   241	            h.update(b)
   242	            n += len(b)
   243	    return h.hexdigest(), n

   347	def stability_refusals(projected):
   348	    """resource-stability-addendum v1 rule over the v2-pinned v3 monitor; strict schema."""
   349	    st = os.lstat(MONITOR)
   350	    if not stat.S_ISREG(st.st_mode):
   351	        return ["monitor is not a regular file"]
   352	    with open(MONITOR, "rb") as f:
   353	        raw = f.read()
   354	    complete = raw.split(b"\n")[:-1]  # last element: b"" after a final newline, or an ignored partial line
   355	    rows = []
   356	    for i, ln in enumerate(complete):
   357	        try:
   358	            o = json.loads(ln)
   359	            if not isinstance(o, dict):
   360	                raise ValueError("not an object")
   361	            if "event" in o:
   362	                if not (i == 0 and set(o) == {"event", "at", "pid", "pgid", "cwd", "source_sha256"}
   363	                        and o["event"] == "observer_start" and o["source_sha256"] == OBSERVER_SOURCE_SHA
   364	                        and o["cwd"] == OBSERVER_CWD):
   365	                    return ["monitor event line %d is not the allowed first observer_start: %r" % (i, ln[:160])]
   366	                _aware(o["at"])
   367	                _int(o["pid"], 1)
   368	                _int(o["pgid"], 1)
   369	                continue
   370	            if set(o) != {"t", "free", "floor_bytes", "scope"} or o["floor_bytes"] != 12_000_000_000 \
   371	                    or o["scope"] != SAMPLE_SCOPE:
   372	                raise ValueError("sample schema/fixed values")
   373	            _int(o["floor_bytes"], 0)
   374	            rows.append((_aware(o["t"]).timestamp(), _int(o["free"], 0)))
   375	        except Exception as exc:
   376	            return ["malformed monitor line %d: %r (%r)" % (i, ln[:160], exc)]
   377	    if not rows:
   378	        return ["no monitor samples"]
   379	    for a, b in zip(rows, rows[1:]):
   380	        if not b[0] > a[0]:
   381	            return ["monitor timestamps not strictly increasing"]
   382	    why = []
   383	    now = time.time()
   384	    t_last, free_last = rows[-1]
   385	    if not (0 <= now - t_last <= 30):
   386	        why.append(f"monitor last sample age {now - t_last:.1f}s outside [0,30]")
   387	    win = [r for r in rows if now - 300 <= r[0] <= now]
   388	    if len(win) < 25:
   389	        why.append(f"only {len(win)} samples in [now-300, now]")
   390	    elif win[-1][0] - win[0][0] < 290:
   391	        why.append(f"window span {win[-1][0] - win[0][0]:.1f}s < 290 s")
   392	    if free_last < 13_500_000_000 + projected:
   393	        why.append(f"last free {free_last} < 13.5e9 + projected {projected}")
   394	    allowed = max(300_000_000, (free_last - 13_500_000_000) // 12)
   395	    peak, drop = None, 0
   396	    for _, fr in win:
   397	        peak = fr if peak is None else max(peak, fr)
   398	        drop = max(drop, peak - fr)
   399	    if drop > allowed:
   400	        why.append(f"peak-to-later drop {drop} > allowed {allowed}")
   401	    return why

   442	    def _run(self):
   443	        try:
   444	            fd = os.open(self.fifo_path, os.O_RDONLY)  # blocks until a writer opens the FIFO
   445	            self.connected = True
   446	            try:
   447	                while True:
   448	                    b = os.read(fd, 1 << 20)
   449	                    if not b:
   450	                        self.eof = True
   451	                        break

   453	                    if not self.over_cap and self.written + len(b) <= self.cap:
   454	                        self.file.write(b)
   455	                        self.sha.update(b)
   456	                        self.written += len(b)
   457	                    else:
   458	                        self.over_cap = True       # never write beyond the cap; drain so the producer can exit

   461	                os.close(fd)
   462	        except Exception as exc:
   463	            self.error = repr(exc)
   464	        finally:
   465	            try:
   466	                self.file.flush()
   467	                os.fsync(self.file.fileno())
   468	                self.file.close()

   470	                self.error = (self.error or "") + " close:" + repr(exc)

   982	def php_cnf(p, h):
   983	    """Pigeonhole: p pigeons into h holes; UNSAT iff p > h."""
   984	    var = lambda i, j: i * h + j + 1  # noqa: E731
   985	    cl = [[var(i, j) for j in range(h)] for i in range(p)]
   986	    for j in range(h):
   987	        for i in range(p):
   988	            for k in range(i + 1, p):
   989	                cl.append([-var(i, j), -var(k, j)])
   990	    return ("p cnf %d %d\n" % (p * h, len(cl)) + "".join(" ".join(map(str, c)) + " 0\n" for c in cl)).encode()
