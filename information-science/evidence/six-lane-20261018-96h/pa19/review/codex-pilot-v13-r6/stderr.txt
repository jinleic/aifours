   186	def bounded(x):
   187	    """r3#7: bound strings/lists recursively, replacing removed text by its SHA-256."""
   188	    if isinstance(x, str):
   189	        if len(x) <= STR_MAX:
   190	            return x
   191	        return x[:STR_MAX] + "...[truncated %d chars sha256 %s]" % (len(x), hashlib.sha256(x.encode()).hexdigest())
   192	    if isinstance(x, (list, tuple)):
   193	        y = [bounded(v) for v in x[:LIST_MAX]]
   194	        if len(x) > LIST_MAX:
   195	            y.append("...[%d more items]" % (len(x) - LIST_MAX))
   196	        return y
   197	    if isinstance(x, dict):
   198	        return {str(k)[:200]: bounded(v) for k, v in x.items()}
   199	    return x

   226	def sha_file(path):
   227	    st = os.lstat(path)
   228	    if not stat.S_ISREG(st.st_mode):
   229	        raise RuntimeError("not a regular file: " + path)
   230	    h = hashlib.sha256()
   231	    n = 0
   232	    with open(path, "rb") as f:
   233	        for b in iter(lambda: f.read(1 << 20), b""):
   234	            h.update(b)
   235	            n += len(b)
   236	    return h.hexdigest(), n

   340	def stability_refusals(projected):
   341	    """resource-stability-addendum v1 rule over the v2-pinned v3 monitor; strict schema."""
   342	    st = os.lstat(MONITOR)
   343	    if not stat.S_ISREG(st.st_mode):
   344	        return ["monitor is not a regular file"]
   345	    with open(MONITOR, "rb") as f:
   346	        raw = f.read()
   347	    complete = raw.split(b"\n")[:-1]  # last element: b"" after a final newline, or an ignored partial line
   348	    rows = []
   349	    for i, ln in enumerate(complete):
   350	        try:
   351	            o = json.loads(ln)
   352	            if not isinstance(o, dict):
   353	                raise ValueError("not an object")
   354	            if "event" in o:
   355	                if not (i == 0 and set(o) == {"event", "at", "pid", "pgid", "cwd", "source_sha256"}
   356	                        and o["event"] == "observer_start" and o["source_sha256"] == OBSERVER_SOURCE_SHA
   357	                        and o["cwd"] == OBSERVER_CWD):
   358	                    return ["monitor event line %d is not the allowed first observer_start: %r" % (i, ln[:160])]
   359	                _aware(o["at"])
   360	                _int(o["pid"], 1)
   361	                _int(o["pgid"], 1)
   362	                continue
   363	            if set(o) != {"t", "free", "floor_bytes", "scope"} or o["floor_bytes"] != 12_000_000_000 \
   364	                    or o["scope"] != SAMPLE_SCOPE:
   365	                raise ValueError("sample schema/fixed values")
   366	            _int(o["floor_bytes"], 0)
   367	            rows.append((_aware(o["t"]).timestamp(), _int(o["free"], 0)))
   368	        except Exception as exc:
   369	            return ["malformed monitor line %d: %r (%r)" % (i, ln[:160], exc)]
   370	    if not rows:
   371	        return ["no monitor samples"]
   372	    for a, b in zip(rows, rows[1:]):
   373	        if not b[0] > a[0]:
   374	            return ["monitor timestamps not strictly increasing"]
   375	    why = []
   376	    now = time.time()
   377	    t_last, free_last = rows[-1]
   378	    if not (0 <= now - t_last <= 30):
   379	        why.append(f"monitor last sample age {now - t_last:.1f}s outside [0,30]")
   380	    win = [r for r in rows if now - 300 <= r[0] <= now]
   381	    if len(win) < 25:
   382	        why.append(f"only {len(win)} samples in [now-300, now]")
   383	    elif win[-1][0] - win[0][0] < 290:
   384	        why.append(f"window span {win[-1][0] - win[0][0]:.1f}s < 290 s")
   385	    if free_last < 13_500_000_000 + projected:
   386	        why.append(f"last free {free_last} < 13.5e9 + projected {projected}")
   387	    allowed = max(300_000_000, (free_last - 13_500_000_000) // 12)
   388	    peak, drop = None, 0
   389	    for _, fr in win:
   390	        peak = fr if peak is None else max(peak, fr)
   391	        drop = max(drop, peak - fr)
   392	    if drop > allowed:
   393	        why.append(f"peak-to-later drop {drop} > allowed {allowed}")
   394	    return why

   398	    """Blocking part of admission (registry, monitor, statvfs, fit). Returns refusal reasons."""

   416	class ProofSink:
   417	    """Owned FIFO reader writing the proof FILE with a HARD byte cap."""
   418	
   419	    def __init__(self, fifo_path, file_path, cap):
   420	        if os.path.lexists(fifo_path) or os.path.lexists(file_path):
   421	            raise RuntimeError("proof fifo/file path exists")
   422	        os.mkfifo(fifo_path, 0o600)
   423	        self.fifo_ino = os.lstat(fifo_path).st_ino
   424	        self.fifo_path, self.file_path, self.cap = fifo_path, file_path, cap
   425	        self.file = open(file_path, "xb")
   426	        self.sha = hashlib.sha256()
   427	        self.written = self.received = self.discarded = 0
   428	        self.over_cap = self.connected = self.eof = False
   429	        self.error = None
   430	        self.unblocked_by_runner = False
   431	        self.finished = None
   432	        self.thread = threading.Thread(target=self._run, daemon=True)
   433	        self.thread.start()
   434	
   435	    def _run(self):
   436	        try:
   437	            fd = os.open(self.fifo_path, os.O_RDONLY)  # blocks until a writer opens the FIFO
   438	            self.connected = True
   439	            try:
   440	                while True:
   441	                    b = os.read(fd, 1 << 20)
   442	                    if not b:
   443	                        self.eof = True
   444	                        break
   445	                    self.received += len(b)
   446	                    if not self.over_cap and self.written + len(b) <= self.cap:
   447	                        self.file.write(b)
   448	                        self.sha.update(b)
   449	                        self.written += len(b)
   450	                    else:
   451	                        self.over_cap = True       # never write beyond the cap; drain so the producer can exit
   452	                        self.discarded += len(b)
   453	            finally:
   454	                os.close(fd)
   455	        except Exception as exc:
   456	            self.error = repr(exc)
   457	        finally:
   458	            try:
   459	                self.file.flush()
   460	                os.fsync(self.file.fileno())

 succeeded in 0ms:
   461	                self.file.close()
   462	            except Exception as exc:
   463	                self.error = (self.error or "") + " close:" + repr(exc)
   464	
   465	    def finish(self):
   466	        """After the producer is reaped: unblock a never-connected reader, bounded join; latches on an
   467	        unsettled thread, sink I/O error, re-hash failure or a replaced FIFO."""
   468	        if self.finished is not None:
   469	            return self.finished
   470	        deadline = time.time() + SINK_UNBLOCK
   471	        while self.thread.is_alive() and not self.connected and time.time() < deadline:
   472	            try:
   473	                fd = os.open(self.fifo_path, os.O_WRONLY | os.O_NONBLOCK)
   474	                self.unblocked_by_runner = True
   475	                os.close(fd)
   476	                break
   477	            except OSError as exc:
   478	                if exc.errno != errno.ENXIO:
   479	                    self.error = (self.error or "") + " unblock:" + repr(exc)
   480	                    break
   481	                time.sleep(0.05)
   482	        self.thread.join(STREAM_JOIN)
   483	        alive = self.thread.is_alive()
   484	        rec = {"fifo": self.fifo_path, "file": self.file_path, "cap": self.cap, "connected": self.connected,
   485	               "eof": self.eof, "received": self.received, "written": self.written, "discarded": self.discarded,
   486	               "over_cap": self.over_cap, "error": self.error, "unblocked_by_runner": self.unblocked_by_runner,
   487	               "thread_alive_after_join": alive, "sha256_of_written": self.sha.hexdigest()}
   488	        try:
   489	            st = os.lstat(self.fifo_path)
   490	            rec["fifo_same_inode_still_fifo"] = bool(stat.S_ISFIFO(st.st_mode) and st.st_ino == self.fifo_ino)
   491	        except Exception as exc:
   492	            rec["fifo_same_inode_still_fifo"] = False
   493	            rec["fifo_check_error"] = repr(exc)
   494	        if not rec["fifo_same_inode_still_fifo"]:
   495	            latch("proof FIFO replaced or missing: " + self.fifo_path)
   496	        if alive:
   497	            latch("proof sink thread unsettled: " + self.fifo_path)
   498	        else:
   499	            try:
   500	                rec["file_rehash"] = list(sha_file(self.file_path))
   501	            except Exception as exc:
   502	                rec["file_rehash_error"] = repr(exc)
   503	                latch("proof file re-hash failed %r" % exc)
   504	        if self.error:
   505	            latch("proof sink I/O error %s" % self.error)
   506	        rec["sink_settled"] = (not alive and self.error is None and "file_rehash" in rec
   507	                               and rec["fifo_same_inode_still_fifo"])
   508	        self.finished = rec
   509	        return rec

   765	def supervise(child, sink):
   766	    """Monitor until exit or a single trip; any monitor exception is itself a trip."""
   767	    me = os.getpid()
   768	    last_rss = last_reg = 0.0
   769	    peak = 0

   772	            now = time.time()
   773	            if cancelled():
   774	                child.term("owner_cancel_sigterm")
   775	            elif now > child.deadline:
   776	                child.term("absolute_or_wall_deadline")
   777	            elif sink is not None and sink.over_cap:
   778	                child.term("proof_cap")
   779	            elif sink is not None and sink.error:
   780	                child.term("proof_sink_error")
   781	            elif child.over_cap:
   782	                child.term("capture_cap")
   783	            elif free_bytes() < FREE_FLOOR:
   784	                child.term("inflight_disk_floor")
   785	            if child.term_sent_at is None and now - last_rss >= 1.0:
   786	                last_rss = now
   787	                found, rss = child.census()
   788	                rows = ps_table()
   789	                runner_rss = sum(r[3] for r in rows if r[0] == me) * 1024
   790	                if found:
   791	                    child.descendants.extend(found)
   792	                    child.term("descendant_process")
   793	                total = rss + runner_rss
   794	                peak = max(peak, total)
   795	                if runner_rss == 0 or (rss == 0 and child.alive()):
   796	                    child.term("rss_unreadable")
   797	                elif total > LANE_RSS_TRIP:
   798	                    child.term("lane_rss")
   799	            if child.term_sent_at is None and now - last_reg >= 5.0:
   800	                last_reg = now
   801	                if registry_refusals():
   802	                    child.term("stop_registry")
   803	        except Exception as exc:
   804	            child.term("monitor_exception:%r" % exc)
   805	        if child.term_sent_at is not None:
   806	            break
   807	        time.sleep(0.25)
   808	    if child.stop is not None and child.stop not in NONLATCHING_TRIPS:
   809	        latch("trip %s on %s" % (child.stop, child.label))
   810	    return child.stop, peak
