   177	def bounded(x):
   178	    """r3#7: bound strings/lists recursively, replacing removed text by its SHA-256."""
   179	    if isinstance(x, str):
   180	        if len(x) <= STR_MAX:
   181	            return x

   183	    if isinstance(x, (list, tuple)):
   184	        y = [bounded(v) for v in x[:LIST_MAX]]
   185	        if len(x) > LIST_MAX:
   186	            y.append("...[%d more items]" % (len(x) - LIST_MAX))
   187	        return y
   188	    if isinstance(x, dict):
   189	        return {str(k)[:200]: bounded(v) for k, v in x.items()}
   190	    return x

   217	def sha_file(path):
   218	    st = os.lstat(path)
   219	    if not stat.S_ISREG(st.st_mode):
   220	        raise RuntimeError("not a regular file: " + path)
   221	    h = hashlib.sha256()
   222	    n = 0
   223	    with open(path, "rb") as f:
   224	        for b in iter(lambda: f.read(1 << 20), b""):
   225	            h.update(b)
   226	            n += len(b)
   227	    return h.hexdigest(), n

   381	        peak = fr if peak is None else max(peak, fr)
   382	        drop = max(drop, peak - fr)
   383	    if drop > allowed:
   384	        why.append(f"peak-to-later drop {drop} > allowed {allowed}")
   385	    return why

   388	def full_admission(projected, child_deadline):
   389	    """Blocking part of admission (registry, monitor, statvfs, fit). Returns refusal reasons."""
   390	    why = []
   391	    try:
   392	        why += registry_refusals()
   393	        why += stability_refusals(projected)
   394	        fb = free_bytes()
   395	        if fb < FRESH_BASE + projected:
   396	            why.append(f"fresh free {fb} < 12.2e9 + {projected}")
   397	        if fb - STATE["remaining_reserve"] < FREE_FLOOR:
   398	            why.append(f"free {fb} - remaining reservations {STATE['remaining_reserve']} < 12e9")
   399	    except Exception as exc:
   400	        why.append("admission exception %r" % exc)
   401	    if child_deadline + SETTLE_TAIL > STATE["abs_end"] or child_deadline + SETTLE_TAIL + MARGIN > STATE["chunk_end"] \
   402	            or child_deadline + SETTLE_TAIL > OWNED_SETTLED_BY - 180:
   403	        why.append("per-child chunk fit / absolute end (with full settlement tail)")
   404	    return why

   407	class ProofSink:
   408	    """Owned FIFO reader writing the proof FILE with a HARD byte cap."""
   409	
   410	    def __init__(self, fifo_path, file_path, cap):
   411	        if os.path.lexists(fifo_path) or os.path.lexists(file_path):
   412	            raise RuntimeError("proof fifo/file path exists")
   413	        os.mkfifo(fifo_path, 0o600)
   414	        self.fifo_ino = os.lstat(fifo_path).st_ino
   415	        self.fifo_path, self.file_path, self.cap = fifo_path, file_path, cap
   416	        self.file = open(file_path, "xb")
   417	        self.sha = hashlib.sha256()
   418	        self.written = self.received = self.discarded = 0
   419	        self.over_cap = self.connected = self.eof = False
   420	        self.error = None
   421	        self.unblocked_by_runner = False
   422	        self.finished = None
   423	        self.thread = threading.Thread(target=self._run, daemon=True)
   424	        self.thread.start()
   425	
   426	    def _run(self):
   427	        try:
   428	            fd = os.open(self.fifo_path, os.O_RDONLY)  # blocks until a writer opens the FIFO
   429	            self.connected = True
   430	            try:
   431	                while True:
   432	                    b = os.read(fd, 1 << 20)
   433	                    if not b:
   434	                        self.eof = True
   435	                        break
   436	                    self.received += len(b)
   437	                    if not self.over_cap and self.written + len(b) <= self.cap:
   438	                        self.file.write(b)
   439	                        self.sha.update(b)
   440	                        self.written += len(b)
   441	                    else:
   442	                        self.over_cap = True       # never write beyond the cap; drain so the producer can exit
   443	                        self.discarded += len(b)
   444	            finally:
   445	                os.close(fd)
   446	        except Exception as exc:
   447	            self.error = repr(exc)
   448	        finally:
   449	            try:
   450	                self.file.flush()
   451	                os.fsync(self.file.fileno())
   452	                self.file.close()
   453	            except Exception as exc:
   454	                self.error = (self.error or "") + " close:" + repr(exc)
   455	
   456	    def finish(self):
   457	        """After the producer is reaped: unblock a never-connected reader, bounded join; latches on an
   458	        unsettled thread, sink I/O error, re-hash failure or a replaced FIFO."""
   459	        if self.finished is not None:
   460	            return self.finished

 succeeded in 0ms:
   461	        deadline = time.time() + SINK_UNBLOCK
   462	        while self.thread.is_alive() and not self.connected and time.time() < deadline:
   463	            try:
   464	                fd = os.open(self.fifo_path, os.O_WRONLY | os.O_NONBLOCK)
   465	                self.unblocked_by_runner = True
   466	                os.close(fd)
   467	                break
   468	            except OSError as exc:
   469	                if exc.errno != errno.ENXIO:
   470	                    self.error = (self.error or "") + " unblock:" + repr(exc)
   471	                    break
   472	                time.sleep(0.05)
   473	        self.thread.join(STREAM_JOIN)
   474	        alive = self.thread.is_alive()
   475	        rec = {"fifo": self.fifo_path, "file": self.file_path, "cap": self.cap, "connected": self.connected,
   476	               "eof": self.eof, "received": self.received, "written": self.written, "discarded": self.discarded,
   477	               "over_cap": self.over_cap, "error": self.error, "unblocked_by_runner": self.unblocked_by_runner,
   478	               "thread_alive_after_join": alive, "sha256_of_written": self.sha.hexdigest()}
   479	        try:
   480	            st = os.lstat(self.fifo_path)
   481	            rec["fifo_same_inode_still_fifo"] = bool(stat.S_ISFIFO(st.st_mode) and st.st_ino == self.fifo_ino)
   482	        except Exception as exc:
   483	            rec["fifo_same_inode_still_fifo"] = False
   484	            rec["fifo_check_error"] = repr(exc)
   485	        if not rec["fifo_same_inode_still_fifo"]:
   486	            latch("proof FIFO replaced or missing: " + self.fifo_path)
   487	        if alive:
   488	            latch("proof sink thread unsettled: " + self.fifo_path)
   489	        else:
   490	            try:
   491	                rec["file_rehash"] = list(sha_file(self.file_path))
   492	            except Exception as exc:
   493	                rec["file_rehash_error"] = repr(exc)
   494	                latch("proof file re-hash failed %r" % exc)
   495	        if self.error:
   496	            latch("proof sink I/O error %s" % self.error)
   497	        rec["sink_settled"] = (not alive and self.error is None and "file_rehash" in rec
   498	                               and rec["fifo_same_inode_still_fifo"])
   499	        self.finished = rec
   500	        return rec

   566	    def _read(self):
   567	        try:
   568	            while True:
   569	                b = self.proc.stdout.read1(65536)
   570	                if not b:
   571	                    break
   572	                room = LOG_CAP - self.captured
   573	                if room > 0:
   574	                    self.capture.write(b[:room])
   575	                self.captured += len(b)
   576	                if self.captured > LOG_CAP:
   577	                    self.over_cap = True
   578	        except Exception as exc:
   579	            self.reader_error = repr(exc)
   580	        finally:
   581	            try:
   582	                self.capture.flush()
   583	                os.fsync(self.capture.fileno())
   584	                self.capture.close()
   585	                self.proc.stdout.close()
   586	            except Exception as exc:
   587	                self.reader_error = (self.reader_error or "") + " close:" + repr(exc)
   588	
   589	    def alive(self):
   590	        return self.proc.poll() is None
   591	
   592	    def term(self, reason):
   593	        if self.term_sent_at is None and self.proc.returncode is None:
   594	            if self.stop is None:
   595	                self.stop = reason
   596	            self.proc.send_signal(signal.SIGTERM)  # Popen polls first; never signals a reaped PID
   597	            self.term_sent_at = time.time()        # r4#1: marked only after delivery returned
   598	
   599	    def reap_robust(self):
   600	        """r3#1/r4#1: exception-resistant reap state machine.  The whole retry path is protected; it
   601	        polls (never blocks) so a pending quarantine receipt keeps being retried; a SIGTERM is (re)tried
   602	        only while undelivered and only if termination was requested; it returns only after the retained
   603	        Popen is reaped."""
   604	        receipt_pending = False
   605	        while True:
   606	            try:
   607	                if self.stop is not None and self.term_sent_at is None and self.proc.returncode is None:
   608	                    self.proc.send_signal(signal.SIGTERM)
   609	                    self.term_sent_at = time.time()
   610	                if self.term_sent_at is not None and not self.quarantined and self.proc.poll() is None \
   611	                        and time.time() >= self.term_sent_at + TERM_WAIT:
   612	                    self.quarantined = True
   613	                    receipt_pending = True
   614	                    try:
   615	                        latch("quarantine pid %d" % self.pid)
   616	                    except BaseException:
   617	                        pass
   618	                if receipt_pending:
   619	                    try:
   620	                        write_x(self.quarantine_path, {"pid": self.pid, "pgid": self.pgid, "argv": self.argv,
   621	                                                       "epoch": time.time(), "note": "survived SIGTERM + 60 s; "
   622	                                                       "no further signal; runner keeps waiting"})
   623	                        receipt_pending = False
   624	                    except FileExistsError:
   625	                        receipt_pending = False
   626	                    except BaseException:
   627	                        self.quarantine_receipt_failures = getattr(self, "quarantine_receipt_failures", 0) + 1
   628	                rc = self.proc.poll()
   629	                if rc is not None and not receipt_pending:
   630	                    return self.proc.wait()
   631	                time.sleep(0.1)
   632	            except BaseException:
   633	                try:
   634	                    time.sleep(0.05)
   635	                except BaseException:
   636	                    pass
   637	
   638	    def census(self):
   639	        """Other processes with ppid==child or pgid==child (excluding the child itself)."""
   640	        rows = ps_table()
   641	        found = [r for r in rows if r[0] != self.pid and (r[1] == self.pid or r[2] == self.pgid)]
   642	        rss = sum(r[3] for r in rows if r[0] == self.pid or r in found) * 1024
   643	        return found, rss

   645	    def settle(self):
   646	        """Always reaches wait(); settlement final only after streams, sink, FIFO/cwd/lane checks,
   647	        hashes and the receipt; any failure clears 'settled' and latches."""
   648	        if self.settled_rec is not None:
   649	            return self.settled_rec
   650	        problems = []
   651	        rc = self.reap_robust()
   652	        stream_alive = None
   653	        if self.reader is not None:
   654	            self.reader.join(STREAM_JOIN)
   655	            stream_alive = self.reader.is_alive()
   656	        else:
   657	            problems.append("no capture reader")
   658	            try:
   659	                self.capture.close()
   660	            except Exception as exc:
   661	                problems.append("capture close %r" % exc)
   662	        group_empty = None
   663	        try:
   664	            os.killpg(self.pgid, 0)
   665	            group_empty = False
   666	        except ProcessLookupError:
   667	            group_empty = True
   668	        except BaseException as exc:
   669	            problems.append("killpg probe %r" % exc)
   670	        try:
   671	            after, _ = self.census()
   672	            if after:
   673	                problems.append("descendants after reap %r" % after)
   674	        except BaseException as exc:
   675	            problems.append("post-reap census failed %r" % exc)
   676	        if self.descendants:
   677	            problems.append("descendants seen during run %r" % self.descendants[:5])
   678	        sink_rec = None
   679	        if self.sink_obj is not None:
   680	            try:
   681	                sink_rec = self.sink_obj.finish()
   682	                if not sink_rec["sink_settled"]:
   683	                    problems.append("proof sink not settled")
   684	            except BaseException as exc:
   685	                problems.append("sink finish failed %r" % exc)
   686	        try:
   687	            if os.listdir(self.cwd):
   688	                problems.append("child cwd not empty: %r" % os.listdir(self.cwd)[:10])
   689	        except BaseException as exc:
   690	            problems.append("cwd check failed %r" % exc)
   691	        if self.lane_before is not None:
   692	            try:
   693	                expected = {self.out_path, self.launch_path, self.settle_path, self.quarantine_path}
   694	                if self.sink_obj is not None:
   695	                    expected.add(self.sink_obj.file_path)
   696	                now = lane_files()
   697	                unexpected = [p for p, v in now.items()
   698	                              if p not in expected and (p not in self.lane_before or self.lane_before[p] != v)]
   699	                if unexpected:
   700	                    problems.append("unexpected lane file changes %r" % unexpected[:10])
   701	            except BaseException as exc:
   702	                problems.append("lane census failed %r" % exc)
   703	        diag = []
   704	        out_sha = None
   705	        if stream_alive is False:
   706	            try:
   707	                out_sha = sha_file(self.out_path)[0]
   708	                with open(self.out_path, "rb") as f:
   709	                    diag = sorted(set(m.group(0) for m in RESOURCE_DIAG.finditer(f.read().decode("utf-8", "replace"))))
   710	            except BaseException as exc:
   711	                problems.append("capture hash/scan failed %r" % exc)
   712	        if diag:
   713	            latch("native memory diagnostic in %s: %r" % (self.label, diag))  # r3#3
   714	        rec = {"seq": self.seq, "label": self.label, "pid": self.pid, "pgid": self.pgid,
   715	               "identity_at_start": self.identity_at_start, "argv": self.argv, "returncode": rc,
   716	               "stop": self.stop or "self", "seconds": round(time.time() - self.started, 3),
   717	               "captured_bytes": self.captured, "capture_over_cap": self.over_cap, "reader_error": self.reader_error,
   718	               "stream_thread_alive_after_join": stream_alive, "group_empty_after_reap": group_empty,
   719	               "quarantined": self.quarantined, "problems": problems, "resource_diagnostics": diag,
   720	               "proof_sink": sink_rec, "launch_record": self.launch_path, "output_path": self.out_path,
   721	               "output_sha256": out_sha, "settled": False}
   722	        ok = (stream_alive is False and group_empty is True and not self.quarantined and not problems
   723	              and self.reader_error is None and out_sha is not None)
   724	        try:
   725	            write_x(self.settle_path, dict(rec, settled=ok))
   726	            rec["settled"] = ok
   727	        except BaseException as exc:
   728	            rec["settle_receipt_error"] = repr(exc)
   729	        if not rec["settled"]:
   730	            latch("child %d %s not settled" % (self.seq, self.label))
   731	        self.settled_rec = rec
   732	        return rec

   741	def supervise(child, sink):
   742	    """Monitor until exit or a single trip; any monitor exception is itself a trip."""
   743	    me = os.getpid()
   744	    last_rss = last_reg = 0.0
   745	    peak = 0
   746	    while child.alive():
   747	        try:
   748	            now = time.time()
   749	            if cancelled():
   750	                child.term("owner_cancel_sigterm")
   751	            elif now > child.deadline:
   752	                child.term("absolute_or_wall_deadline")
   753	            elif sink is not None and sink.over_cap:
   754	                child.term("proof_cap")
   755	            elif sink is not None and sink.error:
   756	                child.term("proof_sink_error")
   757	            elif child.over_cap:
   758	                child.term("capture_cap")
   759	            elif free_bytes() < FREE_FLOOR:
   760	                child.term("inflight_disk_floor")
   761	            if child.term_sent_at is None and now - last_rss >= 1.0:
   762	                last_rss = now
   763	                found, rss = child.census()
   764	                rows = ps_table()
   765	                runner_rss = sum(r[3] for r in rows if r[0] == me) * 1024
   766	                if found:
   767	                    child.descendants.extend(found)
   768	                    child.term("descendant_process")
   769	                total = rss + runner_rss
   770	                peak = max(peak, total)
   771	                if runner_rss == 0 or (rss == 0 and child.alive()):
   772	                    child.term("rss_unreadable")
   773	                elif total > LANE_RSS_TRIP:
   774	                    child.term("lane_rss")
   775	            if child.term_sent_at is None and now - last_reg >= 5.0:
   776	                last_reg = now
   777	                if registry_refusals():
   778	                    child.term("stop_registry")
   779	        except Exception as exc:
   780	            child.term("monitor_exception:%r" % exc)
   781	        if child.term_sent_at is not None:
   782	            break
   783	        time.sleep(0.25)
   784	    if child.stop is not None and child.stop not in NONLATCHING_TRIPS:
   785	        latch("trip %s on %s" % (child.stop, child.label))
   786	    return child.stop, peak

   789	def run_child(argv, label, wall, reserve, inputs, sink=None):
   790	    """Prepared outputs -> admission -> spawn -> supervise -> settle.  Returns (rec|None, lines, refusal)."""
   791	    ch = None
   792	    try:
   793	        cwd = os.path.join(STATE["out"], "cwd", "%02d-%s" % (STATE["seq"] + 1, label))
   794	        os.makedirs(cwd)  # r4#4: inside the cleanup guard
   795	        ch = Child(argv, label, wall, reserve, inputs, sink, cwd)
   796	        if ch.refused:
   797	            return None, [], ch.refused
   798	        peak = None
   799	        try:
   800	            if cancelled():
   801	                ch.term("owner_cancel_sigterm")
   802	            _, peak = supervise(ch, sink)
   803	        except BaseException as exc:
   804	            latch("runner exception during supervision %r" % exc)
   805	            ch.term("runner_exception")
   806	        rec = ch.settle()
   807	        rec["trip"] = ch.stop
   808	        rec["peak_sampled_lane_rss_bytes"] = peak
   809	        lines = segments(ch.out_path) if rec.get("stream_thread_alive_after_join") is False else []
   810	        return rec, lines, []
   811	    finally:
   812	        for c in OWNED:
   813	            if c.proc is not None and c.settled_rec is None:
   814	                if c.stop is None and c.proc.poll() is None:
   815	                    c.stop = "runner_escape_cleanup"  # r4#1: request termination before settling
   816	                c.settle()  # any escape above still reaps and settles every owned child
   817	        if sink is not None and (ch is None or ch.proc is None):
   818	            sink.finish()  # never-spawned producer: unblock and join the reader

   848	def qualify_unsat(cnf_expect, solver_rec, kissat_lines, proof_sha, proof_bytes, proof_cap, drat_rec, drat_lines,
   849	                  cnf_post, proof_post):
   850	    bad = []
   851	    if solver_rec["returncode"] != 20 or solver_rec["stop"] != "self":
   852	        bad.append("solver not self-exit rc20")
   853	    if status_lines(kissat_lines) != ["s UNSATISFIABLE"]:
   854	        bad.append("solver status lines not exactly ['s UNSATISFIABLE']")
   855	    if not sink_complete(solver_rec.get("proof_sink")):
   856	        bad.append("proof sink incomplete/over cap/error")
   857	    if proof_bytes <= 0 or proof_bytes > proof_cap:
   858	        bad.append("proof size outside (0, cap]")
   859	    if drat_rec["returncode"] != 0 or drat_rec["stop"] != "self":
   860	        bad.append("checker not self-exit rc0")
   861	    if status_lines(drat_lines) != ["s VERIFIED"]:
   862	        bad.append("checker status lines not exactly ['s VERIFIED']")
   863	    v, c = cnf_expect[2].split()[2:4]
   864	    if ("c parsing input formula with %s variables and %s clauses" % (v, c)) not in drat_lines:
   865	        bad.append("checker parse header line missing/mismatched")
   866	    if ("c finished parsing, read %d bytes from proof file" % proof_bytes) not in drat_lines:
   867	        bad.append("checker-consumed proof bytes != proof file bytes")
   868	    for pat in ("early EOF", "clauses less than expected", "trivial UNSAT", "ERROR", "MEMOUT",
   869	                "wrong binary prefix", "no empty clause detected", "NOT VERIFIED", "TIMEOUT"):
   870	        if any(pat in s for s in drat_lines):
   871	            bad.append("checker output contains %r" % pat)
   872	    if cnf_post != (cnf_expect[0], cnf_expect[1]):
   873	        bad.append("CNF file digest/bytes != expected")
   874	    if proof_post != (proof_sha, proof_bytes):
   875	        bad.append("proof file digest/bytes != sink measurement")
   876	    for rec in (solver_rec, drat_rec):
   877	        if not rec["settled"] or rec["capture_over_cap"]:
   878	            bad.append("settlement/capture defect seq %s" % rec["seq"])
   879	    return bad

   882	def verify_sat(cnf_path, kissat_lines, label):
   883	    """Two independent checks of a SAT model: clause satisfaction and both PA19 definitions."""
   884	    lits = set()
   885	    for s in kissat_lines:
   886	        if s.startswith("v "):
   887	            lits.update(int(x) for x in s[2:].split() if x != "0")
   888	    sat_all = True
   889	    with open(cnf_path, "rb") as f:
   890	        for ln in f:
   891	            if ln.startswith(b"p") or ln.startswith(b"c"):
   892	                continue
   893	            cl = [int(x) for x in ln.split()[:-1]]
   894	            if not any(l in lits for l in cl):
   895	                sat_all = False
   896	                break
   897	    A = [[None] * 8 for _ in range(19)]
   898	    cells_ok = True
   899	    for r in range(19):
   900	        for c in range(8):

 succeeded in 0ms:
   901	            ss = [s for s in range(6) if (1 + 48 * r + 6 * c + s) in lits]
   902	            if len(ss) != 1:
   903	                cells_ok = False
   904	            else:
   905	                A[r][c] = ss[0]
   906	    rowpair_ok = cells_ok and all(sum(A[r][c] == A[q][c] for c in range(8)) <= 1
   907	                                  for r in range(19) for q in range(r + 1, 19))
   908	    colpair_ok = cells_ok and all(len({(A[r][c], A[r][d]) for r in range(19)}) == 19
   909	                                  for c in range(8) for d in range(c + 1, 8))
   910	    cand = os.path.join(STATE["out"], label + ".SAT-CANDIDATE.json")
   911	    write_x(cand, {"host_clock": clock(), "array": A, "clauses_all_satisfied": sat_all,

   913	                   "column_pair_ordered_pairs_distinct": colpair_ok,
   914	                   "status": "CANDIDATE_UNVERIFIED - notify parent immediately"})
   915	    latch_path = os.path.join(LANE, "control", "SAT-CANDIDATE-" + label)
   916	    write_bytes_x(latch_path, cand.encode())
   917	    return {"candidate": cand, "latch": latch_path, "clauses_all_satisfied": sat_all, "cells": cells_ok,
   918	            "rowpair": rowpair_ok, "colpair": colpair_ok}

   921	def php_cnf(p, h):
   922	    """Pigeonhole: p pigeons into h holes; UNSAT iff p > h."""
   923	    var = lambda i, j: i * h + j + 1  # noqa: E731
   924	    cl = [[var(i, j) for j in range(h)] for i in range(p)]
   925	    for j in range(h):
   926	        for i in range(p):
   927	            for k in range(i + 1, p):
   928	                cl.append([-var(i, j), -var(k, j)])
   929	    return ("p cnf %d %d\n" % (p * h, len(cl)) + "".join(" ".join(map(str, c)) + " 0\n" for c in cl)).encode()

  1077	def commit_record(path, rec_unknown, rec_unsat):
  1078	    """Explicit commit boundary.  Both serializations and the O_EXCL fd exist before the boundary; the
  1079	    boundary is one atomic next(ORDER); a cancel stamped before it (or any latch) forces UNKNOWN."""
  1080	    unknown_bytes = serialize(rec_unknown)
  1081	    unsat_bytes = None if rec_unsat is None else serialize(rec_unsat)
  1082	    fd = os.open(path, os.O_WRONLY | os.O_CREAT | os.O_EXCL, 0o644)
  1083	    try:
  1084	        boundary = next(ORDER)
  1085	        co = STATE["cancel_order"]
  1086	        chosen = unsat_bytes if (unsat_bytes is not None and not LATCH and (co is None or co > boundary)) else unknown_bytes
  1087	        view = memoryview(chosen)
  1088	        while view:
  1089	            k = os.write(fd, view)
  1090	            view = view[k:]
  1091	        os.fsync(fd)
  1092	    finally:
  1093	        os.close(fd)
  1094	    return {"boundary": boundary, "cancel_order": STATE["cancel_order"], "committed_unsat": chosen is unsat_bytes,
  1095	            "sha256": hashlib.sha256(chosen).hexdigest()}
