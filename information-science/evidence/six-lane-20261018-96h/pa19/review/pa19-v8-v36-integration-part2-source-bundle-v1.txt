    def _close_endpoint(self, name, propagate=False):
        side = "read" if name == "rfd" else "write"
        closed_key, attempted_key = side + "_endpoint_closed", side + "_close_attempted"
        error_key = side + "_close_error"
        if getattr(self, attempted_key):
            prior = getattr(self, error_key)
            if prior is not None and propagate:
                raise RuntimeError(prior)         # uncertain descriptor may have been reused: never retry it
            return prior
        fd = getattr(self, name)
        if fd is None:
            if getattr(self, closed_key) is True:
                return None
            prior = getattr(self, error_key)
            if prior is None:
                prior = side + " endpoint identity unavailable before closure"
                setattr(self, error_key, prior)
                self.close_errors.append(prior)
                self.errors.append(prior)
            setattr(self, closed_key, None)
            if propagate:
                raise RuntimeError(prior)
            return prior
        setattr(self, attempted_key, True)
        setattr(self, side + "_close_fd", fd)
        try:
            os.close(fd)
        except BaseException as exc:
            message = "%s endpoint closure uncertain: %r" % (side, exc)
            setattr(self, closed_key, None)
            setattr(self, error_key, message)
            self.close_errors.append(message)
            self.errors.append(message)
            if propagate:
                raise
            return message
        else:
            setattr(self, closed_key, True)
            setattr(self, name, None)
            return None

def php_cnf(p, h):
    """Pigeonhole: p pigeons into h holes; UNSAT iff p > h."""
    var = lambda i, j: i * h + j + 1  # noqa: E731
    cl = [[var(i, j) for j in range(h)] for i in range(p)]
    for j in range(h):
        for i in range(p):
            for k in range(i + 1, p):
                cl.append([-var(i, j), -var(k, j)])
    return ("p cnf %d %d\n" % (p * h, len(cl)) + "".join(" ".join(map(str, c)) + " 0\n" for c in cl)).encode()
