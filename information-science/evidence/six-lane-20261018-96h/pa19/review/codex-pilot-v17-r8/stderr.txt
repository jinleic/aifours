   982	    cl = [[var(i, j) for j in range(h)] for i in range(p)]

   986	                cl.append([-var(i, j), -var(k, j)])

  1135	def commit_record(path, rec_unknown, rec_unsat):
  1136	    """Explicit commit boundary.  Both serializations and the O_EXCL fd exist before the boundary; the
  1137	    boundary is one atomic next(ORDER); a cancel stamped before it (or any latch) forces UNKNOWN."""
  1138	    unknown_bytes = serialize(rec_unknown)
  1139	    unsat_bytes = None if rec_unsat is None else serialize(rec_unsat)
  1140	    fd = os.open(path, os.O_WRONLY | os.O_CREAT | os.O_EXCL, 0o644)
  1141	    try:
  1142	        boundary = next(ORDER)
  1143	        co = STATE["cancel_order"]
  1144	        chosen = unsat_bytes if (unsat_bytes is not None and not LATCH and (co is None or co > boundary)) else unknown_bytes
  1145	        view = memoryview(chosen)
  1146	        while view:
  1147	            k = os.write(fd, view)
  1148	            view = view[k:]
  1149	        os.fsync(fd)
  1150	    finally:
  1151	        os.close(fd)
  1152	    return {"boundary": boundary, "cancel_order": STATE["cancel_order"], "committed_unsat": chosen is unsat_bytes,
  1153	            "sha256": hashlib.sha256(chosen).hexdigest()}
