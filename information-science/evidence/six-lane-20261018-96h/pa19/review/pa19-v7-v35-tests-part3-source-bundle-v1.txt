def d2_assert_sink_endpoints_closed(sink, file_fd, writer=None):
    """Independent OS checks of EVERY acquired endpoint, not merely the sink's settlement assertions. Each observed
    acquisition keeps an enumerable descriptor number; a number that is open again must name a DIFFERENT file
    (legitimate reuse), never the observed endpoint itself; an open number of unknown identity refuses."""
    assert sink.thread is not None and not sink.thread.is_alive(), sink.accounting()
    assert sink.file is not None and sink.file.closed is True, sink.accounting()
    assert sink.reader_fd_closed is True and sink.unblock_fd_closed is True and sink.rehash_fd_closed is True
    assert sink.flush_completed is True and sink.fsync_completed is True and sink.close_completed is True
    endpoints = [("file", file_fd, None if sink.file_identity is None
                  else {"dev": sink.file_identity[0], "ino": sink.file_identity[1]})]
    for side in ("reader", "unblock", "rehash"):
        number, observed = getattr(sink, side + "_fd"), getattr(sink, side + "_fd_observed_identity")
        if observed is not None:
            assert isinstance(observed, dict) and "unavailable" not in observed, (side, observed)
            assert type(number) is int and number >= 0, ("observed endpoint lost its descriptor number", side, number)
        endpoints.append((side, number, observed))
    if writer is not None:
        assert writer.get("closed") is True and "close_error" not in writer, writer
        endpoints.append(("writer", writer.get("fd"), writer.get("observed_identity")))
    checked = []
    for side, fd, observed in endpoints:
        if fd is None:
            continue
        try:
            st = os.fstat(fd)
        except OSError as exc:
            assert exc.errno == errno.EBADF, (side, fd, exc)
        else:
            if observed is None or [st.st_dev, st.st_ino] == [observed["dev"], observed["ino"]]:
                raise AssertionError("D2 fixture endpoint still open or unidentifiable: %s %d" % (side, fd))
        checked.append(fd)
    return sorted(set(checked))
