292|def output_room(additional=0, stream=None):
293|    """Conservative distinct-inode RUN bound, including outgoing parent captures.
294|
295|    4MiB remains for bounded child evidence and later parent/oracle records.
296|    This is a cooperative fixture limit, not a reservation-release accountant.
297|    """
298|    if RUN is None:
299|        return
300|    seen, entries, capture_sizes = set(), [0], {"stdout": 0, "stderr": 0}
301|    def walk(path, depth=0):
302|        if depth > 16:
303|            raise OSError("FIXTURE_OUTPUT_DEPTH")
304|        fd = directory(path)
305|        total = 0
306|        try:
307|            for name in os.listdir(fd):
308|                check("bounded entire current-run output census")
309|                entries[0] += 1
310|                if entries[0] > 4096:
311|                    raise OSError("FIXTURE_OUTPUT_ENTRY_CAP")
312|                value = os.stat(name, dir_fd=fd, follow_symlinks=False)
313|                child = path + "/" + name
314|                if stat.S_ISDIR(value.st_mode):
315|                    total += walk(child, depth + 1)
316|                elif stat.S_ISREG(value.st_mode):
317|                    key = (value.st_dev, value.st_ino)
318|                    if key in seen:
319|                        continue
320|                    seen.add(key)
321|                    if child in (RUN + "/driver.stdout", RUN + "/driver.stderr"):
322|                        capture_sizes[name.split(".")[-1]] = value.st_size
323|                    else:
324|                        total += value.st_size
325|                else:
326|                    raise OSError("FIXTURE_OUTPUT_UNDECLARED_NONREGULAR")
327|        finally:
328|            os.close(fd)
329|        return total
330|    total = walk(RUN)
331|    for name in EMITTED:
332|        total += max(capture_sizes[name], EMITTED[name] + (additional if stream == name else 0))
333|    if total + (additional if stream is None else 0) > 12 << 20:
334|        raise OSError("FIXTURE_OUTPUT_12M_PLUS_4M_PARENT_CHILD_RESERVE")
