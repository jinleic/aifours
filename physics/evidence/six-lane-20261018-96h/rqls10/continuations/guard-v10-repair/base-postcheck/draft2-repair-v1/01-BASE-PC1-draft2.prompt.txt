154|def snapshot(fdroot, name, cap=64 << 20, optional=False):
155|    deadline = time.monotonic() + 30
156|    def signature(s):
157|        return (s.st_dev, s.st_ino, s.st_mode, s.st_size, s.st_nlink, s.st_mtime_ns, s.st_ctime_ns)
158|    try:
159|        st = os.stat(name, dir_fd=fdroot, follow_symlinks=False)
160|    except FileNotFoundError:
161|        if optional:
162|            return None
163|        raise
164|    if not stat.S_ISREG(st.st_mode) or not 0 <= st.st_size <= cap:
165|        raise Invalid("not a bounded regular file: " + name)
166|    fd = os.open(name, os.O_RDONLY | os.O_NOFOLLOW | os.O_NONBLOCK, dir_fd=fdroot)
167|    try:
168|        s1 = os.fstat(fd)
169|        if signature(s1) != signature(st):
170|            raise Invalid("file changed at open")
171|        parts, n = [], 0
172|        while n < st.st_size:
173|            b = os.read(fd, min(1 << 20, st.st_size - n))
174|            if not b:
175|                raise Invalid("short read")
176|            parts.append(b)
177|            n += len(b)
178|            if time.monotonic() > deadline:
179|                raise Invalid("read deadline")
180|        tail, s2 = os.read(fd, 1), os.fstat(fd)
181|        named_end = os.stat(name, dir_fd=fdroot, follow_symlinks=False)
182|        if tail or signature(s2) != signature(st) or signature(named_end) != signature(st):
183|            raise Invalid("file changed during read")
184|    finally:
185|        os.close(fd)
186|    if time.monotonic() > deadline:
187|        raise Invalid("read completion deadline")
188|    raw = b"".join(parts)
189|    return {"raw": raw, "sha256": hashlib.sha256(raw).hexdigest(), "bytes": len(raw),
190|            "inode": [st.st_dev, st.st_ino], "nlink": st.st_nlink, "birth_epoch": st.st_birthtime}
