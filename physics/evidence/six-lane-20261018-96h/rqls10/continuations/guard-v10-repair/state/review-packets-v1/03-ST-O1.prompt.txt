488+|def validate_model(encoded, m, schemas):
489+|    """Required MODEL/step/operation/state fields must exist exactly before any predicate; unavailable evidence is HOLD."""
490+|    if type(encoded) is not dict or set(encoded) != set(schemas["model_fields"]) or set(m) != set(encoded):
491+|        raise Invalid("exact model field set")
492+|    if any(type(encoded.get(k)) is not dict or encoded[k].get("encoding") != "UTF8_HEX" for k in ("stdout", "stderr")):
493+|        raise Invalid("model stdout/stderr require their explicit UTF8_HEX encodings")
494+|    for item in encoded["steps"] if type(encoded["steps"]) is list else [None]:
495+|        error = item.get("exception") if type(item) is dict else None

497+|                                  error["text"].get("encoding") != "UTF8_HEX"):

502+|            type(m["inputs"]) is not dict or type(m["config"]) is not dict or type(m["clock_rules"]) is not list or \
