314|def read_file(path, cap, dl, dir_fd=None, keep=False):
315|    dl.check("before file open")
316|    st = os.stat(path, dir_fd=dir_fd, follow_symlinks=False)
317|    if not stat.S_ISREG(st.st_mode) or not 0 <= st.st_size <= cap:
318|        raise GuardError("NOT_BOUNDED_REGULAR %s" % path)

323|        if (fs.st_dev, fs.st_ino, fs.st_size) != (st.st_dev, st.st_ino, st.st_size):

452|def walk_fd(anchor, dev, dl, seen=None):

462|            for e in it:

634|def spawn_registered(s, argv, env, role, dl, required_s):
635|    """Spawn with an explicitly empty signal mask while cancellation is protected in the parent.
636|    Registration is the FIRST operation after posix_spawn returns a PID; no Popen finalizer or second reaper exists.
637|    Returned pipe descriptors remain owned even when post-spawn close/deadline work fails."""
638|    allocated = []

640|              "wait_lost": False, "term_attempted": False, "settled": False, "stdout_fd": None,

643|    if role == "helper":

647|        allocated.extend((out_read, out_write))
648|        if role == "helper":
649|            null = os.open("/dev/null", os.O_RDWR)
650|            allocated.append(null)

662|            raise GuardError("STANDARD_DESCRIPTORS_MUST_BE_OPEN")
663|        actions = [(os.POSIX_SPAWN_DUP2, in_read, 0), (os.POSIX_SPAWN_DUP2, out_write, 1),
664|                   (os.POSIX_SPAWN_DUP2, err_write, 2)]
665|        actions += [(os.POSIX_SPAWN_CLOSE, fd) for fd in allocated]

679|                                 setsigdef=tuple(CANCEL_SIGNALS | {signal.SIGPIPE, signal.SIGCHLD}))

682|                s.helper_calls += 1
683|            if role != "helper":
