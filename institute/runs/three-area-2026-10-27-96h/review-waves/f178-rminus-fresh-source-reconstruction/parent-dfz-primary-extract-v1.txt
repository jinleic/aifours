S1|334|\begin{enumerate}
S1|335|\item Start with a set of arbitrary random variables.
S1|336|\item Add auxiliary random variables

S1|338|\item Apply known information inequalities to the enlarged set of random variables.
S1|339|\end{enumerate}

S1|348|\begin{lemma}[Copy Lemma]\label{copyLemma}
S1|349|Let $A$, $B$, $C$, $D$ be jointly distributed random variables.  Then there is another random variable
S1|350|$R$, jointly distributed with $A,B,C,D$  with the following properties.
S1|351|
S1|352|\begin{description}
S1|353|\item [C1.] The marginal distributions of $(A,B,C)$ and $(A,B,R)$ are the same with $R$ replacing $C$.
S1|354|%\item [C2.] $(C,D) \to (A,B) \to R$ is a Markov chain.
S1|355|\item [C2.] $I(CD;R|AB)=0$
S1|356|\end{description}
S1|357|In this case we say that $R$ is a $D$-copy of $C$ over $(A,B)$.
S1|358|\end{lemma}
S1|359|\begin{proof}
S1|360|Let $A$, $B$, $C$, $D$, denote the alphabets of the random variables $A$, $B$, $C$, $D$ resp.
S1|361|Let $a,b,c,d$ denote arbitrary elements of $A,B,C,D$, resp.  with probability $p(a,b,c,d)$.
S1|362|Let $R$ be a new random variable and let $r$ denote an arbitrary element of its alphabet, which is $C$.
S1|363|Define the joint probability distribution of $A,B,C,D,R$ by
S1|364|$$
S1|365|p'(a,b,c,d,r) = \frac{p(a,b,c,d)\sum_d p(a,b,r,d)}{\sum_{c,d}p(a,b,c,d)}.
S1|366|$$
S1|367|It is clear that these are nonnegative.
S1|368|Summing over $r$ we get
S1|369|\begin{eqnarray*}
S1|370|\sum_r  p'(a,b,c,d,r) &=& \frac{p(a,b,c,d)\sum_{r,d} p(a,b,r,d)}{\sum_{c,d}p(a,b,c,d)}\\
S1|371|&=& p(a,b,c,d)
S1|372|\end{eqnarray*}
S1|373|so that $p'$ is an extension of $p$, which also implies that the sum of all of the probabilities $p'$ is 1.
S1|374|Similarly, the marginal distribution of $(A,B,R)$ is given by
S1|375|\begin{eqnarray*}
S1|376|\sum_{c,d}  p'(a,b,c,d,r) &=& \frac{\sum_{c,d} p(a,b,c,d)\sum_{d} p(a,b,r,d)}{\sum_{c,d}p(a,b,c,d)}\\
S1|377|&=& \sum_{d} p(a,b,r,d)
S1|378|\end{eqnarray*}
S1|379|while the marginal distribution of $(A,B,C)$ is given by $\sum_{d} p(a,b,c,d)$, demonstrating (C1).
S1|380|
S1|381|If we write (C2) in terms of entropies, we get
S1|382|$H(ABCD)+H(ABR)-H(AB)-H(ABCDR)=0$.  But $H(A,B,R)=H(A,B,C)$ by (C1), so it remains to show that
S1|383|$H(ABCDR)=H(ABCD)+H(ABC)-H(AB)$.  We compute $H(ABCDR)$ as
S1|384|\begin{eqnarray*}
S1|385|&=& \sum_{a,b,c,d,r} -p'(a,b,c,d,r)\log_2(p'(a,b,c,d,r))\\
S1|386|&=&  \sum_{a,b,c,d,r} -p'(a,b,c,d,r)\log_2(p(a,b,c,d))\\
S1|387|&+&  \sum_{a,b,c,d,r} -p'(a,b,c,d,r)\log_2(\sum_d p(a,b,r,d))\\
S1|388|&-&  \sum_{a,b,c,d,r} -p'(a,b,c,d,r)\log_2(\sum_{c,d} p(a,b,c,d)).\\
S1|389|\end{eqnarray*}
S1|390|But
S1|391|\begin{eqnarray*}
S1|392|&& \sum_{a,b,c,d,r} -p'(a,b,c,d,r)\log_2(p(a,b,c,d))\\
S1|393|&=&  \sum_{a,b,c,d} -p(a,b,c,d)\log_2(p(a,b,c,d))\\
S1|394|&=& H(ABCD)
S1|395|\end{eqnarray*}
S1|396|\begin{eqnarray*}
S1|397|&&\sum_{a,b,c,d,r} -p'(a,b,c,d,r)\log_2(\sum_d p(a,b,r,d))\\
S1|398|&=&  \sum_{a,b,r} \sum_{c,d}-p'(a,b,c,d,r)\log_2(\sum_d p(a,b,r,d))\\
S1|399|&=&  \sum_{a,b,r} \sum_d -p(a,b,r,d)\log_2(\sum_d p(a,b,r,d))\\
S1|400|&=& H(ABC)
S1|401|\end{eqnarray*}
S1|402|\begin{eqnarray*}
S1|403|&&  \sum_{a,b,c,d,r}-p'(a,b,c,d,r)\log_2(\sum_{c,d} p(a,b,c,d))\\
S1|404|&=&  \sum_{a,b,} \sum_{c,d,r}-p'(a,b,c,d,r)\log_2(\sum_{c,d} p(a,b,c,d))\\
S1|405|&-&  \sum_{a,b,} \sum_{d,r}-p(a,b,r,d)\log_2(\sum_{c,d} p(a,b,c,d))\\
S1|406|&=& H(AB)
S1|407|\end{eqnarray*}
S1|408|Therefore, $H(ABCDR)=H(ABCD)+H(ABC)-H(AB)$ as desired.
S1|409|\end{proof}
