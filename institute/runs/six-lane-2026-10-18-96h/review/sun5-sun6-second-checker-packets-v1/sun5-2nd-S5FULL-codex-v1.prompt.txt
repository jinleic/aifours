164|    return all(seq[i] < seq[i + 1] for i in range(len(seq) - 1))

167|def parse_key(raw):
168|    """KEY = ten alpha digits (edge order EDGES) '/' E2 E3; '-' = no such vertex."""
169|    m = KEY_RE.fullmatch(raw)
170|    if m is None:
171|        return None
172|    digits = tuple(c - 48 for c in m.group(1))
173|    m2i = MARKER_SYMBOLS.index(chr(m.group(2)[0]))
174|    m3i = MARKER_SYMBOLS.index(chr(m.group(3)[0]))
175|    al = [[0] * NV for _ in VERTICES]
176|    a1 = 0
177|    a2 = 0
178|    for k, (i, j) in enumerate(EDGES):
179|        d = digits[k]
180|        al[i][j] = d
181|        al[j][i] = d
182|        if d == 1:
183|            a1 |= 1 << k
184|        elif d == 2:
185|            a2 |= 1 << k
186|    e2 = None if m2i == 0 else m2i - 1
187|    e3 = None if m3i == 0 else m3i - 1
188|    return {"al": al, "a1": a1, "a2": a2, "m2i": m2i, "m3i": m3i, "e2": e2, "e3": e3}

191|def parse_witness(rule, raw):
192|    """Return (witness, None) or (None, reason).  Grammar per rule, no json module."""
193|    m = WITNESS_RE[rule].fullmatch(raw)
194|    if m is None:
195|        return None, "PARSE_WITNESS_SYNTAX"
196|    g = m.groups()
197|    if rule == "T":
198|        tri = (int(g[0]), int(g[1]), int(g[2]))
199|        if not strictly_increasing(tri):
200|            return None, "PARSE_INDEX_ORDER"
201|        return (tri, int(g[3]), int(g[4])), None
202|    if rule == "K" or rule == "PB":
203|        v = int(g[0])
204|        s = tuple(int(c) for c in g[1].split(b","))
205|        if not strictly_increasing(s):
206|            return None, "PARSE_INDEX_ORDER"
207|        if v in s:
208|            return None, "PARSE_VERTEX_IN_SET"
209|        num = int(g[2])
210|        if g[3] is None:
211|            den = 1
212|        else:
213|            den = int(g[3])
214|            if den <= 1 or math.gcd(num, den) != 1:
215|                return None, "PARSE_NONCANONICAL_RATIONAL"
216|        return (v, s, Fraction(num, den)), None
217|    if rule == "FH2":
218|        a = int(g[0])
219|        b = int(g[1])
220|        v2 = tuple(int(c) for c in g[2].split(b","))
221|        if not a < b or not strictly_increasing(v2):
222|            return None, "PARSE_INDEX_ORDER"
223|        if a in v2 or b in v2:
224|            return None, "PARSE_VERTEX_IN_SET"
225|        return ((a, b), v2, int(g[3]), int(g[4])), None
226|    return (), None  # N2 '[]' and SURVIVE '-'

262|def parse_file(state, data, guard):
263|    """Account for every byte: file-level defects plus one Rec per line."""
264|    if len(data) == 0:
265|        state.file_reasons.append("FILE_EMPTY")
266|        return
267|    bad = [i for i, c in enumerate(data) if not (c == 0x09 or c == 0x0a or 0x20 <= c <= 0x7e)]
268|    if bad:
269|        state.file_reasons.append("FILE_FORBIDDEN_BYTE:count=%d,first_offset=%d" % (len(bad), bad[0]))
270|    pieces = data.split(b"\n")
271|    truncated = data[-1:] != b"\n"
272|    if truncated:
273|        state.file_reasons.append("FILE_NO_FINAL_NEWLINE")
274|    else:
275|        pieces.pop()  # the empty string after the final newline
276|    state.structure["input_lines"] = len(pieces)
277|    first_line_of_key = {}
278|    for n, line in enumerate(pieces):
279|        if (n & 255) == 0:
280|            guard.poll("parse")
281|        fields = line.split(b"\t")
282|        rec = Rec(n + 1, fields[0], fields[1] if len(fields) > 1 else b"")
283|        state.recs.append(rec)
284|        if truncated and n == len(pieces) - 1:
285|            rec.fail("PARSE_TRUNCATED_FINAL_LINE")
286|        if line == b"":
287|            rec.fail("PARSE_EMPTY_LINE")
288|            continue
289|        if len(fields) != 3:
290|            rec.fail("PARSE_FIELD_COUNT")
291|            continue
292|        key = parse_key(fields[0])
293|        if key is None:
294|            rec.fail("PARSE_KEY")
295|        else:
296|            rec.key = key
297|            if fields[0] in first_line_of_key:
298|                rec.fail("PARSE_DUPLICATE_KEY:first_line=%d" % first_line_of_key[fields[0]])
299|            else:
300|                first_line_of_key[fields[0]] = rec.line
301|        rule = RULE_NAMES.get(fields[1])
302|        if rule is None:
303|            rec.fail("PARSE_UNKNOWN_RULE")
304|            continue
305|        rec.rule = rule
306|        w, why = parse_witness(rule, fields[2])
307|        if why is not None:
308|            rec.fail(why)
309|        else:
310|            rec.w = w

316|def build_group():
317|    """Closure of {(0 1), (0 1 2 3 4)} under composition.  In a finite group the
318|    closure under multiplication by generators is the generated subgroup; 120
319|    distinct bijections of a 5-set is all of S5."""
320|    identity = tuple(VERTICES)
321|    generators = ((1, 0, 2, 3, 4), (1, 2, 3, 4, 0))
322|    seen = set([identity])
323|    elements = [identity]
324|    frontier = [identity]
325|    while frontier:
326|        nxt = []
327|        for g in frontier:
328|            for s in generators:
329|                h = tuple(s[g[i]] for i in VERTICES)  # h = s o g
330|                if h not in seen:
331|                    seen.add(h)
332|                    elements.append(h)
333|                    nxt.append(h)
334|        frontier = nxt
335|    ok = len(elements) == GROUP_ORDER and all(sorted(p) == list(VERTICES) for p in elements)
336|    return elements, ok

360|def build_universe(guard):
361|    """All colourings c: E -> {1,2,3} as disjoint mask pairs (A1, A2) = (c^-1(1),
362|    c^-1(2)): A1 runs over every subset of E and A2 over every subset of E \\ A1.
363|    Returns the direct-address rank table, the rank-ordered list, and whether
364|    every colouring appeared exactly once with the definitional count 3^10."""
365|    rank = array("i", [-1]) * (1 << (2 * NE))
366|    colourings = []
367|    repeated = False
368|    for a1 in range(1 << NE):
369|        if (a1 & 63) == 0:
370|            guard.poll("universe")
371|        rest = EDGE_MASK_ALL ^ a1
372|        a2 = rest
373|        while True:
374|            slot = (a2 << NE) | a1
375|            if rank[slot] != -1:
376|                repeated = True
377|            else:
378|                rank[slot] = len(colourings)
379|                colourings.append((a1, a2))
380|            if a2 == 0:
381|                break
382|            a2 = (a2 - 1) & rest

503|def pb_value(al, e2, e3, v, s):
504|    """Left side of (P5) at |X_j cap X_k| = hi_jk, in units of |G| (rules-v1 L4)."""
505|    x = {}
506|    for j in s:
507|        x[j] = Fraction(al[v][j], 4)  # (P1) |X_j| = alpha_vj |G| / 4
508|    value = Fraction(0)
509|    for j in s:
510|        value += 1 - x[j]
511|    his = {}
512|    for j, k in combinations(s, 2):
513|        u = Fraction(1, 4 * gcd_alpha_r(al[j][k], v, e2, e3))  # (P2) unit
514|        low = min(x[j], x[k])
515|        hi = (low // u) * u  # largest multiple of u <= min(|X_j|, |X_k|)
516|        if hi == low:
517|            # (P4): equality would force X_j <= X_k (or the reverse, or both);
518|            # K2 for U_vU_j <= U_vU_k requires alpha_vj + alpha_jk <= 3.
519|            if x[j] < x[k]:
520|                barred = al[v][j] + al[j][k] >= 4
521|            elif x[k] < x[j]:
522|                barred = al[v][k] + al[j][k] >= 4
523|            else:
524|                barred = (al[v][j] + al[j][k] >= 4) or (al[v][k] + al[j][k] >= 4)
525|            if barred:
526|                hi = hi - u
527|        his["%d%d" % (j, k)] = str(hi)
528|        value -= 1 - x[j] - x[k] + hi
529|    return value, his

560|def lambda_exponents(l, t, m, e2, e3):
561|    """(v2, v3) of lambda_l = 4 r_l t_l / m (rules-v1 L6 F1), as the supremum over
562|    every admissible r with this marker status: v2(r_l) = 0 unless l = E2, where
563|    it is any value >= 1 (None = unbounded); likewise v3 with E3.  Primes >= 5
564|    are never common to two lambdas (t <= 4, r pairwise coprime)."""

566|    base3 = p_adic(t, 3)

568|        raise InternalError("negative 2-adic exponent")

588|def fh2_rejected(asg, al, e2, e3, m):
589|    """asg[l] = (t_l, met-coset bitmask).  Rejected iff (F3) fails at a vertex with
590|    t = 1, or some coset with s >= 2 members has every pairwise gcd < s (F2)."""
591|    for i in VERTICES:

594|                # (F3): alpha_il / 4 <= t_l / m

596|                    return True

601|            if all(lambda_gcd_sup(l, asg[l][0], k, asg[k][0], m, e2, e3) < s
602|                   for l, k in combinations(members, 2)):

## 2. Independent representation (deliberately not a canonical-form cube)

- Colourings as edge-colour masks: (A1, A2) 10-bit masks of the α=1 and α=2 edges (A3 = complement). Markers as one-hot 5-bit masks (0 = none).
- Universe constructed by subset decomposition: A1 over all subsets of E, A2 over all submasks of E∖A1. Each colouring gets a sequential rank; a direct-address table maps (A1,A2) to rank and detects any repeat.
- S5 constructed as the closure of the generators (0 1) and (0 1 2 3 4) under composition; the checker verifies 120 distinct bijections (hence all of S5). Induced edge permutations act on masks via precomputed 1024-entry tables.
- Orbits by forward expansion from each record (apply all 120 group elements), with orbit-stabiliser consistency |orbit|·|stab| = 120 checked per record.
- Coverage: an owner array over the dense index rank·36 + marker index. Every labelled triple must be owned by exactly one record: duplicate keys, overlapping orbits, missing triples and out-of-universe images are separate failure reasons.
- Independent cross-check (not load-bearing): Burnside count (1/120)·Σ_π 3^{edge cycles}·(1+fixed vertices)^2 must equal the number of records accepted as distinct orbits.
- No lexicographic canonical form, no canonical-key comparison, no old counts.

## Theorem

Let G be any group. Let a_1G_1, …, a_5G_5 be pairwise disjoint left cosets of subgroups of finite index. Then gcd([G:G_i], [G:G_j]) ≥ 5 for some i < j.

## 1. R0: reduction to finite groups

core_G(G_i) is the kernel of the action of G on G/G_i, so its index is at most [G:G_i]!. Hence N = ∩_i core_G(G_i) is a normal subgroup of finite index contained in every G_i.

Each coset a_iG_i is the full preimage of its image in G/N. So indices, intersection indices and disjointness all pass to the finite group G/N.

## 2. R1: reduction to Core(5), and a proof of M-S Prop 4.5

**Core(5).** No finite group G has subgroups U_1, …, U_5 with [G:U_i] = 4r_i, the r_i pairwise coprime, and pairwise disjoint cosets g_iU_i.

**The Sun(5) ⇒ Core(5) direction.** A Core(5) configuration has every pairwise gcd equal to 4 < 5.

**The Core(5) ⇒ Sun(5) direction, group by group.** Let n_i = [G:G_i], and assume five disjoint cosets in G have all pairwise gcds ≤ 4.

1. **Each gcd is 2, 3 or 4.** By L0.2 and L0.6 no gcd equals 1, so each lies in {2, 3, 4}.
2. **All divisible by 3, or all even.** If some n_i is odd, every gcd with it is odd and so equals 3. Then 3 divides all five indices.
3. **All divisible by 3.** Every gcd is then 3 < 4. Any four of the cosets contradict Zhu Theorem 1.1 at k = 4.
4. **All even.** Write n_i = 2^{e_i}m_i with m_i odd. Then the m_i are pairwise coprime, every e_i ≥ 1, and at most one e_i ≥ 3.
   - **Two indices with e = 1.** With any third index they have all pairwise gcds equal to 2 < 3, contradicting Zhu Theorem 1.1 at k = 3.
   - **Exactly one index with e = 1.** We get (2r_1, 4r_2, 4r_3, 4r_4) with r_1 odd and the r's pairwise coprime. This is excluded by Proposition 4.5 below.
   - **Otherwise.** n_i = 4r_i with r_i = 2^{e_i−2}m_i pairwise coprime: a Core(5) configuration in the same group G, with the same subgroups.

**Proposition 4.5 (M-S, Beitr. Algebra Geom. 60 (2019); proof patched per F-2 and F-8).**

*Status of this proof.* It is self-contained modulo rules-v1 L0.3–L0.5 and L1–L3. "Harmonic" means: there exist g_i such that the cosets g_iU_i are pairwise disjoint (F-9).

*Statement.* Let r_1, …, r_4 be pairwise coprime with r_1 odd. Then no group has subgroups with indices [G:U_1] = 2r_1 and [G:U_i] = 4r_i (i = 2, 3, 4) and pairwise disjoint cosets g_iU_i.

*Proof.* By R0 we may take G finite. Sizes are in units of |G|, and α(i,j) := [G:U_i∩U_j] / lcm(n_i, n_j). Lemma K (rules-v1 L3) at a vertex v with set S requires: for each pair {i, j} ⊆ S, either U_vU_i ⊆ U_jU_i or U_vU_j ⊆ U_iU_j. Its conclusion is Σ_{i∈S} |U_iU_v| < 1.

- **(a) α(1,i) = 1 and |U_1U_i| = 1/2 for i ≥ 2.** gcd(2r_1, 4r_i) = 2, since r_1 is odd and coprime to r_i. So α ≤ 1 by L0.3.
- **(b) α(i,j) ≠ 2 for i, j ≥ 2.** Suppose α(2,3) = 2, so [G:U_2∩U_3] = 8r_2r_3.

  - So |U_1(U_2∩U_3)| = T/(2r_1·8r_2r_3) ≥ 1/2 = |U_1U_2| = |U_1U_3|. Hence U_1U_2 = U_1(U_2∩U_3) = U_1U_3.
  - **Inverting** gives U_2U_1 = U_3U_1. In particular U_2U_1 ⊆ U_3U_1, which is L3's first alternative at v = 2 for the pair {1, 3}. This is K2 with (v, j, k) = (1, 2, 3).
  - Lemma K at v = 2 with S = {1, 3} then requires |U_1U_2| + |U_3U_2| < 1, but the sum is 1/2 + 1/2 = 1. Contradiction.
- **(c) If α(2,3) = 3, then 3 | r_1 and |U_2U_1 ∩ U_2U_3| = 1/4.** (F-2 patch.)
  - T = [G:U_1∩U_2∩U_3] is divisible by lcm(2r_1, 4r_2, 4r_3) = 4r_1r_2r_3. Write T = 4r_1r_2r_3τ.

  - Its size is not 1/2. Otherwise U_2U_1 ⊆ U_2U_3. **Inverting** gives U_1U_2 ⊆ U_3U_2, which is L3's first alternative at v = 1 for the pair {2, 3} (K2 with (v, j, k) = (2, 1, 3)). Lemma K at v = 1 with S = {2, 3} would then give |U_2U_1| + |U_3U_1| = 1, a contradiction. Hence the size is exactly 1/4.
- **(d) Not all of α(2,3), α(2,4), α(3,4) equal 1.** Suppose they all do.
  - In the triangle {2, 3, 4}, (T2) gives τ ≤ 1, so τ = 1. Then |U_2(U_4∩U_3)| = 1/4 = |U_2U_4| (T3), so U_2U_4 ⊆ U_2U_3. **Inverting** gives U_4U_2 ⊆ U_3U_2. Likewise U_4U_3 ⊆ U_2U_3.
  - In the triangle {1, 2, 4}, T is a multiple of lcm(2r_1, 4r_2, 4r_4) = 4r_1r_2r_4. So |U_1(U_4∩U_2)| = T/(2r_1·4r_2r_4) ≥ 1/2 = |U_1U_4|, giving U_1U_4 ⊆ U_1U_2. **Inverting** gives U_4U_1 ⊆ U_2U_1. Likewise U_4U_1 ⊆ U_3U_1.
  - So for every pair {i, j} ⊆ S = {1, 2, 3}, the inclusion U_4U_i ⊆ U_jU_i holds for at least one ordering of the pair. That is L3's pair condition at v = 4.
  - Lemma K requires 1/2 + 1/4 + 1/4 < 1, which fails.
- **(e) Choosing labels.** By (b) and (d), together with α ≤ 3 from L0.3, some α(i,j) with i, j ≥ 2 equals 3. Relabel so that α(2,3) = 3. By (c), 3 | r_1.
  - α(2,4) and α(3,4) cannot both be 1. Otherwise the triangle {2, 3, 4} has τ ≤ α(2,4)α(3,4) = 1, and then 12r_2r_3 | T = 4r_2r_3r_4 gives 3 | r_4. That contradicts 3 | r_1.
  - By (b) neither of them equals 2. So, after swapping 2 and 3 if necessary, α(2,4) = 3.
- **(f) α(3,4) = 3.** By (b), α(3,4) ≠ 2. Suppose α(3,4) = 1.
  - In the triangle {2, 3, 4}, write T = 4r_2r_3r_4τ. Then 12r_2r_3 | T and 12r_2r_4 | T. Since 3 ∤ r_3r_4 (because 3 | r_1), we get 3 | τ.
  - So |U_3(U_2∩U_4)| = τ/(4·3) ≥ 1/4 = |U_3U_4|, giving U_3U_4 ⊆ U_3U_2. **Inverting** gives U_4U_3 ⊆ U_2U_3, which is L3's second alternative at v = 4 for the pair {2, 3} (K2 with (v, j, k) = (3, 4, 2)).
  - Lemma K at v = 4 with S = {2, 3} requires 3/4 + 1/4 < 1, which fails.
- **(g) Final count.** Now α(1,i) = 1 and α(i,j) = 3 for all i, j ∈ {2, 3, 4}, and 3 | r_1.
  - By (c), applied to the pairs (2,3) and (2,4): |U_2U_1 ∪ U_2U_j| = 1/2 + 3/4 − 1/4 = 1 for j = 3, 4.
  - |U_2U_3 ∩ U_2U_4| is a multiple of 1/gcd(4r_2, 12r_3r_4) = 1/4 (by L0.5 + L0.4, using 3 ∤ r_2). This is F-8.
  - That size is at least 3/4 + 3/4 − 1 = 1/2. It is not 3/4: otherwise U_2U_3 = U_2U_4, and K2 with (v, j, k) = (2, 3, 4) — that is, Lemma K at v = 3 with S = {2, 4}, using the inverse U_3U_2 ⊆ U_4U_2 — would give 3/4 + 3/4 < 1. So the size is 1/2, and U_2U_3 ∪ U_2U_4 = G.
  - Put 𝒜 = U_2U_1, ℬ = U_2U_3, 𝒞 = U_2U_4. All three pairwise unions are G. So G∖ℬ and G∖𝒞 are disjoint subsets of 𝒜, and 𝒜 ∩ ℬ ∩ 𝒞 = 𝒜 ∖ ((G∖ℬ) ⊔ (G∖𝒞)) (M-S Lemma 3.1, re-proved here). Hence
    |𝒜 ∩ ℬ ∩ 𝒞| = 1/2 − 1/4 − 1/4 = 0.
  - But U_2 ⊆ 𝒜 ∩ ℬ ∩ 𝒞, a contradiction. ∎

**F-3.** The Core(5) ⇒ Sun(5) direction holds group by group. So the minimal-order G over the full target carries a Core(5) configuration.

## 5. Lemma R: right-coset fibres over a vertex subgroup [NEW]

**Setting.** G is finite. U_1, …, U_5 ≤ G have indices n_i. **Assume the cosets g_1U_1, …, g_5U_5 are pairwise disjoint (F-9).** Fix a vertex v and left-translate the whole configuration so that g_v = 1; disjointness is preserved. For j ≠ v put D_j := U_v g_j U_j, a (U_v, U_j)-double coset, and d_j := |D_j|/|U_j|.

**Statement.**
- **(R1)** D_j ∩ U_vU_j = ∅.
- **(R2)** |D_j| is a positive multiple of |G|/gcd(n_v, n_j).
- **(R3)** Every right coset U_vy contained in D_j meets g_jU_j. If w ∈ g_jU_j ∩ U_vy, then g_jU_j ∩ U_vy = (U_v ∩ wU_jw⁻¹)w, and [U_v : U_v ∩ wU_jw⁻¹] = d_j.
- **(R5′)** If j ≠ k and D_j ∩ D_k ≠ ∅, then gcd(d_j, d_k) ≥ 2.
- (R4) from earlier versions is deleted, since it is unused.

**Proof of (R1).** Since g_vU_v = U_v and g_jU_j are disjoint, L0.2 gives g_j ∉ U_vU_j. Now U_vU_j = U_v·1·U_j is the double coset of 1, and D_j is the double coset of g_j. Distinct double cosets are disjoint, so D_j ∩ U_vU_j = ∅.

**Proof of (R2).** By L0.4, |U_v g_j U_j| is a multiple of |G|/gcd([G:U_v], [G:U_j]). It is positive because g_j ∈ D_j.

**Proof of (R3).**
- Let U_vy ⊆ D_j. Then y = c g_j u with c ∈ U_v and u ∈ U_j. So U_vy = U_v g_j u contains g_ju, which lies in g_jU_j.
- For w ∈ g_jU_j ∩ U_vy we have g_jU_j = wU_j and U_vy = U_vw. Then x ∈ wU_j ∩ U_vw ⟺ xw⁻¹ ∈ wU_jw⁻¹ and xw⁻¹ ∈ U_v. So g_jU_j ∩ U_vy = (U_v ∩ wU_jw⁻¹)w.
- For the index: [U_v : U_v ∩ wU_jw⁻¹] = |U_v · wU_jw⁻¹| / |wU_jw⁻¹|. Right multiplication by w preserves size, so |U_v · wU_jw⁻¹| = |U_vwU_j|. Since w ∈ D_j, U_vwU_j = D_j. Hence the index is |D_j|/|U_j| = d_j.

**Proof of (R5′).**
- D_j and D_k are both left U_v-invariant, so D_j ∩ D_k is too. Being non-empty, it contains a whole right coset C = U_vy.
- By (R3), choose w_j ∈ g_jU_j ∩ C and w_k ∈ g_kU_k ∩ C. The two pieces are P_j = g_jU_j ∩ C = A w_j and P_k = g_kU_k ∩ C = B w_k, where A = U_v ∩ w_jU_jw_j⁻¹ and B = U_v ∩ w_kU_kw_k⁻¹. Their indices in U_v are d_j and d_k.
- P_j ∩ P_k = ∅, because g_jU_j ∩ g_kU_k = ∅.
- **Right-translate into U_v.** Multiply both pieces on the right by w_j⁻¹. This gives A and Bc, where c = w_kw_j⁻¹ lies in U_v because w_j and w_k lie in the same right coset C. So A and Bc are disjoint subsets of U_v.
- **Invert.** A⁻¹ = A and (Bc)⁻¹ = c⁻¹B. These are disjoint left cosets in U_v of subgroups of indices d_j and d_k.
- **Apply k = 2 (L0.6) in U_v.** If gcd(d_j, d_k) = 1, then AB = U_v, so c⁻¹ ∈ AB, i.e. c⁻¹ = ab with a ∈ A and b ∈ B. Then a = c⁻¹b⁻¹ ∈ A ∩ c⁻¹B, a contradiction.
- Hence gcd(d_j, d_k) ≥ 2. ∎

**Core(5) form.** In Core(5), gcd(n_v, n_j) = 4. So |D_j| = q_j |G|/4 with q_j a positive integer. By (R1), D_j ⊆ G ∖ U_vU_j, which has size (4 − α_vj)|G|/4, so 1 ≤ q_j ≤ 4 − α_vj. Then d_j = q_j|G|/4 ÷ (|G|/(4r_j)) = q_jr_j.

## 6. Refutation of the pattern S-D

**S-D:**
- α01 = α02 = 1, α03 = α04 = 2, and α12 = α13 = α14 = α23 = α24 = α34 = 3;
- E2 = 1, so r_1 is even and 3 ∤ r_1;
- E3 = 0, so 3 | r_0 and r_0 is odd;
- r_2, r_3, r_4 are coprime to 6.

Assume a Core(5) configuration with this pattern. Take v = 2, translate so that g_2 = 1, and put X_j = U_2U_j for j ∈ {0, 1, 3, 4}.

**Step 1: sizes, and D_j = Y_j for j ∈ {1, 3, 4}.**
- By L0.3, |X_0| = α_20|G|/4 = |G|/4 and |X_1| = |X_3| = |X_4| = 3|G|/4.
- For j ∈ {1, 3, 4} put Y_j = G ∖ X_j, so |Y_j| = |G|/4.
- By (R1), D_j ⊆ Y_j. By (R2), |D_j| is a positive multiple of |G|/4.
- Hence D_j = Y_j, q_j = 1 and d_j = r_j.

**Step 2: Y_1, Y_3, Y_4 are pairwise disjoint.** If Y_j ∩ Y_k ≠ ∅ for distinct j, k ∈ {1, 3, 4}, then D_j ∩ D_k ≠ ∅. By (R5′), gcd(r_j, r_k) ≥ 2, contradicting coprimality.

**Step 3: X_0 ⊆ X_1 ∩ X_3 ∩ X_4.** Use rules-v1 L1 in the triangles {0, 1, 2}, {0, 2, 3} and {0, 2, 4}.
- In {0, 1, 2}: by (T2), τ ≤ α01·α02 = 1, so τ_012 = 1.
- In {0, 2, 3}: (T1) gives α03 = 2 | r_2 τ, and r_2 is odd, so τ is even. (T2) gives τ ≤ α02·α03 = 2. Hence τ_023 = 2. The same argument gives τ_024 = 2.
- By (T3) with apex 2:
  - |U_2(U_0∩U_1)| = τ_012|G|/(4α01) = |G|/4;
  - |U_2(U_0∩U_3)| = τ_023|G|/(4α03) = 2|G|/8 = |G|/4;
  - |U_2(U_0∩U_4)| = |G|/4.
- Each of these sets lies inside X_0 = U_2U_0, which has size |G|/4. So X_0 = U_2(U_0∩U_j) ⊆ U_2U_j = X_j for j = 1, 3, 4.

**Step 4: G ∖ X_0 = Y_1 ⊔ Y_3 ⊔ Y_4.**
- By Step 3, X_0 ∩ Y_j = ∅ for j = 1, 3, 4.
- By Step 2, |Y_1 ⊔ Y_3 ⊔ Y_4| = 3|G|/4 = |G ∖ X_0|.

**Step 5: the coset g_0U_0.**
- **Location.** By (R1), D_0 ⊆ G ∖ X_0. By (R2), |D_0| = q_0|G|/4 with q_0 ∈ {1, 2, 3}, and d_0 = q_0r_0.
- **D_0 meets some Y_j.** D_0 is non-empty, so by Step 4 it meets some Y_j = D_j with j ∈ {1, 3, 4}.
- **Gcd condition.** (R5′) gives gcd(q_0r_0, r_j) ≥ 2. Since gcd(r_0, r_j) = 1, gcd(q_0r_0, r_j) = gcd(q_0, r_j).
  - For j = 3, 4 the number r_j is coprime to 6, so gcd(q_0, r_j) = 1 for q_0 ∈ {1, 2, 3}. This is impossible.
  - So D_0 does not meet Y_3 or Y_4, and therefore D_0 ⊆ Y_1.
- **Contradiction (q_0 = 1 route).** Because |Y_1| = |G|/4, D_0 ⊆ Y_1 forces q_0 = 1. Then gcd(q_0, r_1) = gcd(1, r_1) = 1, which contradicts (R5′) for the pair {0, 1}.
- **Equivalent route (proof-v1).** gcd(q_0, r_1) ≥ 2 with r_1 even and 3 ∤ r_1 forces q_0 = 2, and then |D_0| = |G|/2 > |Y_1|.

So no configuration has the pattern S-D. ∎

Suppose Sun(5) fails. By §3, a minimal-order group G carries a Core(5) configuration. Its pattern (α, E2, E3) lies, up to relabelling, in one of the 21,150 orbits.

## 0. Setting and notation

**Harmonic configuration.** G is a finite group. U_1, ..., U_k ≤ G are subgroups with n_i = [G:U_i], and g_1, ..., g_k ∈ G are elements such that the cosets g_iU_i are pairwise disjoint. Finiteness of G is justified by R0 in the README.

**Pairwise data.** For each pair i ≠ j put
- g_ij = gcd(n_i, n_j);
- I_ij = [G : U_i ∩ U_j];
- α_ij = I_ij / lcm(n_i, n_j). This is an integer, because n_i and n_j both divide I_ij.

**The Core(5) setting.**
- k = 5 and n_i = 4 r_i, where the r_i are pairwise coprime positive integers.
- E2 is the unique vertex with r_i even, if there is one. E3 is the unique vertex with 3 | r_i, if there is one. E2 = E3 is allowed.
- Consequently g_ij = 4, lcm(n_i, n_j) = 4 r_i r_j and I_ij = 4 r_i r_j α_ij.

**Minimal counterexample (used in F2).** Suppose Core(5) fails. Fix a counterexample in a group G of minimal order among all finite groups that contain five pairwise disjoint cosets of subgroups whose pairwise index-gcds are all ≤ 4. Then every proper subgroup of G satisfies Sun's statement for k = 5.

## L0. Basic facts

**L0.1 [standard] Product formula.** For subgroups X, Y: |XY| = |X||Y| / |X ∩ Y|.

**L0.2 [PUB, M-S Lemma 3.2(3)].** gU ∩ hV = ∅ if and only if g⁻¹h ∉ UV.

*Proof.* gu = hv holds if and only if g⁻¹h = u v⁻¹ ∈ UV.

**L0.3.** In a harmonic configuration, |U_iU_j| = α_ij |G| / g_ij and 1 ≤ α_ij ≤ g_ij − 1. In Core(5) this reads α_ij ∈ {1, 2, 3} and |U_iU_j| = α_ij |G| / 4.

*Proof.*
- By L0.1, |U_iU_j| = |G| I_ij / (n_i n_j) = |G| lcm(n_i, n_j) α_ij / (n_i n_j) = α_ij |G| / g_ij.
- By L0.2 and disjointness, g_i⁻¹g_j ∉ U_iU_j. So U_iU_j ≠ G, which gives α_ij < g_ij.

**L0.4 [PUB, M-S Lemma 3.4(1)].** For subgroups A, B and any x ∈ G, |AxB| is a multiple of |G| / gcd([G:A], [G:B]).

*Proof.*
- |AxB| = |A||B| / |A ∩ xBx⁻¹| = |G| [G : A ∩ xBx⁻¹] / ([G:A][G:B]).
- [G : A ∩ xBx⁻¹] is a multiple of lcm([G:A], [G:B]).
- Substituting gives |AxB| = (integer) · |G| / gcd([G:A], [G:B]).

**L0.5 [PUB, mirror of M-S Lemma 3.4(2)].** For subgroups A, B, C, the set AB ∩ AC is a union of (A, B∩C)-double cosets.

*Proof.* Let x = ab = a'c, with y ∈ A and w ∈ B∩C. Then yxw = (ya)(bw) = (ya')(cw), which lies in AB ∩ AC.

**L0.6 [PUB, Sun 2006 Lemma 2.1 / M-S Lemma 2.2].** If [G:A] and [G:B] are coprime, then AB = G, so every left coset of A meets every left coset of B.

## L1. Triangle lemma T [NEW form; subsumes M-S Lemma 3.10(a),(c) and Zhu (3.1), (3.4)]

Fix distinct vertices a, b, c. Put L = lcm(n_a, n_b, n_c) and T = [G : U_a∩U_b∩U_c].

**Statement.** L divides T. The quotient τ = T/L is a positive integer satisfying:

- **(T1)** I_xy divides L·τ for each pair {x, y} ⊆ {a, b, c}.
- **(T3)** For every labelling {x, y, z} = {a, b, c}:
  |U_y(U_x∩U_z)| = |G| · L · τ / (n_y · I_xz).
- **(T2)** For every labelling, |U_y(U_x∩U_z)| ≤ min(|U_yU_x|, |U_yU_z|).

*Proof.*
- The triple intersection lies in each U_x and in each U_x∩U_y. Hence every n_x divides T, so L | T. Likewise every I_xy divides T = Lτ, which is (T1).
- (T3) is L0.1 applied with |U_y| = |G|/n_y, |U_x∩U_z| = |G|/I_xz and |U_x∩U_y∩U_z| = |G|/T.
- (T2) holds because U_y(U_x∩U_z) ⊆ U_yU_x ∩ U_yU_z.

**Definitions.**
- τ_min is the least positive τ satisfying (T1). The set of solutions of (T1) is exactly the set of multiples of τ_min, since I_xy | Lτ ⟺ (I_xy / gcd(I_xy, L)) | τ.
- τ_max is the largest τ satisfying (T2) at all three apexes.

**T-kill.** If τ_min > τ_max, no harmonic configuration has these data.

**Core(5) form.**
- L = 4 r_a r_b r_c.
- (T1) ⟺ α_xy | r_z τ.
- (T3) reads |U_y(U_x∩U_z)| = τ |G| / (4 α_xz).
- (T2) ⟺ τ ≤ α_xy α_xz and τ ≤ α_yz α_xz.

So τ_max is the smallest product of two of the three α's. Also τ_min = lcm over the edges xy of α_xy / gcd(α_xy, r_z). Here gcd(α, r_z) = α exactly when α = 2 and z = E2, or α = 3 and z = E3; otherwise it is 1.

## L2. Linking [NEW form of the step used in M-S 3.7, 3.8, 3.10(b)]

**Statement.** If |U_y(U_x∩U_z)| = |U_yU_x|, then U_yU_x ⊆ U_yU_z, and also U_xU_y ⊆ U_zU_y.

*Proof.*
- U_y(U_x∩U_z) is contained in U_yU_x and has the same size, so the two sets are equal. It is also contained in U_yU_z.
- Taking inverses of U_yU_x ⊆ U_yU_z gives U_xU_y ⊆ U_zU_y.

**Core(5) form.** The hypothesis holds exactly when τ = α_xy α_xz, the product of the two α's at x.

## L3. Lemma K [NEW generalisation of M-S Lemma 3.5; subsumes Cor 3.6, Prop 3.8, Cor 3.9, Lemma 3.10(b), Cor 3.11]

**Statement.** Let v be a vertex and S a set of other vertices. Suppose that for every pair {i, j} ⊆ S,

  U_vU_i ⊆ U_jU_i   or   U_vU_j ⊆ U_iU_j.

Then Σ_{i∈S} |U_iU_v| ≤ |G| − |U_v| < |G|, that is, Σ_{i∈S} α_iv / g_iv < 1. In Core(5) this reads Σ_{i∈S} α_iv ≤ 3.

*Proof.*
1. **The sets g_iU_iU_v (i ∈ S) are pairwise disjoint.** Suppose g_iu_iu = g_ju_ju' with u, u' ∈ U_v.
   - Then g_j⁻¹g_i = u_j u' u⁻¹ u_i⁻¹ ∈ U_jU_vU_i.
   - If U_vU_i ⊆ U_jU_i, then U_jU_vU_i ⊆ U_jU_i. This contradicts L0.2 for the disjoint pair g_jU_j, g_iU_i.
   - If instead U_vU_j ⊆ U_iU_j, then g_i⁻¹g_j ∈ U_iU_vU_j ⊆ U_iU_j, the same contradiction.
2. **None of these sets meets g_vU_v.** Each g_iU_iU_v is right U_v-invariant. If it met g_vU_v it would contain g_v, say g_v = g_iu_iu. Then g_vu⁻¹ = g_iu_i lies in g_vU_v ∩ g_iU_i, which is empty.
3. **Counting.** Summing the sizes |g_iU_iU_v| = |U_iU_v| = α_iv |G| / g_iv gives the claim.

**K1 (links).** The pair condition for {i, j} at v holds if |U_i(U_v∩U_j)| = |U_iU_v|, or if |U_j(U_v∩U_i)| = |U_jU_v|.
- By L2, the first equality gives U_vU_i ⊆ U_jU_i, and the second gives U_vU_j ⊆ U_iU_j.
- Core(5) form: τ_{ivj} = α_iv α_vj, where the product is of the two α's at v.

**K2.** If U_vU_j ⊆ U_vU_k (v, j, k distinct), then α_vj / g_vj + α_kj / g_kj < 1. In Core(5): α_vj + α_kj ≤ 3.

*Proof of K2.* Apply L3 at vertex j with S = {v, k}. Its pair condition requires U_jU_v ⊆ U_kU_v, which is the inverse of the hypothesis.

**Corrections to M-S.**
- *M-S Cor 3.11.* The printed proof asserts "by Lemma 3.10a) gcd(r_v, α(U,W)) = 1". With α(U,W) = α(V,W) that step does not follow, because 3.10(a) becomes trivial. The statement is still true whenever α(U,W) is 1 or a prime:
  - since r_u and r_v are coprime, gcd(r_v, α) = 1 or gcd(r_u, α) = 1;
  - in the second case apply 3.10(b) with U and V swapped; the quantity |VU| + |UW| equals |UV| + |VW|.

  In this lane the statement is an instance of K1 + L3: τ ≤ τ_max forces a link at the vertex shared by the two equal edges. This settles the screen caveat.
- *M-S Lemma 4.4.* The first bullet is misprinted. It should read "α(U_2, U_3) ≠ 2", and that is what its proof shows.

## L4. Vertex count P [NEW form; subsumes the Lemma 3.1 counts in M-S Props 4.3/4.5 and screen N1]

Fix a vertex v. For each other vertex j put X_j = U_vU_j and Y_j = G ∖ X_j.

**Statement.**
- **(P1)** |X_j| = α_vj |G| / g_vj, and U_v ⊆ X_j.
- **(P2)** |X_j ∩ X_k| is a multiple of u_vjk = |G| / gcd(n_v, I_jk). In Core(5), u_vjk = |G| / (4 gcd(r_v, α_jk)).
- **(P3)** |X_j ∩ X_k| ≥ max(|U_v(U_j∩U_k)|, |X_j| + |X_k| − |G|). The first term is given by (T3).
- **(P4)** If |X_j ∩ X_k| = |X_j|, then X_j ⊆ X_k, so K2 must hold for that containment.
- **(P5)** For every set S of other vertices:
  Σ_{j∈S} |Y_j| − Σ_{j<k in S} |Y_j ∩ Y_k| < |G|,
  where |Y_j ∩ Y_k| = |G| − |X_j| − |X_k| + |X_j ∩ X_k|.

*Proof.*
- (P2): by L0.5, X_j ∩ X_k is a union of (U_v, U_j∩U_k)-double cosets. By L0.4, each has size a multiple of |G| / gcd(n_v, I_jk). In Core(5), gcd(4r_v, 4r_jr_kα_jk) = 4 gcd(r_v, α_jk), because gcd(r_v, r_jr_k) = 1.
- (P3): U_v(U_j∩U_k) ⊆ X_j ∩ X_k, and inclusion–exclusion gives the second bound.
- (P4): immediate from K2.
- (P5): this is the Bonferroni inequality. Also ∪_j Y_j ⊆ G ∖ U_v, which is a proper subset of G.

**How the engine uses P (sound).** For each pair {j, k}:
- hi_jk is the largest multiple of u that is ≤ min(|X_j|, |X_k|). It is lowered by one unit u when equality would force a containment that K2 forbids. If |X_j| = |X_k|, equality would force X_j = X_k, and then both K2 conditions must hold.
- lo_jk is (P3) rounded up to a multiple of u.
- If lo_jk > hi_jk the pattern is killed: **P-pair**.
- Otherwise the left side of (P5) is non-increasing in each |X_j ∩ X_k|. If (P5) already fails with every |X_j ∩ X_k| set to hi_jk, it fails for the true values, and the pattern is killed: **P-Bonferroni**.

## L5. N2: all ten α = 2 is impossible in Core(5) [NEW, unreviewed]

*Proof.*
1. **Every triangle has τ = 2.** Each edge has α = 2, and at most one vertex is E2. So by (T1) τ is even, and by (T2) τ ≤ 4.
   - If τ = 4, then (T2) holds with equality at every apex. So every vertex is linked for the other two (K1), and L3 at any vertex of the triangle gives 2 + 2 ≤ 3, which is false.
   - Hence τ = 2. Therefore [G : U_a∩U_b∩U_c] = 8 r_a r_b r_c for every triangle.
2. **Fix a vertex v with r_v odd.** One exists, since at most one r is even. For the other vertices j put X_j = U_vU_j; then |X_j| = |G|/2.
3. **Pairwise intersections.** For other vertices j ≠ k put Z_jk = U_v(U_j∩U_k).
   - By (T3), |Z_jk| = 2|G| / (4·2) = |G|/4, and Z_jk ⊆ X_j ∩ X_k.
   - By (P2), |X_j ∩ X_k| is a multiple of |G| / (4 gcd(r_v, 2)) = |G|/4.
   - |X_j ∩ X_k| ≠ |G|/2: otherwise X_j = X_k, and K2 would give 2 + 2 ≤ 3.
   - Hence X_j ∩ X_k = Z_jk.
4. **The three Z's of a triple coincide.** Take distinct other vertices j, k, l and put W = U_j∩U_k∩U_l, so [G:W] = 8 r_j r_k r_l.
   - Q = [G : U_v ∩ W] is divisible by the three triangle indices 8 r_v r_j r_k, 8 r_v r_j r_l and 8 r_v r_k r_l. By coprimality, Q is divisible by 8 r_v r_j r_k r_l.
   - So |U_vW| = |G| Q / (4 r_v · 8 r_j r_k r_l) ≥ |G|/4.
   - But U_vW ⊆ Z_jk ∩ Z_jl ∩ Z_kl, and each Z has size |G|/4. Hence Z_jk = Z_jl = Z_kl = U_vW.
5. **Contradiction.** The six sets Z among the four other vertices are therefore all one set Z with |Z| = |G|/4.
   - The four sets X_j ∖ Z are pairwise disjoint, each of size |G|/4.
   - So |∪_j X_j| = 4·|G|/4 + |G|/4 = 5|G|/4 > |G|, a contradiction.

## L6. Fibre lemma F [NEW form; the case of H1 below subsumes screen N3]

Let H ≤ G with [G:H] = m. For each vertex l put t_l = [U_l : U_l∩H] and λ_l = [H : U_l∩H].

**Statement.**
- **(F1)** n_l t_l = m λ_l and 1 ≤ t_l ≤ m. The coset g_lU_l meets exactly t_l left cosets of H.
- **(F2)** Fix a left coset C of H and z ∈ C, and let S_C = {l : g_lU_l ∩ C ≠ ∅}.
  - For each l ∈ S_C, the set z⁻¹(g_lU_l ∩ C) is a left coset in H of U_l∩H. These cosets are pairwise disjoint.
  - Hence, if |S_C| = s ≥ 2, then gcd(λ_l, λ_l') ≥ s for some l ≠ l' in S_C. This uses L0.7 for s ≤ 4, and for s = 5 the minimality of G (§0; H is proper).
- **(F3)** If U_i ⊆ H, then |U_iU_l| ≤ |HU_l| = t_l |G| / m, that is, α_il / g_il ≤ t_l / m. In particular, if U_i, U_j ⊆ H then α_ij / g_ij ≤ 1/m.

*Proof.*
- (F1): [G : U_l∩H] = n_l t_l = m λ_l. The set g_lU_lH is a union of |U_lH| / |H| = t_l left H-cosets. Also t_l ≤ m, since distinct cosets of U_l∩H in U_l lie in distinct cosets of H.
- (F2): take y ∈ g_lU_l ∩ C. Then g_lU_l = yU_l and C = yH, so g_lU_l ∩ C = y(U_l∩H). Since z⁻¹y ∈ H, z⁻¹y(U_l∩H) is a left coset in H. Disjointness is preserved by translation.
- (F3): U_iU_l ⊆ HU_l, and |HU_l| = |H||U_l| / |H∩U_l| = t_l |H|.

## L7. Sources of a subgroup H

**H1 [PUB, M-S Lemma 3.7].** Suppose [G:U] = a r_u, [G:V] = a r_v, [G:W] = a r_w with r_u, r_v, r_w pairwise coprime, and α(U,V) = α(U,W) = α(V,W) = 1. Then UV = VU = UW = WU = VW = WV is a subgroup of index a. It contains U, V and W. In Core(5), a = 4.

**H2 [NEW, unreviewed].** In Core(5), suppose α_ab = 1, and let V2 be the non-empty set of vertices v with α_va = α_vb = 2. Put W = ⟨U_a, U_b⟩. Then one of the following holds:
- **(i)** [G:W] = 4, W = U_aU_b, and t_v = [U_v : U_v∩W] = 2 for every v ∈ V2;
- **(ii)** [G:W] = 2, and U_v ⊆ W for every v ∈ V2.

*Proof.* Fix v ∈ V2 and work in the triangle {v, a, b}.
1. **τ = 2.** By (T2), τ ≤ min(2·2, 2·1, 2·1) = 2. By (T1), 2 | r_b τ and 2 | r_a τ. Since r_a and r_b are not both even, τ is even. So τ = 2.
2. **U_vU_a = U_vU_b.** By (T3), |U_v(U_a∩U_b)| = 2|G| / (4·1) = |G|/2 = |U_vU_a| = |U_vU_b|. Hence U_vU_a = U_v(U_a∩U_b) = U_vU_b. Call this set X.
3. **X = U_vW.** X is right-invariant under U_a and under U_b, hence under W. Since U_v ⊆ X, this gives U_vW ⊆ X. Conversely X = U_vU_a ⊆ U_vW. So X = U_vW, and |X| = |G|/2 shows W ≠ G.
4. **[G:W] ∈ {2, 4}.** [G:W] divides gcd(4r_a, 4r_b) = 4.
5. **Index 4.** U_aU_b ⊆ W, and |U_aU_b| = |G|/4 = |W|, so W = U_aU_b. Also |U_vW| = t_v |W| = |G|/2, so t_v = 2.
6. **Index 2.** |U_vW| = |G|/2 = |W|, so U_v ⊆ W.

W does not depend on v, so the same alternative holds for all v ∈ V2.

## L8. Soundness of the engine's evaluation points [NEW]

**(M1) Evaluating at τ = τ_min is enough.** Every admissible τ is a multiple of τ_min, so τ ≥ τ_min. Every rule the engine uses is upward closed in each triangle's τ:
- T-kill means τ_min > τ_max.
- A link (K1) at τ_min means (T2) holds with equality there. Then τ = τ_min is forced, because a larger τ would violate (T2). So the links present at τ_min are present for every admissible τ.
- The lower bounds in (P3) increase with τ. The values hi_jk and the K2 conditions do not depend on τ.
- F, H1, H2 and N2 do not use the program's τ.

So a kill at τ_min is a kill for every admissible τ.

**(M2) Representative values of r are enough.** A Core(5) pattern (α, E2, E3) quantifies over all pairwise coprime r. The engine instead fixes

  r_i = p_i · 16^[i=E2] · 81^[i=E3],   with distinct primes p_i ∈ {5, 7, 11, 13, 17}.

Every quantity it uses depends on r only through the following:
- g_ij = 4;
- whether α_xy | r_z, and gcd(r_v, α_jk). Both depend only on the E2/E3 status.
- In F, gcd(λ_l, λ_l') = 2^min(v_2) · 3^min(v_3).
  - No prime p ≥ 5 is common to the two, since the r's are coprime, t ≤ 4 and m ∈ {2, 4}.
  - With v_2(r_E2) = 4 ≥ 3 and v_3(r_E3) = 4 ≥ 2, each min equals the largest value over all instances.
  - The F-kill condition "no pair reaches s" is monotone, so a kill at the representative is a kill for every instance.

**Order.** Rules are applied in the order T, K, P-pair, P-Bonferroni, N2, F-H1, F-H2. The certificate of a pattern is the first rule that fires, together with its witness:

# sun-coset5: Sun's disjoint-coset conjecture at k = 5 in every group

**Slot:** math-1, transition 42. **Created:** 2026-10-18, about 09:40 PDT by the host clock, then frozen read-only. Later corrections and results go only into new files: `README-addendum-vN.md`, `proof-vN.md`, `evidence/...`.

## 1. Exact target

Let G be any group. Let a_1G_1, …, a_5G_5 be pairwise disjoint left cosets of subgroups G_i of finite index. The G_i need not be distinct. Then gcd([G:G_i], [G:G_j]) ≥ 5 for some i < j.

- **Source:** Sun, Internat. J. Math. 17 (2006), Conjecture 1.2, at k = 5 (`sources/arxiv-math0501451v4.pdf`).
- **Sharpness:** five cosets of one subgroup of index 5.
- **Deliverable:** a complete written proof, or an explicit counterexample checked by two independent exact computations. If neither is reached, a PARTIAL report.

**Known before this lane (from primary sources):**
- k = 2, and all p-groups: Sun 2006.
- k = 3, 4 in every group: Zhu, arXiv:0807.2207, Theorem 1.1. Reproved by Margolis–Schnabel (M-S), arXiv:1803.03569, Theorem B with Props 4.2, 4.3, 4.5.
- k ≤ 20 for the group Z only: O'Bryant.
- Menon, arXiv:2608.15873 (2026-08-16), realises (6,6,6,10,15) in A5 although it is not Z-harmonic. So results for Z do not transfer to groups at length 5.

The priority re-search of 2026-10-18 08:34 PDT found no claim of k = 5 for groups; see `evidence/priority-search-v1/summary.json`. A search miss is not novelty.

## 2. Reductions

**R0. Reduction to finite groups.** This is standard; Menon §2 gives the same argument.
- Put N = ∩_i core_G(G_i). It is a normal subgroup of finite index, contained in every G_i.
- Each coset a_iG_i is the full preimage of its image in G/N.
- Passing to G/N therefore preserves all indices [G:G_i], all indices of intersections, and disjointness.
- Hence a counterexample in any group yields one in a finite group with the same data. From here on G is finite.

**R1. Reduction to Core(5).** Sun at k = 5 in all groups is equivalent to the following statement, which is correct as stated in the screen.

> **Core(5).** No finite group G has subgroups U_1, …, U_5 with [G:U_i] = 4r_i, the r_i pairwise coprime, and with pairwise disjoint cosets g_iU_i.

*Proof.*
- (⇐) Core(5) is a special case of the target. In such a configuration every pairwise gcd equals 4, which is less than 5.
- (⇒) Suppose five disjoint cosets have all pairwise gcds ≤ 4. Write n_i = [G:G_i].
  1. **Each gcd is 2, 3 or 4.** By L0.6 no pair is coprime. So each pairwise gcd lies in {2, 3, 4}, and 6 divides none of them.
  2. **All n_i are divisible by 3, or all are even.** If some n_i is odd, it shares the factor 3 with every n_j, so 3 divides all five.
  3. **The case where 3 divides every n_i.** Then every gcd equals 3. Writing n_i = 3r_i, the r_i are pairwise coprime. Four of them contradict M-S Prop 4.3, which is also Zhu Theorem 3.1.
  4. **The case where every n_i is even.** Write n_i = 2^{e_i} m_i with m_i odd. From gcd ∈ {2, 4} we get: the m_i are pairwise coprime, every e_i ≥ 1, and min(e_i, e_j) ≤ 2. So at most one e_i is ≥ 3.
     - If two indices have e = 1, then together with any third index they form (2r, 2r', 2r'') with pairwise coprime r's. This contradicts M-S Prop 4.2, which is also Zhu Theorem 2.2.
     - If exactly one index has e = 1, say n_1 = 2m_1, take any three of the others. They are 4r_j with r_j = 2^{e_j−2} m_j, pairwise coprime because at most one e_j ≥ 3. With r_1 = m_1 odd, this contradicts M-S Prop 4.5.
     - Otherwise every e_i ≥ 2, so n_i = 4r_i with r_i = 2^{e_i−2} m_i pairwise coprime. That is a Core(5) configuration. ∎

**How R1 depends on published results.**
- R1 uses M-S Prop 4.5, which is not implied by Sun's statement for k = 4: in (2r_1, 4r_2, 4r_3, 4r_4) the pairwise gcd 4 already occurs.
- The printed proof of Prop 4.5 relies on Lemma 4.4. Its first bullet is misprinted and should read "α ≠ 2", as its proof shows; see `rules-v1.md` §L3.
- Control C1 re-derives Props 4.2, 4.3 and 4.5 with this lane's own rules T, K, P, as an independent check.

**R2. Restriction (not used for the gate).**
- *Statement.* Let K ≤ G with KG_i = G for every i. Then the configuration restricts to K with the same indices.
- *Proof.* Since g_i ∈ KG_i, the coset g_iG_i meets K, and g_iG_i ∩ K = k_i(K ∩ G_i). Also [K : K∩G_i] = |KG_i| / |G_i| = [G:G_i]. Disjointness is inherited.
- *Consequence.* A minimal counterexample has no proper subgroup K with KG_i = G for all i. If G is solvable, a Hall π-subgroup K, where π is the set of primes dividing the indices, satisfies KG_i = G. So a minimal solvable counterexample is a π-group.

**R3. Coprime-fibre descent (not used for the gate).**
- *Statement.* Let Q be a normal subgroup of G such that the numbers [Q : Q∩G_i] are pairwise coprime. Then the images g_iG_iQ/Q are pairwise disjoint cosets of subgroups of G/Q, and their indices divide the original indices.
- *Proof.* Suppose g_iG_iQ meets g_jG_jQ. Then x = g_i⁻¹g_j ∈ G_iQG_j, so x = u_i q u_j with q ∈ Q. By L0.6 applied in Q, (Q∩G_i)(Q∩G_j) = Q. Hence q ∈ G_iG_j, so x ∈ G_iG_j. By L0.2 this contradicts disjointness.
- *Core(5) application.* Let Q be normal of odd order. Then [Q : Q∩U_i] is odd and divides 4r_i, so it divides r_i, and these numbers are pairwise coprime. So a minimal counterexample to Core(5) has O(G) = 1.
