265|def tri_bounds(a_ij, a_ik, a_jk, i, j, k, st):
266|    """(tau_min, tau_max) of triangle {i,j,k} (rules-k6-v1 L1; CERT-FORMAT 2)."""
267|    t1 = a_ij // gcd_alpha_r(a_ij, st, k)     # edge ij, opposite vertex k
268|    t2 = a_ik // gcd_alpha_r(a_ik, st, j)     # edge ik, opposite vertex j
269|    t3 = a_jk // gcd_alpha_r(a_jk, st, i)     # edge jk, opposite vertex i
270|    tmin = lcm(lcm(t1, t2), t3)
271|    tmax = min(a_ij * a_ik, a_ij * a_jk, a_ik * a_jk)   # products at i, j, k
272|    return tmin, tmax

309|def triangle_tables(st):
310|    tables = {}
311|    for (i, j, k) in TRIANGLES:
312|        ok = set()

323|def enumeration_schedule():
324|    """15 edge steps; each triangle is tested exactly once, when its last edge is set."""
325|    order = (5, 4, 3, 2, 1, 0)
326|    steps = []
327|    tested = []
328|    for idx in range(1, 6):
329|        k = order[idx]
330|        placed = order[:idx]
331|        for pos in range(idx):
332|            m = placed[pos]
333|            tris = []
334|            for m2 in placed[:pos]:
335|                tri = tuple(sorted((k, m, m2)))
336|                tris.append(tri)
337|                tested.append(tri)
338|            steps.append((k, m, tuple(tris)))
339|    if len(steps) != 15 or sorted(tested) != sorted(TRIANGLES):
340|        raise RuntimeError("enumeration schedule does not test each triangle exactly once")
341|    if sorted(tuple(sorted((k, m))) for (k, m, _t) in steps) != sorted(LEX_PAIRS):
342|        raise RuntimeError("enumeration schedule does not set each edge exactly once")
343|    return tuple(steps)

347|    """Set of lex keys of all T-consistent labellings for the marker statuses st."""
348|    tables = triangle_tables(st)
349|    steps = enumeration_schedule()
350|    A = [[0] * 6 for _ in range(6)]
351|    leaves = set()
352|    nodes = [0]

354|    def rec(s):
355|        if s == 15:
356|            leaves.add("".join(chr(48 + A[i][j]) for (i, j) in LEX_PAIRS))
357|            return
358|        k, m, tris = steps[s]
359|        for a in ALPHAS:

362|                guard.check()

417|def check_coverage(items, leaves, getters, guard):
418|    """items: list of (lineno, lexkey). Exact disjoint-orbit cover of `leaves`."""
419|    owner = {}
420|    out = {"rep_not_leaf": [], "image_not_leaf": [], "duplicate": [], "orbits_owned": 0}
421|    for (lineno, key) in items:
422|        guard.check()
423|        if key not in leaves:
424|            out["rep_not_leaf"].append(lineno)
425|            continue
426|        imgs = orbit_of(key, getters)
427|        if any(im not in leaves for im in imgs):
428|            out["image_not_leaf"].append(lineno)
429|            continue
430|        clash = sorted(set(owner[im] for im in imgs if im in owner))
431|        if clash:
432|            out["duplicate"].append([lineno, clash[0]])
433|            continue
434|        for im in imgs:
435|            owner[im] = lineno
436|        out["orbits_owned"] += 1
437|    covered = set(owner)
438|    missing = leaves - covered
439|    out["covered"] = len(covered)
440|    out["missing"] = len(missing)
441|    out["missing_orbits"] = missing_orbit_count(missing, getters, guard)
442|    out["missing_examples"] = sorted(code_from_lexkey(k) for k in missing)[:20]
443|    out["set_equal"] = (covered == leaves)
444|    out["clean"] = (out["set_equal"] and not out["rep_not_leaf"] and not out["image_not_leaf"]
445|                    and not out["duplicate"])
446|    return out

505|def hereditary_ok(members, pairval):
506|    """Every T subset of members with |T| >= 2 has a pair with pairval >= |T| (rules-k6-v2 L9)."""
507|    n = len(members)
508|    for size in range(2, n + 1):

515|def allowed_masks(q, st5, ctx):
516|    """Non-empty supports S (bitmasks over N positions) allowed with maxgcd (CERT-FORMAT 3 RLP)."""
517|    ck = (q, st5)
518|    hit = ctx.allowed_cache.get(ck)
519|    if hit is not None:
520|        return hit

522|    e3 = [v_p(q[x], 3) + (SAT if st5[x][1] else 0) for x in range(5)]

610|def p_bounds(A, st, v, j, k):
611|    xj = Fraction(A[v][j], 5)
612|    xk = Fraction(A[v][k], 5)
613|    ajk = A[j][k]
614|    u = Fraction(1, 5 * gcd_alpha_r(ajk, st, v))
615|    tmin, _tmax = tau_min_max(A, st, v, j, k)
616|    lb = max(Fraction(tmin, 5 * ajk), xj + xk - 1, Fraction(0))
617|    lo = u * math.ceil(lb / u)
618|    mn = min(xj, xk)
619|    hi = u * math.floor(mn / u)
620|    if hi == mn:
621|        j_in_k_ok = A[v][j] + ajk <= 4      # K2 for X_j subset X_k
622|        k_in_j_ok = A[v][k] + ajk <= 4      # K2 for X_k subset X_j
623|        if xj < xk:
624|            barred = not j_in_k_ok
625|        elif xk < xj:
626|            barred = not k_in_j_ok
627|        else:
628|            barred = not (j_in_k_ok and k_in_j_ok)
629|        if barred:
630|            hi = hi - u
631|    return lo, hi, u

662|    x = {j: Fraction(A[v][j], 5) for j in N}

664|    total -= sum((1 - x[j] - x[k] + hi[(j, k)] for (j, k) in itertools.combinations(Ss, 2)),
665|                 Fraction(0))
666|    if total < 1:
667|        raise WitnessFail("PB_BOUND", "value=%s" % total)
668|    return {"v": v, "S": Ss}

695|# CERT-FORMAT 3 "FH5", 6.3).  Sun(6) in H is used only for |T| = 6 and only
696|# inside a least-order counterexample (H = U_a U_b has index 5, so H < G).

698|def fh5_exponents(st, l, t, e5):
699|    """(v2, v3, v5) exponents used for lambda_l = r_l t_l (saturated, C5/C16)."""
700|    b2 = 0
701|    if st.is_e2(l):
702|        b2 = 1 if st.c2 == 1 else SAT
703|    b3 = SAT if st.is_e3(l) else 0
704|    b5 = SAT if (e5 is not None and l == e5) else 0
705|    return (v_p(t, 2) + b2, v_p(t, 3) + b3, v_p(t, 5) + b5)

708|def fgcd_value(ex, ey):
709|    return 2 ** min(ex[0], ey[0]) * 3 ** min(ex[1], ey[1]) * 5 ** min(ex[2], ey[2])
710|

712|def fgcd_table(st, e5):
713|    ex = {(l, t): fh5_exponents(st, l, t, e5) for l in range(6) for t in range(1, 6)}
714|    fg = {}
715|    for (l, t) in ex:
716|        for (l2, t2) in ex:
717|            if l != l2:
718|                fg[(l, t, l2, t2)] = fgcd_value(ex[(l, t)], ex[(l2, t2)])
719|    return fg

722|def fh5_domains(A, a, b, V):
723|    """Coset-set choices (mask over cosets 0..4, t) per vertex; only a is fixed (C6)."""
724|    dom = {a: ((1, 1),)}
725|    dom[b] = tuple((1 << c, 1) for c in COSETS)
726|    for v in V:
727|        c = A[v][a]
728|        if c == 1:
729|            dom[v] = tuple((1 << x, 1) for x in COSETS)
730|        else:
731|            dom[v] = tuple((sum(1 << x for x in comb), c)
732|                           for comb in itertools.combinations(COSETS, c))
733|    for l in range(6):
734|        if l not in dom:
735|            dom[l] = tuple((m, bin(m).count("1")) for m in range(1, 32))
736|    return dom

739|def fh5_search(A, a, b, V, fg, guard):
740|    """Exhaustive search with safe monotone pruning; returns accounting and any survivor."""
741|    free = [l for l in range(6) if l != a and l != b and l not in V]
742|    order = [a, b] + sorted(V) + sorted(free)
743|    dom = fh5_domains(A, a, b, V)
744|    sizes = [len(dom[l]) for l in order]
745|    suffix = [1] * 7
746|    for d in range(5, -1, -1):
747|        suffix[d] = suffix[d + 1] * sizes[d]
748|    t_of = [0] * 6
749|    mask_of = [0] * 6
750|    members = [[] for _ in COSETS]
751|    acct = {"nodes": 0, "pruned_weight": 0, "complete": 0}
752|    found = []
753|
754|    def rejected(l):
755|        t = t_of[l]
756|        m = mask_of[l]
757|        for i in range(6):
758|            if i == l:
759|                continue
760|            ti = t_of[i]
761|            if ti == 0:
762|                continue
763|            if t == 1 and A[l][i] > ti:        # (F3): U_l inside H
764|                return True
765|            if ti == 1 and A[i][l] > t:        # (F3): U_i inside H
766|                return True
767|        for c in COSETS:
768|            if not (m >> c) & 1:
769|                continue
770|            M = members[c]
771|            for size in range(1, len(M) + 1):
772|                k = size + 1
773|                for R in itertools.combinations(M, size):
774|                    T = R + (l,)
775|                    if not any(fg[(x, t_of[x], y, t_of[y])] >= k
776|                               for (x, y) in itertools.combinations(T, 2)):
777|                        return True            # (F2) hereditary Sun(|T|) fails in coset c
778|        return False
779|
780|    def rec(d):
781|        if d == 6:
782|            acct["complete"] += 1
783|            found.append([[l, list(mask_members(mask_of[l])), t_of[l]] for l in range(6)])
784|            return True
785|        l = order[d]
786|        for (m, t) in dom[l]:
787|            acct["nodes"] += 1
788|            if acct["nodes"] % GUARD_STRIDE == 0:
789|                guard.check()
790|            t_of[l] = t
791|            mask_of[l] = m
792|            if rejected(l):
793|                acct["pruned_weight"] += suffix[d + 1]
794|                t_of[l] = 0
795|                mask_of[l] = 0
796|                continue
797|            for c in COSETS:
798|                if (m >> c) & 1:
799|                    members[c].append(l)
800|            hit = rec(d + 1)
801|            for c in COSETS:
802|                if (m >> c) & 1:
803|                    members[c].pop()
804|            t_of[l] = 0
805|            mask_of[l] = 0
806|            if hit:
807|                return True
808|        return False
809|
810|    rec(0)
811|    return {"domain": suffix[0], "nodes": acct["nodes"], "pruned_weight": acct["pruned_weight"],
812|            "complete": acct["complete"], "survivor": (found[0] if found else None)}

815|def verify_FH5(A, st, w, ctx, tp):
816|    if type(w) is not list or len(w) != 3:
817|        raise WitnessFail("FH5_SHAPE", "witness")
818|    ab, Vw, counts = w
819|    if type(ab) is not list or len(ab) != 2 or not all(is_vertex(x) for x in ab):
820|        raise WitnessFail("FH5_SHAPE", "[a,b]")
821|    a, b = ab
822|    if a == b:
823|        raise WitnessFail("FH5_SHAPE", "a == b")
824|    if type(Vw) is not list or any(not is_vertex(x) for x in Vw) or len(set(Vw)) != len(Vw):
825|        raise WitnessFail("FH5_SHAPE", "V")
826|    if type(counts) is not list or len(counts) != 7 or \
827|            any(type(c) is not int or c < 0 for c in counts):
828|        raise WitnessFail("FH5_SHAPE", "counts")
829|    if A[a][b] != 1:
830|        raise WitnessFail("FH5_AB_NOT_ONE", "alpha_ab=%d" % A[a][b])
831|    V = sorted(v for v in range(6) if v != a and v != b and A[v][a] == A[v][b])
832|    if set(Vw) != set(V):
833|        raise WitnessFail("FH5_V_MISMATCH", "witness %s recomputed %s" % (sorted(Vw), V))
834|    if not V:
835|        raise WitnessFail("FH5_V_EMPTY", "")
836|    per = []
837|    for e5 in E5_PLACEMENTS:
838|        fk = (tp, e5)
839|        fg = ctx.fg_cache.get(fk)
840|        if fg is None:
841|            fg = fgcd_table(st, e5)
842|            ctx.fg_cache[fk] = fg
843|        res = fh5_search(A, a, b, V, fg, ctx.guard)
844|        ctx.stats["fh5_placements"] += 1
845|        ctx.stats["fh5_nodes"] += res["nodes"]
846|        ctx.stats["fh5_domain_total"] += res["domain"]
847|        if res["survivor"] is not None:
848|            raise WitnessFail("FH5_SURVIVING_ASSIGNMENT",
849|                              "e5=%s assignment=%s" % (e5, res["survivor"]))
850|        if res["complete"] != 0 or res["pruned_weight"] != res["domain"]:
851|            raise WitnessFail("FH5_ACCOUNTING", "e5=%s %s/%s" % (e5, res["pruned_weight"],
852|                                                                 res["domain"]))
853|        ctx.stats["fh5_pruned_weight"] += res["pruned_weight"]
854|        per.append([("None" if e5 is None else e5), res["domain"], res["nodes"]])
855|    return {"ab": [a, b], "V": V, "placements": per}

859|# RPLP (rules-k6-v3 L12; proof-k6-lemmas-v1 4; CERT-FORMAT 3 "RPLP", 6.4, 6.5).
860|# ---------------------------------------------------------------------------
861|def rplp_frame(A, st, v):
862|    """Forced ordered pairs (positions), admissible A-masks, and ylo/yhi per pair."""
863|    N, st5 = neighbours(st, v)
864|    forced = []
865|    for x in range(5):
866|        for y in range(5):
867|            if x == y:
868|                continue
869|            wv, jv = N[x], N[y]
870|            tmin, _tmax = tau_min_max(A, st, v, wv, jv)
871|            if tmin >= A[v][wv] * A[wv][jv]:      # product taken at w (ordered)
872|                forced.append((x, y))
873|    amasks = [am for am in range(1, 32)
874|              if not any(((am >> y) & 1) and not ((am >> x) & 1) for (x, y) in forced)]
875|    ylo = []
876|    yhi = []
877|    for (x, y) in FIVE_PAIRS:
878|        lo, hi, _u = p_bounds(A, st, v, N[x], N[y])
879|        base = 1 - Fraction(A[v][N[x]] + A[v][N[y]], 5)
880|        ylo.append(base + lo)
881|        yhi.append(base + hi)
882|    return N, st5, forced, amasks, ylo, yhi

885|def verify_RPLP(A, st, w, ctx):
886|    if type(w) is not list or len(w) != 2 or not is_vertex(w[0]) or type(w[1]) is not list:
887|        raise WitnessFail("RPLP_SHAPE", "witness")
888|    v, certs = w
889|    N, st5, forced, amasks, ylo, yhi = rplp_frame(A, st, v)
890|    qs = q_vectors(A, v, N)
891|    if len(certs) != len(qs):
892|        raise WitnessFail("RPLP_CERT_COUNT", "%d != %d" % (len(certs), len(qs)))
893|    pairs_in = {am: tuple(pi for pi, (x, y) in enumerate(FIVE_PAIRS)
894|                          if (am >> x) & 1 and (am >> y) & 1) for am in amasks}
895|    nR = 0
896|    nP = 0
897|    for idx, q in enumerate(qs):
898|        e = certs[idx]
899|        if type(e) is not list or len(e) != 2 or type(e[0]) is not str:
900|            raise WitnessFail("RPLP_ENTRY_SHAPE", "entry %d" % idx)
901|        if e[0] == "R":
902|            if type(e[1]) is not str or e[1] != dual_key(q, st5):
903|                raise WitnessFail("RPLP_R_KEY_MISMATCH", "entry %d q=%s" % (idx, list(q)))
904|            check_q_dual(q, st5, ctx, "RPLP_R")
905|            ctx.stats["rplp_R_refs"] += 1
906|            nR += 1
907|        elif e[0] == "P":
908|            vec = e[1]
909|            if type(vec) is not list or len(vec) != 30:
910|                raise WitnessFail("RPLP_ENTRY_SHAPE", "entry %d vector" % idx)
911|            vals = [parse_rational(s) for s in vec]
912|            if any(r is None for r in vals):
913|                raise WitnessFail("RPLP_BAD_RATIONAL", "entry %d" % idx)
914|            av, bv, uv, lv = vals[0:5], vals[5:10], vals[10:20], vals[20:30]
915|            if any(r < 0 for r in uv) or any(r < 0 for r in lv):
916|                raise WitnessFail("RPLP_P_NEGATIVE", "entry %d" % idx)
917|            allowed = set(allowed_masks(q, st5, ctx))
918|            allowed.add(0)
919|            for am in amasks:
920|                base = sum((bv[x] for x in mask_members(am)), Fraction(0))
921|                base -= sum((uv[pi] - lv[pi] for pi in pairs_in[am]), Fraction(0))
922|                sm = am
923|                while True:
924|                    if sm in allowed:
925|                        ctx.stats["rplp_columns_checked"] += 1
926|                        coef = base + sum((av[x] for x in mask_members(sm)), Fraction(0))
927|                        if coef > 1:
928|                            raise WitnessFail("RPLP_P_COLUMN", "entry %d A=%s S=%s" % (
929|                                idx, [N[x] for x in mask_members(am)],
930|                                [N[x] for x in mask_members(sm)]))
931|                    if sm == 0:
932|                        break
933|                    sm = (sm - 1) & am
934|            obj = sum((av[x] * Fraction(q[x], 5) for x in range(5)), Fraction(0))
935|            obj += sum((bv[x] * Fraction(5 - A[v][N[x]], 5) for x in range(5)), Fraction(0))
936|            obj -= sum((uv[pi] * yhi[pi] for pi in range(10)), Fraction(0))
937|            obj += sum((lv[pi] * ylo[pi] for pi in range(10)), Fraction(0))
938|            if obj < 1:
939|                raise WitnessFail("RPLP_P_OBJECTIVE", "entry %d value=%s" % (idx, obj))
940|            ctx.stats["rplp_P_vectors"] += 1
941|            nP += 1
942|        else:
943|            raise WitnessFail("RPLP_ENTRY_SHAPE", "entry %d tag" % idx)
944|    return {"v": v, "q": len(qs), "R": nR, "P": nP}

950|def verify_record(tp, rep, rule, witness, ctx):
951|    """Verify one (type, rep, rule, witness); returns info or raises WitnessFail/StopRequested."""
952|    if tp not in TYPES:
953|        raise WitnessFail("BAD_TYPE", str(tp))
954|    if type(rep) is not str or CODE_RE.fullmatch(rep) is None:
955|        raise WitnessFail("BAD_REP", str(rep)[:40])
956|    st = Status(tp)
957|    A = matrix_from_code(rep)
958|    if not t_consistent(A, st):
959|        raise WitnessFail("REP_NOT_T_CONSISTENT", rep)
960|    if rule == "K":
961|        return verify_K(A, st, witness)
962|    if rule == "RLP":
963|        return verify_RLP(A, st, witness, ctx)
964|    if rule == "PB":
965|        return verify_PB(A, st, witness)
966|    if rule == "Ppair":
967|        return verify_Ppair(A, st, witness)
968|    if rule == "FH5":
969|        return verify_FH5(A, st, witness, ctx, tp)
970|    if rule == "N26":
971|        return verify_N26(A, st, witness)
972|    if rule == "RPLP":
973|        return verify_RPLP(A, st, witness, ctx)
974|    if rule == "SURVIVE":
975|        raise WitnessFail("SURVIVE_LINE", rep)
976|    raise WitnessFail("BAD_RULE", str(rule)[:20])

**(Sun(6) ⇒ Core5(6)).** A Core5(6) configuration has every pairwise gcd equal to 5, and 5 < 6.

**(Core5(6) ⇒ Sun(6)), group by group.** Assume six disjoint cosets in G have pairwise gcds all ≤ 5, and write n_i = [G:G_i].

2. **The cover by A_p.** Put A_p = {i : p | n_i} for p ∈ {2, 3, 5}. Each of the 15 pairs lies in exactly one A_p. Hence Σ_p C(|A_p|, 2) = 15, and |A_p ∩ A_q| ≤ 1 for p ≠ q.

## 0. Standing setting and imports

**Setting.** These hypotheses hold throughout.
- G is a finite group with subgroups U_0, …, U_5.
- [G:U_i] = 5r_i, where r_i ≥ 1 and the r_i are pairwise coprime. The case r_i = 1 is allowed, and so are arbitrary prime powers.

- The subgroups need not be distinct. In Core5(6), two equal subgroups force both cofactors to equal 1.

**Markers.** E2 is the vertex with r even, together with its class c2 (v₂(r) = 1, or v₂(r) ≥ 2). E3 is the vertex with 3 | r. E5 is the vertex with 5 | r. Each marker may be absent. Any two of them, or all three, may coincide.

**Measures.** All sizes are normalised: for X ⊆ G, write μ(X) = |X|/|G|. Then:
- μ(U_iU_j) = α_ij/5, where α_ij = [G:U_i∩U_j]/(5r_ir_j) ∈ {1, 2, 3, 4};
- μ(U_i) = 1/(5r_i).

**Imports** (rules-k6-v1 §0), with H a subgroup:
- **Sun(2):** L0.6, the Chinese remainder theorem for coprime indices.
- **Sun(3) and Sun(4):** Zhu Theorem 1.1.

## 1. rules-k6-v2 L9: the hereditary Sun condition

**Statement.** Two settings are covered.
- **(a) Left cosets of H.** Let H ≤ G, and assume H < G if |S| = 6 is used. Let C be a left coset of H, and let S be the set of l whose coset g_lU_l meets C. Put λ_l = [H : H∩U_l].
- **(b) Right cosets of U_v.** Let v be a vertex and translate so that g_v = 1. Let C = U_vy be a right coset, and let S be the set of j ≠ v with C ⊆ D_j = U_vg_jU_j. Put d_j = [U_v : U_v ∩ wU_jw⁻¹], where w ∈ g_jU_j ∩ C; the value does not depend on the choice of w.

In both settings: for every T ⊆ S with |T| ≥ 2, some pair l ≠ l′ in T has gcd(λ_l, λ_l′) ≥ |T| (respectively gcd(d_l, d_l′) ≥ |T|).

**Proof of (a).**
1. **Pieces are cosets of H∩U_l.** Fix z ∈ C. For l ∈ S choose y_l ∈ g_lU_l ∩ C. Then g_lU_l = y_lU_l and C = y_lH, so g_lU_l ∩ C = y_l(U_l∩H).
2. **Move into H.** Left-multiplying by z⁻¹ gives z⁻¹y_l(U_l∩H). Here z⁻¹y_l ∈ H, so this is a left coset in H of the subgroup U_l∩H, which has index λ_l in H.
3. **Disjointness.** For l ≠ l′ the sets g_lU_l ∩ C and g_l′U_l′ ∩ C are disjoint, because g_lU_l ∩ g_l′U_l′ = ∅. Left translation preserves disjointness.
4. **Apply Sun.** For T ⊆ S, the cosets {z⁻¹y_l(U_l∩H) : l ∈ T} are |T| pairwise disjoint left cosets in H of finite-index subgroups. Apply Sun(|T|) in H:
   - |T| ≤ 5: by the imports;
   - |T| = 6: by minimality, since H < G.

   So some pair has gcd ≥ |T|.

**Proof of (b).**
1. **Pieces are right cosets.** For j ∈ S, the coset C lies in D_j, so it meets g_jU_j (rules-k6-v1 L7 R3). Choose w_j ∈ g_jU_j ∩ C. Then g_jU_j = w_jU_j and C = U_vw_j, so g_jU_j ∩ C = (U_v ∩ w_jU_jw_j⁻¹)w_j =: A_jw_j.
2. **Index of A_j.** [U_v : A_j] = |U_v·w_jU_jw_j⁻¹| / |U_j|. Right multiplication by w_j preserves size, so this equals |U_vw_jU_j| / |U_j| = |D_j|/|U_j| = d_j.
3. **Move into U_v.** Fix w_0 ∈ C. Right-multiplying by w_0⁻¹ gives A_jc_j with c_j = w_jw_0⁻¹ ∈ U_v.
4. **Invert.** Inversion turns A_jc_j into c_j⁻¹A_j. These are pairwise disjoint left cosets in U_v, of indices d_j, because the g_jU_j are pairwise disjoint and both operations are bijections.
5. **Apply Sun.** For T ⊆ S, |T| ≤ 5. Apply Sun(|T|) in U_v by the imports (Sun(6) is not needed). So some pair has gcd ≥ |T|. ∎

**Consequences for the rules.**

| Rule | Consequence |
|---|---|
| FH5 | In every coset, every subfamily T of the members passes the test. An assignment that fails it for some T is impossible. |
| RLP / RPLP | A support S carries positive mass only if it is **hereditary-allowed**: every T ⊆ S with \|T\| ≥ 2 has a pair with gcd ≥ \|T\|. |

The allowed family is therefore closed under taking subsets.

**Replacing gcd by maxgcd is safe.** The allowed test is monotone, i.e. upward closed, in each pairwise gcd. The true gcd(q_jr_j, q_kr_k) is at most maxgcd, which is computed from the markers:
- only the primes 2 and 3 can be common, since q ≤ 4 and the r's are coprime;
- v₂ of the pair is min(v₂(q_j) + ε_j, v₂(q_k) + ε_k), where ε = 1 for E2 of class 1, ε = SAT for E2 of class 2, and ε = 0 otherwise. Likewise for 3 with E3.

Each minimum **saturates at the partner's q-part**: v₂(q) ≤ 2 and v₃(q) ≤ 1. So maxgcd equals the true gcd for every instance with the given markers. No allowed support is lost.

## 2. rules-k6-v3 L10 (XE)

**Statement.** Let v, j, k be distinct, and suppose U_vU_j = U_vU_k. Then:
- α_jk = 1;
- W := U_jU_k is a subgroup of index 5;
- [U_v : U_v∩W] = α_vj.

**Proof.**
1. **X absorbs W.** X := U_vU_j satisfies XU_j = X, and since X = U_vU_k, also XU_k = X. Hence XW = X for W := ⟨U_j, U_k⟩.
2. **X = U_vW.** Because 1 ∈ X, we have U_v ⊆ X, so U_vW ⊆ XW = X. Conversely, X = U_vU_j ⊆ U_vW.
3. **W ≠ G.** μ(X) = α_vj/5 ≤ 4/5 < 1.
4. **Index 5.** U_j ≤ W and U_k ≤ W, so [G:W] divides [G:U_j] = 5r_j and [G:U_k] = 5r_k. Hence it divides gcd(5r_j, 5r_k) = 5, and since W ≠ G, [G:W] = 5.
5. **α_jk = 1.** U_jU_k ⊆ W, so α_jk/5 = μ(U_jU_k) ≤ μ(W) = 1/5. Hence α_jk = 1 and μ(U_jU_k) = μ(W), so U_jU_k = W.
6. **The index of U_v∩W.** μ(U_vW) = [U_v : U_v∩W]·μ(W), by the product formula. Since μ(U_vW) = μ(X) = α_vj/5, we get [U_v : U_v∩W] = α_vj. ∎

**Corollary.** If α_jk ≥ 2, then U_vU_j ≠ U_vU_k for every vertex v.

## 3. rules-k6-v3 L11 (N2-6)

**Hypotheses.** Let v be a vertex such that:
- (i) v ≠ E2, so r_v is odd;
- (ii) α_vj = 2 for all j ≠ v;
- (iii) α_jk ∈ {2, 4} for all distinct j, k ≠ v;
- (iv) for every triple {j,k,l} ⊆ V∖{v}, gcd(r_v, τ_jkl) = 1.

Condition (iv) holds automatically if v ≠ E3, because every prime factor of τ is 2 or 3. If v = E3, it holds when 3 ∤ τ_jkl for every triple. A sufficient test for that: for each triple, either 3 | τ_min(jkl) — impossible here, since τ_min is a product of 2-powers when all α ∈ {2,4} — or τ_max(jkl) < 3τ_min(jkl). The second suffices because τ is a multiple of τ_min that is at most τ_max.

**Conclusion.** No configuration with these data exists.

**Proof.** Translate so that g_v = 1, and put X_j = U_vU_j for j ≠ v. Then μ(X_j) = 2/5, and U_v ⊆ X_j.

**Step 1: μ(X_j∩X_k) = 1/5 for all j ≠ k.**
- (T1) for the edge jk, whose opposite vertex is v, gives α_jk | r_vτ_vjk. Since r_v is odd and α_jk ∈ {2, 4}, this gives α_jk | τ_vjk, so τ_vjk ≥ α_jk.
- By (T3) at apex v, Z_jk := U_v(U_j∩U_k) has μ(Z_jk) = τ_vjk/(5α_jk) ≥ 1/5. Also Z_jk ⊆ X_j∩X_k.
- By L0.5, X_j∩X_k is a union of (U_v, U_j∩U_k)-double cosets. By L0.4 each has μ a multiple of 1/gcd(5r_v, 5r_jr_kα_jk) = 1/(5·gcd(r_v, α_jk)) = 1/5, since r_v is odd and α_jk ∈ {2, 4}.
- By the XE corollary, X_j ≠ X_k. As both have measure 2/5, this gives μ(X_j∩X_k) < 2/5.
- Hence μ(X_j∩X_k) = 1/5, and so Z_jk = X_j∩X_k with μ(Z_jk) = 1/5.

**Step 2: the three Z's of a triple coincide.** Fix a triple j, k, l of vertices other than v, and put W := U_j∩U_k∩U_l.
- Index of W: [G:W] = 5r_jr_kr_lτ_jkl, using the definition of τ.
- Put Q := [G : U_v∩W]. Then Q is a multiple of [G:W] = 5r_jr_k·(r_lτ_jkl) and of T_vjk = 5r_jr_k·(r_vτ_vjk). So it is a multiple of 5r_jr_k·lcm(r_lτ_jkl, r_vτ_vjk).
- **The coprimality step.** The lcm is a multiple of lcm(r_lτ_jkl, r_v). This equals r_lτ_jkl·r_v, because gcd(r_v, r_lτ_jkl) = gcd(r_v, τ_jkl) = 1. Here gcd(r_v, r_l) = 1, and gcd(r_v, τ_jkl) = 1 by (iv).
- Hence Q ≥ 5r_jr_kr_lτ_jkl·r_v = r_v[G:W].
- By the product formula, μ(U_vW) = Q/(5r_v·[G:W]) ≥ 1/5.
- U_vW ⊆ Z_jk ∩ Z_jl ∩ Z_kl, and each Z has measure 1/5. Hence Z_jk = Z_jl = Z_kl = U_vW.

**Step 3: contradiction.**
- Take any two pairs of vertices from V∖{v}. Two pairs that share a vertex lie in a common triple, and two disjoint pairs {j,k}, {l,m} are linked through {j,l}. So all ten sets Z coincide; call the common set Z, with μ(Z) = 1/5.
- For j ≠ k, X_j∩X_k = Z. So the five sets X_j∖Z are pairwise disjoint, each of measure 1/5.
- Hence μ(∪_j X_j) = μ(Z) + 5·(1/5) = 6/5 > 1, a contradiction. ∎

The markers E3 (when v ≠ E3) and E5 play no role in this proof. If E2 is a neighbour j, nothing changes either: only r_v is involved in the granularity, and only gcd(r_v, ·) enters Step 2.

## 4. rules-k6-v3 L12 (RPLP)

**Setting.** Fix v and translate so that g_v = 1. Write N for the five vertices other than v. For j ∈ N put X_j = U_vU_j, Y_j = G∖X_j and D_j = U_vg_jU_j.

**Facts, by rules-k6-v1 L7 (R1–R3, proved in sun-coset5 proof-v4 §5, which uses only pairwise disjointness and holds verbatim for six cosets):**
- D_j ⊆ Y_j.
- μ(D_j) = q_j/5 with an integer q_j satisfying 1 ≤ q_j ≤ 5 − α_vj.
- d_j = q_jr_j.

Each admissible vector q ∈ ∏_j {1, …, 5 − α_vj} is treated separately. The true configuration has exactly one such q, and it is not known which. So the argument must hold for every q.

**Types.**
- Each of X_j, Y_j and D_j is left U_v-invariant, hence a union of right U_v-cosets.
- For a right coset C, its type is (A_C, S_C), where A_C = {j : C ⊆ Y_j} and S_C = {j : C ⊆ D_j}. Since D_j ⊆ Y_j, S_C ⊆ A_C. By L9(b), S_C is hereditary-allowed whenever |S_C| ≥ 2.
- For each type (A, S) with A ≠ ∅, let x(A,S) be the total measure of right cosets of that type, so x(A,S) ≥ 0.
- The cosets with A_C = ∅ are exactly those in ∩_j X_j, and U_v is one of them.

**Forced containments.** Suppose τ_min(v,w,j) ≥ α_vwα_wj.
- (T2) at apex w gives τ ≤ α_vwα_wj, so τ = α_vwα_wj.
- (T3) at apex v gives μ(U_v(U_w∩U_j)) = τ/(5α_wj) = α_vw/5 = μ(X_w).
- Hence X_w = U_v(U_w∩U_j) ⊆ X_j, that is, Y_j ⊆ Y_w.
- So every type with j ∈ A and w ∉ A has x = 0. The engine omits those columns, which is valid.

**Constraints satisfied by x.**
- **(E1_j)** Σ_{(A,S): j∈S} x(A,S) = μ(D_j) = q_j/5.
- **(E2_j)** Σ_{(A,S): j∈A} x(A,S) = μ(Y_j) = (5 − α_vj)/5.
- **(B_jk)** ylo_jk ≤ Σ_{(A,S): j,k∈A} x(A,S) = μ(Y_j∩Y_k) ≤ yhi_jk.
  - μ(Y_j∩Y_k) = 1 − μ(X_j) − μ(X_k) + μ(X_j∩X_k).
  - lo_jk ≤ μ(X_j∩X_k) ≤ hi_jk, by rules-k6-v1 L4. Those bounds hold for every admissible τ (M1) and every instance (M2).
- **(TOT)** Σ x(A,S) = μ(∪_j Y_j) ≤ 1 − μ(U_v) < 1.

**The LP and its dual.** Let OPT(q) be the minimum of Σ x over all x ≥ 0 on allowed columns that satisfy E1, E2 and B. The true x is feasible, so OPT(q) ≤ μ(∪_j Y_j) < 1. If the system is infeasible, there is a contradiction straight away.

Let a_j, b_j be free rationals and u_jk, l_jk ≥ 0. Suppose that for every allowed column (A,S),
- Σ_{j∈S} a_j + Σ_{j∈A} b_j − Σ_{j<k, j,k∈A} u_jk + Σ_{j<k, j,k∈A} l_jk ≤ 1. ……(★)

**Weak duality.** For any feasible x:
- Σ x ≥ Σ_{(A,S)} x(A,S)·[Σ_{j∈S} a_j + Σ_{j∈A} b_j − Σ_{j,k∈A} u_jk + Σ_{j,k∈A} l_jk], using (★) and x ≥ 0.
- Regrouping the right side by constraint gives Σ_j a_j·(q_j/5) + Σ_j b_j·(5 − α_vj)/5 − Σ u_jk·μ(Y_j∩Y_k) + Σ l_jk·μ(Y_j∩Y_k).
- Since u_jk ≥ 0 and l_jk ≥ 0, this is ≥ Σ_j a_jq_j/5 + Σ_j b_j(5 − α_vj)/5 − Σ u_jk·yhi_jk + Σ l_jk·ylo_jk =: Obj.

So if Obj ≥ 1, then Σ x ≥ 1 for every feasible x, contradicting (TOT).

**Farkas case.** If the primal system has no solution, a dual direction exists with (★) holding with right-hand side 0 and positive objective. Scaling it so that Obj = 1 satisfies (★) with right-hand side 0 ≤ 1. So the same check (★) together with Obj ≥ 1 certifies this case as well.

**Exactness.** All entries are rationals: q_j/5, (5 − α_vj)/5, lo, hi, and the dual entries. The certificate is checked in exact rational arithmetic.

**Conclusion.** If, for **every** admissible q at v, there is either
- an RLP certificate (rules-k6-v1 L7 dual y with value ≥ 1), or
- a vector (a, b, u, l) satisfying (★) on all allowed columns with Obj ≥ 1,

then vertex v admits no configuration, so the pattern is impossible. ∎

The hereditary-allowed test is applied with maxgcd. That is exact, or at least safe, as shown in §1.

# rules-k6-v1: lemmas for the Core5(6) pattern engine

The host-clock freeze record is in `evidence/freeze-v1.sha256`. Corrections go only into new versions.

**Reading conventions.**
- "[NEW]" means only "written in this lane"; it is not a novelty claim.
- **[GEN]** marks lemmas proved for arbitrary indices n_i in sun-coset5 `rules-v1.md` (cb86c168…390f), together with its cumulative erratum `rules-v1-erratum-v3.md`. Their proofs do not depend on the index form. They are restated below with explicit hypotheses and specialised to Core5(6).

## 0. Setting

**Harmonic configuration.** G is finite. U_1, …, U_k ≤ G have indices n_i = [G:U_i]. The cosets g_1U_1, …, g_kU_k are **pairwise disjoint**.

**Pairwise data.** For i ≠ j:
- g_ij = gcd(n_i, n_j);
- I_ij = [G:U_i∩U_j];
- α_ij = I_ij / lcm(n_i, n_j).

**Core5(6).** k = 6 and n_i = 5r_i, where the r_i are pairwise coprime. Then:
- g_ij = 5, lcm(n_i, n_j) = 5r_ir_j and I_ij = 5r_ir_jα_ij.
- **Special vertices:**
  - E2 is the vertex with r even, if any. Its class is c2 = 1 if v₂(r_E2) = 1, and c2 = 2 if v₂(r_E2) ≥ 2.
  - E3 is the vertex with 3 | r, if any.
  - E5 is the vertex with 5 | r, if any.
  - Any two of these may coincide.

**Inputs from the minimal counterexample frame** (README §2):
- Sun(k) for k ≤ 4 holds in every group, by Zhu Theorem 1.1 and L0.6.
- Sun(5) holds in every group, by sun-coset5 proof-v4. This is a stated dependence.
- Sun(6) holds in every proper subgroup of the minimal G.

## L0. Basic facts [GEN; rules-v1 L0]

- **L0.1** Product formula: |XY| = |X||Y| / |X∩Y|.
- **L0.2** gU ∩ hV = ∅ if and only if g⁻¹h ∉ UV.
- **L0.3** |U_iU_j| = α_ij|G| / g_ij and 1 ≤ α_ij ≤ g_ij − 1. In Core5(6): α_ij ∈ {1, 2, 3, 4} and |U_iU_j| = α_ij|G|/5.
  - *Proof.* L0.1 gives |U_iU_j| = |G| I_ij / (n_i n_j) = α_ij|G| / g_ij. Disjointness and L0.2 give U_iU_j ≠ G.
- **L0.4** Every double coset AxB has size a multiple of |G| / gcd([G:A], [G:B]).
- **L0.5** AB ∩ AC is a union of (A, B∩C)-double cosets.
- **L0.6** If [G:A] and [G:B] are coprime, then AB = G, so every left coset of A meets every left coset of B. This is Sun's statement at k = 2.

## L1. Triangle lemma T [GEN; rules-v1 L1]

For distinct a, b, c put L = lcm(n_a, n_b, n_c) and T = [G : U_a∩U_b∩U_c]. Then L | T, and τ = T/L satisfies:

- **(T1)** I_xy | Lτ for every pair {x, y}.
- **(T3)** |U_y(U_x∩U_z)| = |G| L τ / (n_y I_xz).
- **(T2)** |U_y(U_x∩U_z)| ≤ min(|U_yU_x|, |U_yU_z|).

**Core5(6) form.**
- L = 5r_ar_br_c, because the r's are pairwise coprime.
- (T1) is equivalent to α_xy | r_zτ.
- (T3) reads |U_y(U_x∩U_z)| = τ|G| / (5α_xz).
- (T2) is equivalent to τ ≤ α_xyα_xz and τ ≤ α_yzα_xz.
- So τ_max is the smallest, over the three vertices, of the product of the two α's at that vertex.
- τ_min is the lcm over the three edges of α_e / gcd(α_e, r_opp), where r_opp is the r of the vertex opposite the edge. The table below lists gcd(α, r_z) by E2/E3 status alone; a prime p ≥ 5 never divides α ≤ 4.

| α | gcd(α, r_z) |
|---|---|
| 1 | 1 |
| 2 | 2 if z = E2, else 1 |
| 3 | 3 if z = E3, else 1 |
| 4 | 4 if z = E2 with c2 = 2; 2 if z = E2 with c2 = 1; else 1 |

The admissible τ are exactly the multiples of τ_min that are ≤ τ_max.

**T-kill.** If τ_min > τ_max, no configuration has these data. T-kills are **triangle-local**: whether a triangle is forbidden depends only on its three α's and on the E2-class and E3 status of its three vertices.

## L2. Links [GEN; rules-v1 L2]

If |U_y(U_x∩U_z)| = |U_yU_x|, then U_yU_x ⊆ U_yU_z and U_xU_y ⊆ U_zU_y.

In Core5(6) the hypothesis holds exactly when τ = α_xyα_xz.

## L3. Lemma K [GEN; rules-v1 L3]

**Statement.** Let v be a vertex and S a set of other vertices such that, for every pair {i, j} ⊆ S, U_vU_i ⊆ U_jU_i or U_vU_j ⊆ U_iU_j. Then Σ_{i∈S} |U_iU_v| ≤ |G| − |U_v|. In Core5(6) this reads Σ_{i∈S} α_iv ≤ 4.

**Proof.**
1. The sets g_iU_iU_v (i ∈ S) are pairwise disjoint, by L0.2.
2. Each of them is right U_v-invariant and misses g_vU_v.
3. Adding their sizes gives the bound.

**K1.** The pair condition at v holds for {i, j} when |U_i(U_v∩U_j)| = |U_iU_v| (L2 with apex i), or when |U_j(U_v∩U_i)| = |U_jU_v|. In Core5(6) this is τ_ivj = α_ivα_vj.

**K2.** If U_vU_j ⊆ U_vU_k, then α_vj + α_kj ≤ 4. This is L3 applied at j with S = {v, k}, using the inverse U_jU_v ⊆ U_kU_v.

## L4. Vertex count P [GEN; rules-v1 L4]

Fix a vertex v. For each other vertex j put X_j = U_vU_j and Y_j = G ∖ X_j.

- **(P1)** |X_j| = α_vj/5, and U_v ⊆ X_j.
- **(P2)** |X_j ∩ X_k| is a multiple of u = 1/gcd(n_v, I_jk). In Core5(6), u = 1/(5·gcd(r_v, α_jk)), with the gcd read from the table in L1.
- **(P3)** |X_j ∩ X_k| ≥ max(τ_vjk/(5α_jk), |X_j| + |X_k| − 1).
- **(P4)** If X_j ⊆ X_k, then K2 holds.
- **(P5)** For every set S of other vertices: Σ_{j∈S} |Y_j| − Σ_{j<k in S} |Y_j ∩ Y_k| < 1, with |Y_j ∩ Y_k| = 1 − |X_j| − |X_k| + |X_j ∩ X_k|.

**How the engine applies P.** All sizes are in units of |G|.
- hi_jk is the largest multiple of u that is ≤ min(|X_j|, |X_k|). It is lowered by one unit u if equality would force a containment that K2 forbids. If |X_j| = |X_k|, equality forces equality of the two sets, so both K2 conditions must hold.
- lo_jk is the bound in (P3), rounded up to a multiple of u.
- **P-pair kill:** lo_jk > hi_jk.
- **PB kill:** (P5) fails when every intersection is set to its hi value.

## L5. Fibre lemma F [GEN; rules-v1 L6], with k = 6 inputs

Let H ≤ G with [G:H] = m. For each l put t_l = [U_l : U_l∩H] and λ_l = [H : U_l∩H] = n_l t_l / m.

- **(F1)** g_lU_l meets exactly t_l left cosets of H, and 1 ≤ t_l ≤ m. Also t_l = 1 if and only if U_l ⊆ H.
- **(F2)** Fix a left coset C of H. The vertices whose cosets meet C give pairwise disjoint left cosets in H of the subgroups U_l∩H. So if s ≥ 2 of them meet C, some pair has gcd(λ_l, λ_l') ≥ s. The input used depends on s:
  - s ≤ 4: Zhu's theorem, or L0.6 when s = 2;
  - s = 5: Sun(5), a stated dependence;
  - s = 6: minimality, since H is proper.
- **(F3)** If U_i ⊆ H, then |U_iU_l| ≤ |HU_l| = t_l|G|/m. In Core5(6) with m = 5, this reads α_il ≤ t_l.

Proofs as in rules-v1 L6.

## L6. H5: index-5 subgroups from α = 1 edges [NEW]

**Statement.** In Core5(6), suppose α_ab = 1 and V := {v ∉ {a, b} : α_va = α_vb} ≠ ∅. Then:
- H := U_aU_b is a subgroup of index 5;
- U_a ⊆ H and U_b ⊆ H;
- for every v ∈ V, t_v = [U_v : U_v∩H] = α_va. In particular U_v ⊆ H when α_va = 1.

This covers M-S Lemma 3.7 with a = 5, which is the case α_va = 1.

**Proof.** Fix v ∈ V and put c = α_va = α_vb ∈ {1, 2, 3, 4}. Work in the triangle {v, a, b}.
1. **The bound τ ≤ c.** The vertex products are c at a, c at b and c² at v, so (T2) gives τ ≤ c.
2. **c divides τ.** Suppose c ≥ 2. By (T1), c | r_b τ and c | r_a τ, using the edges va and vb.
   - If c is prime (2 or 3): r_a and r_b are coprime, so c fails to divide at least one of them. Hence c | τ.
   - If c = 4: one of r_a, r_b is odd, and that one forces 4 | τ.
   - So τ = c. For c = 1 the bound already gives τ = 1.
3. **U_vU_a = U_vU_b.** By (T3) with apex v, |U_v(U_a∩U_b)| = τ/(5α_ab) = c/5 = |U_vU_a| = |U_vU_b|. Since U_v(U_a∩U_b) lies inside both, U_vU_a = U_v(U_a∩U_b) = U_vU_b. Call this set X.
4. **X = U_vW for W = ⟨U_a, U_b⟩.** X is right-invariant under U_a and under U_b, hence under W. So U_vW ⊆ X ⊆ U_vW.
5. **W has index 5.** |X| = c/5 < 1, so W ≠ G. Also [G:W] divides gcd(5r_a, 5r_b) = 5, since U_a ≤ W. Hence [G:W] = 5.
6. **W = U_aU_b.** U_aU_b ⊆ W and |U_aU_b| = 1/5 = |W|.
7. **t_v = c.** |U_vW| = t_v|W| = t_v/5, and |U_vW| = |X| = c/5.

W does not depend on v. ∎

## L7. Lemma R in LP form [NEW]

**Setting.** A harmonic configuration (pairwise disjoint cosets). Fix a vertex v and translate so that g_v = 1. For j ≠ v put D_j = U_vg_jU_j and d_j = |D_j|/|U_j|.

**The fibre facts (R1)–(R5′).** These are proved in sun-coset5 proof-v4 §5; the proofs use only disjointness and L0.
- **(R1)** D_j ∩ U_vU_j = ∅.
- **(R2)** |D_j| is a positive multiple of |G|/gcd(n_v, n_j).
- **(R3)** Every right coset U_vy ⊆ D_j meets g_jU_j, in a right coset of a subgroup of U_v of index d_j.
- **(R5′)** If D_j ∩ D_k ≠ ∅, then gcd(d_j, d_k) ≥ 2.

**A strengthening (R4′).** Let C be a right U_v-coset and S_C = {j : C ⊆ D_j}. If |S_C| = s ≥ 2, then:
- every pair in S_C has gcd ≥ 2;
- some pair in S_C has gcd ≥ s.

*Proof of (R4′).* The pieces in C, right-translated into U_v and then inverted, are pairwise disjoint left cosets in U_v. Apply Sun(s) in U_v (s ≤ 5, using the inputs of §0) to all of them. For the "every pair" part, apply L0.6 to each pair.

**Core5(6) form.**
- |D_j| = q_j|G|/5 with 1 ≤ q_j ≤ 5 − α_vj, so d_j = q_jr_j.
- The common primes of d_j and d_k lie in {2, 3}: q ≤ 4, and no two r's share a prime. In particular E5 is irrelevant here.

**LP statement.**
- **Measure.** For each non-empty S, let μ(S) be the total size of the right U_v-cosets C with S_C = S.
- **Coverage.** Σ_{S∋j} μ(S) = |D_j| = q_j/5.
- **Total.** Σ_S μ(S) ≤ 1 − |U_v| < 1. Here U_v·1 lies in no D_j, by (R1).
- **Support.** μ(S) > 0 only for **allowed** S: either |S| = 1, or every pair in S has maxgcd ≥ 2 and some pair has maxgcd ≥ |S|. "maxgcd" is the largest value of gcd(q_jr_j, q_kr_k) over all r with the given special statuses.
- **Lower bound.** Let LPmin(q) be the minimum of Σ_S μ(S) over μ ≥ 0 on allowed S with Σ_{S∋j} μ(S) ≥ q_j/5. This relaxes the true coverage equalities, so LPmin(q) is at most the true total.

**RLP kill.** If LPmin(q) ≥ 1 for every admissible vector q, i.e. every q with 1 ≤ q_j ≤ 5 − α_vj, then vertex v is impossible.

**Evaluation of LPmin.** By LP duality,
  LPmin = max Σ_j (q_j/5) y_j subject to Σ_{j∈S} y_j ≤ 1 for every maximal allowed S, and y ≥ 0.
- Maximal sets suffice: every non-maximal allowed set lies inside an allowed superset, and y ≥ 0 makes its constraint redundant.
- The engine solves this packing LP by exact rational simplex with Bland's rule.

**Soundness of maxgcd.** Using maxgcd only enlarges the allowed family, which lowers LPmin. Hence a kill computed with maxgcd is a kill for every instance.

## L8. Evaluation points, representatives and E5 [NEW]

**(M1) τ_min suffices.** Every admissible τ is a multiple of τ_min. The rules T, K, P and RLP are upward closed in τ:
- a K1 link at τ_min forces τ = τ_min;
- the (P3) lower bounds increase with τ;
- RLP and F do not read τ.

So a kill at τ_min is a kill for every admissible τ.

**(M2) Representatives suffice.** A pattern quantifies over all pairwise coprime r with given statuses: the E2 vertex and its class, the E3 vertex, and the E5 vertex.
- **How r enters each rule.** T, K, P and the support of RLP depend on r only through these statuses. For E2 the class matters for α = 4, and for maxgcd the 2-adic part of q_jr_j is v₂(q_j) + v₂(r_j). In the RLP maxgcd, E2 of class 2 contributes an unbounded v₂.
- **F rules.** gcd(λ_l, λ_l') involves only the primes 2, 3 and 5:
  - t ≤ 5 and m = 5, and the r's are coprime;
  - a class-1 E2 contributes v₂ = 1 exactly;
  - class-2 E2, E3 and E5 contribute exponents larger than any t-part. The representative exponents 2⁴, 3⁴ and 5⁴ dominate.
  - Because the kill condition is monotone in the gcds, a kill at the representative is a kill for every instance.

**E5.** E5 affects only the F rules: no α and no q is divisible by 5, and only one r is divisible by 5. **Every F certificate must refute every E5 placement, including "no E5".** The engine checks all seven placements.

## Certificates and the order of rules

**Enumeration.**
- There are 8 special-placement types up to S6.
- For each type, backtracking runs over the 15 edges, with α ∈ {1, 2, 3, 4}.
- A partial labelling is pruned as soon as a completed triangle is T-forbidden. Its certificate is the triangle-local record: the vertices, the three α's and the statuses.
- Every complete T-consistent labelling (a "leaf") is tested in this order: K, RLP, P (P-pair, then PB), FH5.

**FH5.**
- For each 1-edge (a, b) with V ≠ ∅, use H5 with m = 5.
- Vertices fixed inside H: a, b and every v ∈ V with α_va = 1.
- Forced values: t_v = α_va for the other v ∈ V.
- Free vertices: t_l ranges over 1..5. Every met-coset set M_l with |M_l| = t_l is enumerated, and a's coset is fixed to label 0.
- An assignment is rejected iff (F3) fails at a vertex with t = 1, or (F2) fails in some coset.
- The certificate requires every assignment to be rejected for **every** E5 placement.

**Output.**
- Each leaf yields one certificate or the label SURVIVE.
- Leaves are deduplicated into orbits by a canonical form under the stabiliser of the special placement.
- **Controls** record observations only and do not gate the run. They are:
  - **Core(5) and Core_3(4) reruns.** The same enumerator and the same code paths for T, K, P and RLP are run with the constants changed to (d = 4, k = 5) for Core(5) and (d = 3, k = 4) for Core_3(4). FH5 is omitted, because H5 is specific to the prime index 5.
    - Core(5): the canonical orbit count must be 21,150. Every orbit these rules leave must be one of the orbits that sun-coset5 engine-v1 killed with N2 or FH2, or the orbit S-D. Any other orbit would mean a rule is unsound.
    - Core_3(4) (= M-S Prop 4.3): the expectation is that every orbit is killed.
  - Realisable configurations survive.
  - Selected mutations break their certificates.

## L9. Hereditary Sun condition [NEW]

**Statement.** Take the setting of rules-k6-v1 L5 (F2) or L7 (R4′). Let C be a left coset of H in (F2), or a right coset of U_v in (R4′). Let S_C be the set of vertices whose pieces lie in C, with local indices λ_l in (F2) or d_l = q_lr_l in (R4′). Then for **every** subset T ⊆ S_C with |T| ≥ 2, some pair in T has gcd ≥ |T|.

**Proof.**
1. **The pieces of T are pairwise disjoint.** The pieces of all of S_C are pairwise disjoint, so in particular the pieces indexed by T are.
2. **Move them into the subgroup.**
   - In (F2): left-translate by z⁻¹, where z ∈ C. This gives pairwise disjoint left cosets in H of the subgroups U_l∩H.
   - In (R4′): right-translate by w⁻¹, where w ∈ C, and then invert. This gives pairwise disjoint left cosets in U_v of subgroups of index d_l.

   The translation steps are those of rules-k6-v1 L5 and of sun-coset5 proof-v4 §5 (R5′).
3. **Apply Sun(|T|).** Apply Sun's statement for k = |T| inside H, or inside U_v, using the inputs of rules-k6-v1 §0:
   - |T| = 2: L0.6.
   - |T| ∈ {3, 4}: Zhu Theorem 1.1.
   - |T| = 5: Sun(5). This rests on sun-coset5 proof-v4, a stated dependence.
   - |T| = 6: this can occur only in (F2), because in (R4′) S_C contains at most 5 vertices. Here use the minimality of G, since H is proper.

   So some pair of T has gcd ≥ |T|. ∎

**Effect on the rules.**

| Rule | Change |
|---|---|
| (F2) in FH5 | An assignment is rejected if the condition fails for **some** subset T of the vertices meeting a coset. rules-k6-v1 tested only T = S_C. |
| RLP (rules-k6-v1 L7) | A support S is **allowed** only if every subset T ⊆ S with \|T\| ≥ 2 has a pair with maxgcd ≥ \|T\|. |

The allowed family is therefore closed under taking subsets. The "maximal sets suffice" step and the soundness argument for maxgcd go through unchanged. A smaller allowed family can only raise LPmin, and the new condition is valid for every instance, so every RLP kill remains sound.

**Engine.** `code/core56_engine_v2.py` is `core56_engine_v1.py` with exactly two functions changed:
- `lp_min`: its allowed-support test becomes the hereditary condition;
- `fibre5_feasible`: its coset test becomes the hereditary condition.

The exact diff is in `evidence/engine-v1-to-v2.diff`. Everything else is unchanged: the enumeration, the T pruning, the canonical forms, K, P, M1/M2 and the E5 loop.

# rules-k6-v3: lemmas XE, N2-6 and RPLP, plus the certificate formats (addendum to rules-k6-v1 and v2)

The host-clock freeze record is in `evidence/freeze-v3.sha256`. rules-k6-v1 (3b135fea…b187) and rules-k6-v2 (b45049a0…fa6) stay frozen. Here [NEW] means only "written in this lane". All sizes below are in units of |G|.

**Why these lemmas.** The frozen run engine-v2-core56 left 263 orbits (11:20 PDT). All of them have E2 at vertex 0 and E3 at vertex 1, and almost every α in them is 2 or 4. The lemmas below target that family.

## L10. XE: equal products force an index-5 subgroup [NEW]

**Statement.** In Core5(6), let v, j, k be distinct and suppose U_vU_j = U_vU_k. Then α_jk = 1, W = U_jU_k is a subgroup of index 5, and [U_v : U_v∩W] = α_vj.

**Proof.**
1. X := U_vU_j = U_vU_k is right-invariant under U_j and under U_k, hence under W := ⟨U_j, U_k⟩.
2. Since U_v ⊆ X, we get U_vW ⊆ X. Also X = U_vU_j ⊆ U_vW. So X = U_vW.
3. |X| = α_vj/5 < 1 (L0.3), so W ≠ G.
4. Both U_j and U_k lie in W, so [G:W] divides gcd(5r_j, 5r_k) = 5. Hence [G:W] = 5.
5. U_jU_k ⊆ W, so α_jk/5 = |U_jU_k| ≤ |W| = 1/5. Hence α_jk = 1, and then U_jU_k = W.
6. Finally |U_vW| = [U_v : U_v∩W]·|W|, so α_vj/5 = [U_v : U_v∩W]/5. ∎

**Corollary.** If α_jk ≥ 2, then X_j ≠ X_k at every vertex v.

## L11. N2-6 [NEW]

**Statement.** In Core5(6), let v be a vertex such that:
1. r_v is odd, i.e. v ≠ E2;
2. α_vj = 2 for all five j ≠ v;
3. α_jk ∈ {2, 4} for every pair j, k ≠ v;
4. if v = E3, then 3 ∤ τ_jkl for every triple j, k, l ≠ v. A sufficient check: in every such triangle, τ_max(jkl) < 3·τ_min(jkl) whenever 3 ∤ τ_min(jkl).

Then no configuration exists.

**Proof.** Translate so that g_v = 1. For j ≠ v put X_j = U_vU_j, so |X_j| = 2/5.

1. **Each pairwise intersection is exactly a fifth.** Fix j ≠ k.
   - (T1) on the edge jk, whose opposite vertex is v: α_jk | r_vτ_vjk. Since r_v is odd and α_jk ∈ {2, 4}, α_jk | τ_vjk, so τ_vjk ≥ α_jk.
   - Put Z_jk := U_v(U_j∩U_k). By (T3), |Z_jk| = τ_vjk/(5α_jk) ≥ 1/5, and Z_jk ⊆ X_j∩X_k.
   - By (P2), |X_j∩X_k| is a multiple of 1/(5·gcd(r_v, α_jk)) = 1/5, because r_v is odd.
   - By XE, X_j ≠ X_k, so |X_j∩X_k| < 2/5.
   - Hence |X_j∩X_k| = 1/5. Then |Z_jk| = 1/5 and X_j∩X_k = Z_jk.
2. **The three fifths of a triple coincide.** Fix a triple j, k, l ≠ v and put W = U_j∩U_k∩U_l.
   - [G:W] = 5r_jr_kr_lτ_jkl.
   - Q := [G : U_v∩W] is a common multiple of [G:W] and of T_vjk = 5r_vr_jr_kτ_vjk. So Q is a multiple of 5r_jr_k·lcm(r_lτ_jkl, r_vτ_vjk).
   - gcd(r_v, r_lτ_jkl) = gcd(r_v, τ_jkl). This equals 1: τ involves only the primes 2 and 3, r_v is odd, and either v ≠ E3 or 3 ∤ τ_jkl by condition 4.
   - So r_v·r_lτ_jkl divides that lcm, and Q ≥ r_v[G:W].
   - Therefore |U_vW| = Q/(5r_v[G:W]) ≥ 1/5.
   - U_vW ⊆ Z_jk ∩ Z_jl ∩ Z_kl, and each Z has size 1/5. Hence Z_jk = Z_jl = Z_kl.
3. **Contradiction.** Any two pairs from the five neighbours are linked by a chain of triples, so all ten sets Z_jk are one set Z with |Z| = 1/5. The five sets X_j ∖ Z are pairwise disjoint, each of size 1/5. So |∪_j X_j| = 1/5 + 5/5 = 6/5 > 1. ∎

## L12. RPLP: Lemma R combined with the vertex counts [NEW]

**Setting.** A harmonic configuration in Core5(6). Fix a vertex v and normalise g_v = 1. Write N for the five other vertices. For j ∈ N put:
- X_j = U_vU_j and Y_j = G ∖ X_j;
- D_j = U_vg_jU_j, with |D_j| = q_j/5, where 1 ≤ q_j ≤ 5 − α_vj.

**Which sets are right U_v-cosets unions.**
- X_j is left U_v-invariant, so X_j and Y_j are unions of right U_v-cosets.
- D_j is also left U_v-invariant, so it too is a union of right U_v-cosets.
- By (R1), D_j ⊆ Y_j.

**Types of right cosets.** Give each right U_v-coset C the type (A_C, S_C), where A_C = {j : C ⊆ Y_j} and S_C = {j : C ⊆ D_j}. Then S_C ⊆ A_C, and S_C is hereditary-allowed (rules-k6-v2 L9 with rules-k6-v1 L7, evaluated with maxgcd).

**Forced containments.** Suppose τ_min(v,w,j) ≥ α_vwα_wj. By (T2) this forces τ = α_vwα_wj. Then (T3) gives |U_v(U_w∩U_j)| = |X_w|, so X_w ⊆ X_j, that is, Y_j ⊆ Y_w. Hence a type with j ∈ A and w ∉ A has measure 0.

**The LP.** For every allowed type (A, S) with A ≠ ∅ that is consistent with the forced containments, let x(A,S) be the total size of right cosets of that type. Then:
- **(E1_j)** Σ_{S∋j} x(A,S) = q_j/5.
- **(E2_j)** Σ_{A∋j} x(A,S) = |Y_j| = (5 − α_vj)/5.
- **(U_jk), (L_jk)** ylo_jk ≤ Σ_{A⊇{j,k}} x(A,S) ≤ yhi_jk, where yhi/ylo = 1 − (α_vj + α_vk)/5 + hi_jk/lo_jk, with hi and lo as in the P rule (rules-k6-v1 L4).
- **(TOT)** Σ x(A,S) = |∪_j Y_j| ≤ 1 − |U_v| < 1, since U_v lies in no Y_j.

**RPLP kill.** Fix v. If, for every admissible q, the minimum of Σ x subject to E1, E2, U and L is ≥ 1 (or the system is infeasible), then vertex v is impossible.

**Certificate (weak duality).** Give a dual vector (a_j, b_j free; u_jk ≥ 0; l_jk ≥ 0) such that:
- for every allowed column (A, S): Σ_{j∈S} a_j + Σ_{j∈A} b_j − Σ_{{j,k}⊆A} u_jk + Σ_{{j,k}⊆A} l_jk ≤ 1;
- the dual objective satisfies Σ_j a_j q_j/5 + Σ_j b_j(5 − α_vj)/5 − Σ u_jk yhi_jk + Σ l_jk ylo_jk ≥ 1.

Weak duality then gives min Σ x ≥ 1. If the primal is infeasible, such a vector exists by Farkas' lemma.

**Soundness.**
- maxgcd only enlarges the set of allowed columns, which lowers the minimum.
- The lo and hi bounds are valid for every admissible τ (rules-k6-v1 L8, M1).
- A forced containment holds whenever its hypothesis holds at τ_min, because that hypothesis pins τ.

## Certificate formats (engine v3; every one is checkable without trusting the engine)

1. **T.** For each special type, the enumeration tree is deterministic. A pruned node records its triangle, its three α's and the statuses. The checker regenerates the tree independently. The engine publishes, for each type: the number of labelled leaves, the number of pruned nodes, and the sha256 of the sorted list of labelled leaf codes.
2. **Orbits.** One line per canonical orbit: type, a representative α-matrix (as a code string), the rule, and its witness.
3. **K (v, S), P (v, pair or S), N2-6 (v).** The checker recomputes each.
4. **RLP (v).** For every admissible q at v, the global file `rlp_duals.json` holds an exact dual y ≥ 0 under the key "q|stats". The dual must satisfy Σ_{j∈S} y_j ≤ 1 for **every** allowed S (all of them, not only the maximal ones) and Σ (q_j/5) y_j ≥ 1.
5. **FH5 ((a,b), V).** The checker re-runs the exhaustive fibre search for all seven E5 placements. The engine records the number of assignments examined in each placement.
6. **RPLP (v).** For every admissible q, either an RLP dual (when the RLP LP for that q is already ≥ 1) or an RPLP dual vector (a, b, u, l) in a canonical constraint order: j in N ascending; then pairs (j,k) in lexicographic order. The checker rebuilds the columns, bounds and forced containments itself and verifies the vector exactly.

This file describes the output of `code/core56_engine_v3.py` (sha256 01f0de4c…d0cd), run under rules-k6-v1, v2 and v3. It is written so that an independent checker can be built from this file plus the rules texts, without reading any engine code.

## 0. Mathematical setting (summary of rules-k6-v1 §0)

**The configuration.**
- G is a finite group. There are six subgroups U_0, …, U_5 with [G:U_i] = 5r_i, and the r_i are pairwise coprime.
- The cosets g_iU_i are pairwise disjoint.
- For i ≠ j, put α_ij = [G:U_i∩U_j]/(5r_ir_j). Each α_ij lies in {1, 2, 3, 4}.
- A pattern is the symmetric matrix α together with the special statuses below.

**Special statuses.**
- **E2:** the vertex with r even, if any, together with a class: c2 = 1 if v₂(r) = 1, and c2 = 2 if v₂(r) ≥ 2.
- **E3:** the vertex with 3 | r, if any.
- **E5:** the vertex with 5 | r, if any. It is used only by FH5, which must refute all seven placements: none, or each of the vertices 0..5.

**Units.** All sizes are in units of |G|.

## 1. Types and stabilisers

**The eight types.** A type is a triple (e2, c2, e3). Here e2 is either `None` or the E2 vertex, c2 ∈ {0, 1, 2} with 0 meaning "no E2", and e3 is either `None` or the E3 vertex. The types used are exactly:

```
(None,0,None) (None,0,0) (0,1,None) (0,1,0) (0,1,1) (0,2,None) (0,2,0) (0,2,1)
```

- (0,c,0) means E2 and E3 lie on the same vertex 0.
- (0,c,1) means they lie on different vertices.

**Completeness.** Every placement of E2 (with its class) and E3 on the six vertices is mapped onto exactly one of these types by some permutation in S6.

**Stabiliser of a type.** These are the permutations p of {0, …, 5} with p(e2) = e2 when e2 is not `None`, and p(e3) = e3 when e3 is not `None`. E2 and E3 are never exchanged. Orbits of a type are orbits of this group acting on α-labellings.

**String keys.** In run.json, types appear as Python tuple strings, for example `"(None, 0, None)"` and `"(0, 2, 1)"`.

## 2. Edge order and leaf codes

**Edge order E.** This order is used for every 15-character code:

```
(0,1),(0,2),(1,2),(0,3),(1,3),(2,3),(0,4),(1,4),(2,4),(3,4),(0,5),(1,5),(2,5),(3,5),(4,5)
```

These are the pairs (i,j) with i < j, sorted by (j, i).

**Codes.** A labelling is the 15-character string of its α-values, each in {1, 2, 3, 4}, written in edge order E.

**T-consistent leaves.** For a fixed type, a labelling is a *T-consistent leaf* if every one of the 20 triangles passes rule T (rules-k6-v1 L1).

- For each edge e of a triangle, take the vertex z opposite e and the table entry gcd(α_e, r_z):
  - α = 1: 1.
  - α = 2: 2 if z = E2, else 1.
  - α = 3: 3 if z = E3, else 1.
  - α = 4: 4 if z = E2 with c2 = 2; 2 if z = E2 with c2 = 1; else 1.
- **τ_min** is the lcm over the three edges e of α_e / gcd(α_e, r_z).
- **τ_max** is the minimum, over the three vertices, of the product of the two α's at that vertex.
- **T passes** if τ_min ≤ τ_max.

Every labelling that fails T on some triangle is killed by that triangle, by rules-k6-v1 L1. No further certificate is needed for it.

**What run.json reports per type** (under `summary[type]`):
- `leaves_labelled`: the number of T-consistent leaves.
- `pruned_nodes`: the number of pruned nodes. This count belongs to the engine's search order and does not need to be matched.
- `leafcodes_sha256`: the sha256 of the UTF-8 string formed by sorting all T-consistent leaf codes ascending, joining them with "\n", and appending a final "\n".
- `orbits`: the number of stabiliser-orbits of T-consistent leaves.
- `orbit_rule_counts`: counts by rule.

## 3. orbits.jsonl.gz

The file is gzip-compressed. It has one JSON object per line, and one line for every orbit of every type:

```
{"type":[e2,c2,e3], "code":"<engine canonical code>", "rep":"<15-char labelled representative, edge order E>",
 "rule":"K|RLP|PB|Ppair|FH5|N26|RPLP|SURVIVE", "witness":[...]}   (no "witness" key when rule is SURVIVE)
```

- `code` is the engine's canonical form. A checker should use its own canonical form and check that the representatives (`rep`) meet each orbit exactly once.
- The rule fields below all refer to `rep`, read as a labelled pattern of the given type.

### "K", witness `[v, S]`

This is rules-k6-v1 L3 with K1.
- **Check:** Σ_{i∈S} α_iv ≥ 5, and τ_min(i,v,j) = α_iv·α_vj for every pair {i,j} ⊆ S. Here τ_min is computed on the triangle {i, v, j} as in §2.
- **Proves:** the pattern is impossible, because L3 requires Σ ≤ 4.

### "RLP", witness `[v]`

This is rules-k6-v1 L7 with rules-k6-v2 L9.

**Neighbours.** Write N for the five vertices other than v, in ascending order. For each j ∈ N, set st_j = (E2-class digit, E3 flag). The class digit is 0 if j is not E2 and c2 if j is E2. The flag is 1 if j = E3 and 0 otherwise.

**Admissible q.** Every q ∈ ∏_{j∈N} {1, …, 5 − α_vj}, enumerated in itertools.product order with the first neighbour varying slowest.

**Allowed supports S** (S is a non-empty subset of positions 0..4 of N; every subset must be considered):
1. **Exponents.** For each position x: e2(x) = v₂(q_x) + {0 if digit 0; 1 if digit 1; SAT₂ if digit 2}, and e3(x) = v₃(q_x) + {SAT₃ if flag 1}. Here SAT₂ ≥ 3 and SAT₃ ≥ 2 suffice for RLP and RPLP; any larger value, e.g. 30, gives the same gcds (saturation, erratum-v3 C5/C16).
2. **maxgcd.** maxgcd(x,y) = 2^min(e2(x),e2(y)) · 3^min(e3(x),e3(y)).
3. **Allowed.** S is allowed if |S| = 1, or if every subset T ⊆ S with |T| ≥ 2 contains a pair with maxgcd ≥ |T|.

**The dual.** In `rlp_duals.json`, under key = `"q_1,...,q_5|d1f1;d2f2;...;d5f5"` (digits of st_j in neighbour order), there must be y = list of 5 exact rationals as strings such that:
- y_x ≥ 0 for every x;
- Σ_{x∈S} y_x ≤ 1 for **every** allowed S;
- Σ_x (q_x/5)·y_x ≥ 1.

**Proves:** for every admissible q, the covering LP minimum is ≥ 1 by weak duality. So vertex v is impossible.

`value` in `rlp_duals.json` is informational. The check is the inequalities above.

### "PB", witness `[v, S]`, and "Ppair", witness `[v, [j,k]]`

This is rules-k6-v1 L4.

**Per pair** {j,k} ⊆ N, sizes in units of |G|:
- x_j = α_vj/5.
- u = 1/(5·gcd(r_v, α_jk)), with the gcd read from the §2 table taking z = v.
- **lo:** let Lb = max(τ_min(v,j,k)/(5α_jk), x_j + x_k − 1, 0). Then lo = u·⌈Lb/u⌉, the least multiple of u that is ≥ Lb.
- **hi:** start from hi = u·⌊min(x_j, x_k)/u⌋, the greatest multiple of u that is ≤ min(x_j, x_k). If hi = min(x_j, x_k), lower it by one u when the containment is barred:
  - if x_j < x_k, barred unless α_vj + α_kj ≤ 4;
  - if x_k < x_j, barred unless α_vk + α_jk ≤ 4;
  - if x_j = x_k, barred unless both inequalities hold.

**Ppair:** check lo > hi for the witnessed pair.

**PB:** every pair of N has lo ≤ hi (otherwise a Ppair would apply), and
  Σ_{j∈S}(1 − x_j) − Σ_{{j,k}⊆S}(1 − x_j − x_k + hi_jk) ≥ 1.

**Proves:** the pattern is impossible.

### "FH5", witness `[[a,b], V, counts]`

This is rules-k6-v1 L5 and L6 with rules-k6-v2 L9.

**Hypotheses and setup:**
- α_ab = 1, and V = {v ∉ {a,b} : α_va = α_vb} is non-empty.
- There are m = 5 cosets, labelled 0..4.

**Coset choices for each vertex l:**
- l = a: fixed to t = 1 and coset set {0}.
- l = b, or l ∈ V with α_la = 1: t = 1, with each of the 5 singletons.
- l ∈ V with α_la = c > 1: t = c, with all C(5,c) subsets.
- Every other vertex: t ∈ {1..5}, with all t-subsets.

**Rejection rules.** An assignment is rejected iff (i) or (ii) holds:
- (i) (F3) fails: some vertex i with t_i = 1 has α_il > t_l for some l ≠ i.
- (ii) Some coset c has a member set M_c = {l : c ∈ set_l} of size ≥ 2 that contains a subset T, |T| ≥ 2, in which no pair has fgcd ≥ |T|.

Here fgcd(x,y) = 2^min · 3^min · 5^min of the exponents of λ_l = r_l·t_l:
- 2-exponent: v₂(t) + {1 if l = E2 with c2 = 1; SAT if l = E2 with c2 = 2};
- 3-exponent: v₃(t) + {SAT if l = E3};
- 5-exponent: v₅(t) + {SAT₅ if l = e5}, where **SAT₅ ≥ 1** is required. In FH5, SAT₂ ≥ 2, SAT₃ ≥ 1 and SAT₅ ≥ 1 are required. The value 30 is safe for all three. With SAT₅ = 0 a checker would underestimate fgcd and reject an assignment that may be real, which is unsound.

**The certificate.** Every assignment is rejected, for each of the seven placements e5 ∈ {None, 0, …, 5}. `counts` lists the number of assignments examined per placement. These counts are informational and depend on the engine's loop order.

### "N26", witness `[v]`

This is rules-k6-v3 L11. The check:
- v ≠ E2;
- α_vj = 2 for all j ≠ v;
- α_jk ∈ {2, 4} for all pairs j, k ≠ v;
- If v = E3: τ_max(j,k,l) < 3·τ_min(j,k,l) for **every** triple j, k, l ≠ v.
  - Why this suffices: every α among the neighbours lies in {2, 4}, so every τ_min is a power of 2, and τ is a multiple of τ_min that is ≤ τ_max.
  - The possible τ by triangle type are listed below. "At most one 4-edge" means that at most one 4-edge has its divisibility reduced by E2.

    | triangle | τ_min | possible τ |
    |---|---|---|
    | 2,2,2 | 2 | {2, 4} |
    | 2,2,4 | 2 or 4 | {2, 4} or {4} |
    | 2,4,4 | 4 | {4, 8} |
    | 4,4,4 | 4 | {4, 8, 12, 16} |

  - So the only τ divisible by 3 is 12, in a 4,4,4 triangle. The test fails exactly for such triangles: τ_max = 16 ≥ 12 = 3·τ_min.

### "RPLP", witness `[v, certs]`

This is rules-k6-v3 L12.

**Shape of `certs`.** It has one entry per admissible q at v, in the itertools.product order defined under RLP.
- **["R", key]:** key is the RLP key for this q, and its dual must pass the RLP check above.
- **["P", vec]:** vec is a list of 30 exact rationals as strings, giving (a_j for j ∈ N ascending; b_j for j ∈ N ascending; u_p for pairs p of N in lexicographic order; l_p in the same pair order). The checks below apply.

**Exponents and allowed S** are as for RLP, with q.

**Forced containments.** These are the pairs (w,j) of N with τ_min(v,w,j) ≥ α_vw·α_wj.

**Columns.** All (A, S) such that:
- A ⊆ N is non-empty;
- no forced pair (w,j) has j ∈ A and w ∉ A;
- S ⊆ A, where S is empty, a singleton, or hereditary-allowed.

**Checks for ["P", vec]:**
- u_p ≥ 0 and l_p ≥ 0.
- **Every column:** Σ_{j∈S} a_j + Σ_{j∈A} b_j − Σ_{p⊆A} u_p + Σ_{p⊆A} l_p ≤ 1.
- **Objective:** Σ_j a_j(q_j/5) + Σ_j b_j(5 − α_vj)/5 − Σ_p u_p·yhi_p + Σ_p l_p·ylo_p ≥ 1, where for p = {j,k}:
  - yhi_p = 1 − (α_vj + α_vk)/5 + hi_jk;
  - ylo_p = 1 − (α_vj + α_vk)/5 + lo_jk;
  - lo and hi are as defined under PB.

**Proves:** for every admissible q, the joint LP minimum is ≥ 1, or the LP is infeasible (weak duality, or Farkas). So vertex v is impossible.

### "SURVIVE"

No rule killed the orbit. Such orbits are listed in survivors.json; the expected number is 0.
