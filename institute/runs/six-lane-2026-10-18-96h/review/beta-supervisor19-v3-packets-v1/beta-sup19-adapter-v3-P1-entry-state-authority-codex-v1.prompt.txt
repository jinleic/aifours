 367|        """At most one reader attempt per path per call; the current completed observation must match the bytes."""

 375|            if len(entry['data']) > cap:

 377|            if role not in entry['row']['roles']:

 434|def build_beta_inputs(request_ref, reader, deadline_check):
 435|    """Return READY canonical expectation/observation/context records or HELD with ordered causes and no records."""
 436|    state = _State(reader, deadline_check)
