 41|def _exact(left, right):
 42|    if type(left) is not type(right):
 43|        return False
 44|    if type(left) is dict:
 45|        return set(left) == set(right) and all(_exact(left[k], right[k]) for k in left)

 47|        return len(left) == len(right) and all(_exact(a, b) for a, b in zip(left, right))

 51|def _json(data):
 52|    def pairs(rows):
 53|        value = {}
 54|        for key, entry in rows:
 55|            if key in value:
 56|                raise SnapshotRefusal('duplicate JSON key')

 60|        raise SnapshotRefusal('nonfinite JSON')

 62|        value = float(text)

189|def _exact(left, right):
190|    """Iterative typed JSON equality with no depth cap; reject cycles but allow repeated acyclic aliases."""
191|    stack = [(left, right, False)]
192|    active = set()
193|    while stack:
194|        a, b, leave = stack.pop()
195|        pair = (id(a), id(b))
196|        if leave:
197|            active.remove(pair)
198|            continue
199|        if type(a) is not type(b):
200|            return False
201|        if type(a) in (dict, list):
202|            if pair in active:
203|                return False

208|                        or any(type(k) is not str or k not in b for k in a)):

211|            else:
