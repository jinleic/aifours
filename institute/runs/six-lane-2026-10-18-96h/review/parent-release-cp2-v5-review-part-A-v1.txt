899: def _attempt(api, state, call, attempt):
900:     """Start before acquisition; end once on success or abort. Setup/journal failure is explicitly poisoned, not
901:     claimed to have been journaled. No retries of a failed end-journal operation."""
902:     rows, work = [], None
903:     try:
904:         state.journal('attempt_start', {'call': call, 'attempt': attempt})
905:         work = {'call': call, 'attempt': attempt, 'status': 'ABORTED', 'listed': None, 'identified': 0, 'missing_n': 0,
906:                 'missing': [], 'last_pid': None, 'observer_pid': None, 'observer_ok': False, 'rows_sha256': None,
907:                 'error': None, 'error_chain': []}
908:         me = _native(state, 'getpid', None, os.getpid)
909:         work['observer_pid'] = me
910:         before = _identify(api, me, state)
911:         pids = _all_pids(api, state)
912:         work['listed'] = len(pids)
913:         if len(pids) > MAX_ROWS:
914:             raise CensusRefusal('census process cap')
915:         for pid in pids:
916:             work['last_pid'] = pid
917:             try:
918:                 rows.append(_identify(api, pid, state))
919:             except _NativeError as exc:
920:                 if state.poisoned:
921:                     raise
922:                 work['missing_n'] += 1
923:                 if len(work['missing']) < MAX_MISSING_SAMPLE:
924:                     work['missing'].append([pid, _why(exc)])
925:             work['identified'] = len(rows)
926:         after = _identify(api, me, state)
927:         mine = [p for p in rows if p['pid'] == me]
928:         ok = (len(mine) == 1 and _sv19_fingerprint(before) == _sv19_fingerprint(after) == _sv19_fingerprint(mine[0]))
929:         complete = work['missing_n'] == 0 and ok
930:         work.update(status='COMPLETE' if complete else 'INCOMPLETE', observer_ok=ok,
931:                     rows_sha256=hashlib.sha256(_canonical(rows)).hexdigest())
932:     except BaseException as exc:
933:         state.poisoned = True
934:         state.keep(exc)
935:         if work is not None:
936:             _abort_end(state, work, rows, exc)
937:         raise
938:     # Successful end journaling outside the handler prevents a second attempt on journal failure.
939:     state.journal('attempt_end', work)
940:     return {'complete': complete, 'observer_ok': ok, 'rows': rows, 'before': before, 'after': after,
941:             'observer_pid': me, 'listed': len(pids), 'missing_n': work['missing_n'], 'missing': list(work['missing']),
942:             'rows_sha256': work['rows_sha256']}

1027: def _request(request):
1028:     """Bounded detach (no aliasing, no arbitrary objects/cycles) and exact request validation."""
1029:     stack, nodes = [(request, 0)], 0
1030:     while stack:
1031:         value, depth = stack.pop()
1032:         nodes += 1
1033:         _require(nodes <= REQUEST_NODES and depth <= 8, 'request node/depth cap')
1034:         if type(value) is dict:
1035:             _require(len(value) <= REQUEST_NODES, 'request object size')
1036:             for key, item in list(value.items()):
1037:                 _require(type(key) is str and len(key) <= 64, 'request key')
1038:                 stack.append((item, depth + 1))
1039:         elif type(value) is list:
1040:             _require(len(value) <= REQUEST_NODES, 'request list size')
1041:             stack.extend((item, depth + 1) for item in value)
1042:         elif type(value) is str:
1043:             _require(len(value) <= 4096, 'request string length')
1044:         else:
1045:             _require(value is None or type(value) is bool or (type(value) is int and abs(value) <= UINT64),
1046:                      'request scalar type')
1047:     data = _canonical(request)
1048:     _require(len(data) <= REQUEST_CAP, 'request byte cap')
1049:     request = _json(data)
1050:     _keys(request, ('format', 'item', 'histories'))
1051:     _require(request['format'] == REQUEST_FORMAT, 'request format')
1052:     _text(request['item'])
1053:     histories = request['histories']
1054:     _require(type(histories) is list and 0 < len(histories) <= MAX_HISTORIES, 'histories list 1..16')
1055:     for h in histories:
1056:         _keys(h, HISTORY_REQUEST_KEYS)
1057:         _require(h['family'] in ('main', 'oracle'), 'family main|oracle')
1058:         _ref(h['producer'])
1059:         _ref(h['wait_record'])

1061:             if h[key] is not None:
1062:                 _ref(h[key])

1321: def _baseline_scan(recon, events, baseline):
1322:     """M3 witness from RECORDED sv19 evidence only (no inferred time): the leading ADMISSION identity scans are the
1323:     establish_baseline loop; every one but the last carries a coverage stamp in its window, the last none; no
1324:     observation straddles the last window, none inside it is flagged unproved, and the uniqueids observed inside it
1325:     EQUAL the recorded baseline set, one observation each. Returns that scan window."""
1326:     attempts, coverage = recon['observation_attempts'], recon['coverage_history']
1327:     _require(type(attempts) is list and type(coverage) is list, 'baseline witness: attempt/coverage lists')
1328:     lead = []
1329:     for attempt in attempts:
1330:         if not (type(attempt) is dict and attempt.get('kind') == 'identity_scan'
1331:                 and attempt.get('phase') == 'ADMISSION' and attempt.get('inside_helper') is False):
1332:             break
1333:         lead.append(attempt)
1334:     _require(1 <= len(lead) <= 3, 'baseline witness: 1..3 leading ADMISSION identity scans required')
1335:     stamps = [entry.get('mono') if type(entry) is dict else None for entry in coverage]
1336:     previous = None

1338:         start, end = attempt.get('mono_start'), attempt.get('mono_end')
1339:         _require(attempt.get('aborted') is False and attempt.get('error') is None and _finite(start)
1340:                  and _finite(end) and start <= end and (previous is None or previous <= start),
1341:                  'baseline witness: scan attempt window')
1342:         previous = end
1343:         gapped = any(not _finite(mono) or start <= mono <= end for mono in stamps)
1344:         _require(gapped == (index < len(lead) - 1), 'baseline witness: only the last leading scan is gap-free')
1345:     start, end = lead[-1]['mono_start'], lead[-1]['mono_end']
1346:     inside = []
1347:     for uid, _variant, _status, begin, finish, unproved in events:
1348:         if finish < start or begin > end:
1349:             continue
1350:         _require(start <= begin and finish <= end, 'baseline witness: observation straddles the scan window')
1351:         _require(not unproved, 'baseline witness: unproved observation inside the baseline scan')
1352:         inside.append(uid)
1353:     count = lead[-1].get('identified_count')
1354:     _require(set(inside) == baseline and len(inside) == len(baseline) and _int(count) and count == len(baseline),
1355:              'baseline witness: scan window uniqueids != recorded baseline set (equality required)')

1928: def _build_union(request, read_bytes, deadline_check, retain=None):
