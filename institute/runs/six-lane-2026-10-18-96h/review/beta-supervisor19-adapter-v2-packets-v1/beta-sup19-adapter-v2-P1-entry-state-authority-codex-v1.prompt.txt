 301|class _State:
 302|    """Per-call evidence state. Private coherent cache: each path is acquired once per call."""
 303|
 304|    def __init__(self, reader, deadline_check):
 305|        self.reader, self.check = reader, deadline_check
 306|        self.causes, self.evidence = set(), []
 307|        self.cache, self.consumed = {}, []
 308|        self.stopped = False
 309|
 310|    def hold(self, cause, detail):
 311|        if cause not in HELD_CAUSES:
 312|            cause = 'BRIDGE_INTERNAL_EXCEPTION'
 313|        self.causes.add(cause)
 314|        if len(self.evidence) < EVIDENCE_ROWS:
 315|            self.evidence.append({'cause': cause, 'detail': str(detail)[:DETAIL_CAP]})
 316|
 317|    def tick(self):
 318|        if self.stopped:
 319|            raise _Stop()
 320|        try:
 321|            value = self.check()
 322|        except Exception as exc:
 323|            self.stopped = True
 324|            self.hold('DEADLINE', 'deadline_check raised ' + type(exc).__name__)
 325|            raise _Stop()
 326|        if value is not None:
 327|            self.stopped = True
 328|            self.hold('DEADLINE', 'deadline_check returned a non-None value')
 329|            raise _Stop()
 330|
 331|    def stage(self, function, *args):
 332|        """A _Held records its cause and yields None; _Stop, other Exceptions and BaseExceptions propagate."""
 333|        try:
 334|            return function(self, *args)
 335|        except _Held as held:
 336|            self.hold(held.cause, held.detail)
 337|            return None
 338|
 339|    def acquire(self, role, path, cap, cause):
 340|        """One actual reader acquisition per path; the current completed observation must match the bytes."""
 341|        if not _norm_abs(path):
 342|            raise _Held(cause, role + ': acquisition path is not normalized absolute')
 343|        self.tick()
 344|        entry = self.cache.get(path)
 345|        if entry is not None:
 346|            if len(entry['data']) > cap:
 347|                raise _Held('READ_FAILED', role + ': cached acquisition exceeds a stricter cap')
 348|            if role not in entry['row']['roles']:
 349|                entry['row']['roles'].append(role)
 350|            return entry['data'], entry['after']
 351|        try:
 352|            data = self.reader.read_bytes(path, cap)
 353|        except Exception as exc:
 354|            raise _Held('READ_FAILED', role + ': reader refused or failed: ' + type(exc).__name__)
 355|        try:
 356|            meta = self.reader.last_file(path)
 357|        except Exception as exc:
 358|            raise _Held('READ_FAILED', role + ': last_file failed: ' + type(exc).__name__)
 359|        after = meta.get('after') if type(meta) is dict else None
 360|        if (type(data) is not bytes or len(data) > cap or type(meta) is not dict or meta.get('path') != path
 361|                or meta.get('completed') is not True or meta.get('bytes_seen') != len(data)
 362|                or meta.get('sha256') != _sha(data) or not _metadata(after) or after['size'] != len(data)):
 363|            raise _Held('READ_FAILED', role + ': current completed acquisition not established')
 364|        after = copy.deepcopy(after)
 365|        row = {'roles': [role], 'path': path, 'sha256': _sha(data), 'length': len(data),
 366|               'metadata': {k: after[k] for k in METADATA_KEYS}}
 367|        self.cache[path] = {'data': data, 'after': after, 'row': row}
 368|        self.consumed.append(row)
 369|        self.tick()
 370|        return data, after

 382|        out = {'format': RESULT_FORMAT, 'state': 'READY' if ready else 'HELD',

 385|               'records': copy.deepcopy(records) if ready else None,

 390|               'external_prerequisites_not_verified': list(EXTERNAL_PREREQUISITES),

 396|def build_beta_inputs(request_ref, reader, deadline_check):
 397|    """Return READY canonical expectation/observation/context records or HELD with ordered causes and no records."""
 398|    state = _State(reader, deadline_check)
 399|    records, reader_unresolved, reported = None, None, False
 400|    try:
 401|        if not callable(deadline_check):
 402|            state.stopped = True
 403|            state.hold('DEADLINE', 'deadline_check is not callable')
 404|        else:
 405|            assembled = _evaluate(state, request_ref)
 406|            reader_unresolved, reported = _reader_unresolved(state), True
 407|            if assembled is not None and not state.causes:
 408|                records = _records(assembled)
 409|            state.tick()  # Original deadline AFTER complete assembly; a late result is never READY.
 410|    except _Stop:
 411|        pass
 412|    except Exception as exc:
 413|        state.hold('BRIDGE_INTERNAL_EXCEPTION', repr(exc))
 414|    if not reported:
 415|        reader_unresolved = _reader_unresolved(state)  # In-memory report only; HELD is already certain.
