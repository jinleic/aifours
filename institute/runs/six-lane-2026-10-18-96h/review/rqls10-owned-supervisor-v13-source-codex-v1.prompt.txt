1654: def sink(sup, name, pipe):
1655:     path, cap = sup.cfg[name], sup.cfg[name + "_cap_bytes"]
1656:     fd, base = sup.location(path)
1657:     out, ident = create_file(fd, base)
1658:     sup.created[path] = ident

1660:     sup.streams[name] = rec
1661: 
1662:     def run():
1663:         h = hashlib.sha256()
1664:         try:
1665:             while True:
1666:                 chunk = pipe.read1(65536)
1667:                 if not chunk:
1668:                     break
1669:                 rec["bytes_seen"] += len(chunk)
1670:                 room = max(0, cap - rec["bytes_written"])
1671:                 if room:
1672:                     b = chunk[:room]
1673:                     write_all(out, b, sup.deadline(IO_S))

1679:             rec["error"] = repr(e)
1680:         finally:
1681:             dl = sup.deadline(IO_S)
1682:             errors = []
1683:             for label, action in (("fsync", lambda: os.fsync(out)), ("close", lambda: os.close(out)), ("pipe_close", pipe.close)):
1684:                 try:
1685:                     action()

1692:             rec["mono_close_completed"] = time.monotonic()
1693:             rec["closed"] = rec["error"] is None
1694:     # Every sink permanently inherits blocked cancellation signals; only the main thread receives them.
1695:     with protected():
