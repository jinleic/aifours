1674: def sink(sup, name, pipe):
1675:     path, cap = sup.cfg[name], sup.cfg[name + "_cap_bytes"]

1677:     out, ident = create_file(fd, base)

1682:     def run():
1683:         h = hashlib.sha256()
1684:         try:
1685:             while True:
1686:                 chunk = pipe.read1(65536)
1687:                 if not chunk:
1688:                     break

1690:                 room = max(0, cap - rec["bytes_written"])
