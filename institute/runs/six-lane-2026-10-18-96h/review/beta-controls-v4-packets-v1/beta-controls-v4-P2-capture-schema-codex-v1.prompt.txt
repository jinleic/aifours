def _exact(left, right):
    """Iterative typed JSON equality with no depth cap; reject cycles but allow repeated acyclic aliases."""
    stack = [(left, right, False)]
    active = set()
    while stack:
        a, b, leave = stack.pop()
        pair = (id(a), id(b))
        if leave:
            active.remove(pair)
            continue
        if type(a) is not type(b):
            return False
        if type(a) in (dict, list):
            if pair in active:
                return False
            active.add(pair)
            stack.append((a, b, True))
            if type(a) is dict:
                if (len(a) != len(b) or any(type(k) is not str for k in b)
                        or any(type(k) is not str or k not in b for k in a)):
                    return False
                stack.extend((a[k], b[k], False) for k in a)
            else:
                if len(a) != len(b):
                    return False
                stack.extend((x, y, False) for x, y in zip(a, b))
        elif type(a) is float:
            if not math.isfinite(a) or not math.isfinite(b) or a != b:
                return False
        elif type(a) in (str, int, bool):
            if a != b:
                return False
        elif a is not None:
            return False
    return True

def _replay(gate, plan, jobs, order, calibrations, item, files, prefix, tick):
    """Replay the complete audited prefix in plan order, then check the current item's declared outputs."""
    if item in calibrations or item not in jobs:
        raise _Reject('item outside the twenty-job plan (calibrations are excluded)')
    position = order.index(item)
    if len(prefix) != position:
        raise _Reject('prefix must hold exactly the preceding plan items')
    state = {'ok': {}, 'states': {}, 'records': [], 'controls': None, 'datafiles': None}
    for index in range(position):
        tick()
        name = order[index]
        job = jobs[name]
        record = prefix[index]
        if type(record) is not dict or any(type(key) is not str for key in record) or set(record) != PREFIX_RECORD_KEYS:
            raise _Reject('prefix record must be exactly {item, files, audit_bytes}')
        if type(record['item']) is not str or record['item'] != name:
            raise _Reject('prefix order differs from plan order')
        record_files = _declared_files(job, record['files'], gate['CAP'])
        audit_bytes = record['audit_bytes']
        if type(audit_bytes) is not bytes or len(audit_bytes) > AUDIT_CAP:
            raise _Reject('prefix audit_bytes type/cap')
        hashes = {output: gate['digest'](record_files[output]) for output in job['outputs']}
        tick()
        _check_audit(gate, job, audit_bytes, hashes)
        _check_job(gate, plan, order, job, record_files, state, tick)
        state['ok'][name] = True
        state['records'].append({'item': name, 'problems': [], 'artifacts': hashes})
    tick()
    current = _declared_files(jobs[item], files, gate['CAP'])
    _check_job(gate, plan, order, jobs[item], current, state, tick)
    tick()
    return True


def _declared_files(job, files, cap):
    """Private copy holding exactly the job's declared outputs as immutable bytes within the gate artifact cap."""
    outputs = job['outputs']
    if (type(outputs) is not list or not outputs or any(type(name) is not str for name in outputs)
            or len(set(outputs)) != len(outputs)):
        raise _Reject('plan declared outputs')
    if type(files) is not dict or any(type(name) is not str for name in files) or set(files) != set(outputs):
        raise _Reject('files must map exactly the declared output names')
    copy = {}
    for name in outputs:
        if not name or '/' in name or '\x00' in name or name in ('.', '..'):
            raise _Reject('declared output must be a plain name')
        data = files[name]
        if type(data) is not bytes or len(data) > cap:
            raise _Reject('declared output must be bytes within the gate artifact cap')
        copy[name] = data
    return copy

        raise _Reject('audit keys')

              'verdict': 'AUDITED_PASS', 'settled': True, 'exit_code': job['exit']}

            raise _Reject('audit ' + key)

        raise _Reject('audit output digests differ from the supplied declared outputs')
    if not gate['hex64'](audit['receipt_sha256']):
        raise _Reject('audit receipt_sha256 must be lowercase 64-hex')
    if any(type(audit[key]) is not str for key in ('auditor', 'at', 'notes')) or not audit['auditor'] or not audit['at']:

class EvidenceReader:
    """One parent-owned audit snapshot, with durable in-memory acquisition and close history.

    Ledger entries are retained before the operation and never removed. NOT_ACQUIRED is reserved
    for an os.open/scandir OSError return; an unexpected acquisition exception stays UNKNOWN.
    A close exception is UNKNOWN even when the number later disappears or is reused. No retry.
    This is not a process supervisor. Passing bytes into consumers remains a separate parent act.
    """
    def __init__(self, deadline_check):
        if not callable(deadline_check):
            raise ValueError('deadline_check must be callable')
        self._check = deadline_check
        self._resources = []
        self._observations = []
        self._last = {}
        self._calls = 0

    def _tick(self):
        if self._check() is not None:
            raise EvidenceRefusal('deadline_check must return None or raise')

    def _begin(self, kind, path):
        if len(self._resources) >= MAX_RESOURCES:
            raise EvidenceRefusal('retained resource ledger cap')
        rec = {'id': len(self._resources) + 1, 'kind': kind, 'path': path,
               'state': 'ACQUIRING', 'fd': None, 'close_attempted': False,
               'acquisition_error': None, 'close_error': None}
        self._resources.append(rec)
        return rec

    def _open(self, path, flags, parent=None, display=None):
        rec = self._begin('fd', display if display is not None else path)
        try:
            fd = os.open(path, flags) if parent is None else os.open(path, flags, dir_fd=parent)
        except OSError as exc:
            rec['state'], rec['acquisition_error'] = 'NOT_ACQUIRED', repr(exc)
            raise
        except BaseException as exc:
            rec['state'], rec['acquisition_error'] = 'UNKNOWN', repr(exc)
            raise
        rec['fd'], rec['state'] = fd, 'OWNED'  # no observation or clock may precede this retention
        return rec

    def _close(self, rec):
        if rec['state'] != 'OWNED' or rec['close_attempted']:
            return
        rec['close_attempted'] = True
        try:
            if rec['kind'] == 'fd':
                os.close(rec['fd'])
            else:
                rec['handle'].close()
        except BaseException as exc:
            rec['state'], rec['close_error'] = 'UNKNOWN', repr(exc)
        else:
            rec['state'] = 'CLOSED'  # destructive close result is retained before any next operation

    def _finish(self, owned):
        for rec in reversed(owned):
            self._close(rec)
        if any(r['state'] not in ('CLOSED', 'NOT_ACQUIRED') for r in owned):
            raise EvidenceRefusal('unresolved acquisition/close; inspect reader report')
        self._tick()

    def _start(self):
        self._calls += 1
        if self._calls > MAX_CALLS:
            raise EvidenceRefusal('snapshot call limit')
        self._tick()

    def _directory(self, parts, owned):
        root = self._open('/', os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW)
        owned.append(root)
        current = root['fd']
        location = ''
        for part in parts:
            self._tick()
            location += '/' + part
