MAX_CAP = 16 << 20

class EvidenceRefusal(OSError):
    pass


def _path_parts(path):
    if type(path) is not str or not path.startswith('/') or '\x00' in path:
        raise EvidenceRefusal('normalized absolute path required')
    parts = path.split('/')[1:]

        raise EvidenceRefusal('normalized path/component limit')
    return parts


def _metadata(info):
    return {'dev': info.st_dev, 'ino': info.st_ino, 'mode': info.st_mode, 'nlink': info.st_nlink,
            'size': info.st_size, 'mtime_ns': info.st_mtime_ns, 'ctime_ns': info.st_ctime_ns}


def _same_file(before, after):
    return all(before[k] == after[k] for k in ('dev', 'ino', 'mode', 'nlink', 'size', 'mtime_ns', 'ctime_ns'))


class EvidenceReader:
    """One parent-owned audit snapshot, with durable in-memory acquisition and close history.

    Ledger entries are retained before the operation and never removed. NOT_ACQUIRED is reserved
    for an os.open/scandir OSError return; an unexpected acquisition exception stays UNKNOWN.
    A close exception is UNKNOWN even when the number later disappears or is reused. No retry.
    This is not a process supervisor. Passing bytes into consumers remains a separate parent act.

    def __init__(self, deadline_check):
        if not callable(deadline_check):
            raise ValueError('deadline_check must be callable')
        self._check = deadline_check
        self._resources = []
        self._observations = []
        self._last = {}
        self._calls = 0

    def _tick(self):
        if self._check() is not None:
            raise EvidenceRefusal('deadline_check must return None or raise')

    def _begin(self, kind, path):
        if len(self._resources) >= MAX_RESOURCES:
            raise EvidenceRefusal('retained resource ledger cap')
        rec = {'id': len(self._resources) + 1, 'kind': kind, 'path': path,
               'state': 'ACQUIRING', 'fd': None, 'close_attempted': False,
               'acquisition_error': None, 'close_error': None}
        self._resources.append(rec)
        return rec

    def _open(self, path, flags, parent=None, display=None):
        rec = self._begin('fd', display if display is not None else path)
        try:
            fd = os.open(path, flags) if parent is None else os.open(path, flags, dir_fd=parent)
        except OSError as exc:
            rec['state'], rec['acquisition_error'] = 'NOT_ACQUIRED', repr(exc)
            raise
        except BaseException as exc:
            rec['state'], rec['acquisition_error'] = 'UNKNOWN', repr(exc)
            raise
        rec['fd'], rec['state'] = fd, 'OWNED'  # no observation or clock may precede this retention
        return rec

    def _close(self, rec):
        if rec['state'] != 'OWNED' or rec['close_attempted']:
            return
        rec['close_attempted'] = True
        try:
            if rec['kind'] == 'fd':
                os.close(rec['fd'])
            else:
                rec['handle'].close()
        except BaseException as exc:
            rec['state'], rec['close_error'] = 'UNKNOWN', repr(exc)
        else:
            rec['state'] = 'CLOSED'  # destructive close result is retained before any next operation

        for rec in reversed(owned):
            self._close(rec)
        if any(r['state'] not in ('CLOSED', 'NOT_ACQUIRED') for r in owned):
            raise EvidenceRefusal('unresolved acquisition/close; inspect reader report')
        self._tick()

**FINDINGS**

1. **FATAL — P1-F01: RD05E can MATCH without a crossing after native return.**  
   **Location:** v2 771–777, 780–793; P074; production `EvidenceReader.read_bytes()` native read and following deadline check.  
   If the native data read returns after the one-second control deadline, but before the harness’s 300-second deadline, `sleep_crossing()` enters with its deadline already expired. Its loop executes zero times, yet it sets `real_hook_ran=True`. The resulting observation and `real_crossing_after_native_return=True` can reproduce the frozen expected object, although expiry occurred before the hook. Lines 784–785 only detect a hook that never ran.  
   **Consequence:** MATCH can falsely attribute the specifically claimed post-return crossing.  
   **Minimal repair:** At hook entry, conservatively require the unchanged control deadline still to be unexpired; otherwise raise `InconclusivePrecondition`. Set the crossing witness only after witnessing the transition from an unexpired entry.

2. **FATAL — P1-F02 / CF03: emergency reporting can replace the propagating BaseException and skip bootstrap cleanup.**  
   **Location:** v2 1649–1660 and 1698–1715, especially 1656–1659 and 1709–1715.  
   An `_emit()` failure becomes the propagating exception. In bootstrap, it also bypasses `owned.close_since(0)`. In the per-case handler, cleanup runs, but the subsequent bare `raise` is never reached when emission fails. Report acquisition/retention also lacks complete protection against a secondary BaseException.  
   **Consequence:** The original BaseException need not propagate; bootstrap descriptors can remain without their required cleanup attempt. This exceeds the proposed MINOR severity.  
   **Minimal repair:** Make emergency reporting best effort under explicit primary-exception preservation, guarantee cleanup independently of reporting success, and re-propagate the original exception.

3. **FATAL — P1-F03: direct fixture-close finalizers can suppress an earlier BaseException.**  
   **Location:** v2 532–541 and 558–565; `_Owned.close()` at 246–255.  
   Both finalizers directly call a close method that raises. If writing is already propagating a BaseException and close raises an ordinary exception, close replaces the primary. The run loop can then classify the replacement as ERROR and continue, retaining neither the original exception nor its full evidence in the emitted result.  
   **Consequence:** A primary interrupt can disappear into ordinary error handling.  
   **Minimal repair:** Preserve an active primary exception while latching close uncertainty, retaining the single close attempt and prohibiting retries.

4. **FATAL — P1-F04: a newly arising cleanup BaseException can be swallowed.**
