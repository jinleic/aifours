def _tick(check):

def _hex(value):
    return type(value) is str and len(value) == 64 and all(c in '0123456789abcdef' for c in value)


def _exact(left, right):
    if type(left) is not type(right):
        return False
    if type(left) is dict:
        return set(left) == set(right) and all(_exact(left[k], right[k]) for k in left)
    if type(left) is list:
        return len(left) == len(right) and all(_exact(a, b) for a, b in zip(left, right))
    return left == right


def _json(data):
    def pairs(rows):
        value = {}
        for key, entry in rows:
            if key in value:
                raise SnapshotRefusal('duplicate JSON key')
            value[key] = entry
        return value
    def bad(_):
        raise SnapshotRefusal('nonfinite JSON')
    def number(text):
        value = float(text)
        if not math.isfinite(value):
            bad(text)
        return value
    if type(data) is not bytes or len(data) > CAP:
        raise SnapshotRefusal('JSON type/cap')
    return json.loads(data.decode('utf-8'), object_pairs_hook=pairs,
                      parse_constant=bad, parse_float=number)

class _Reader:
    """Calls read_bytes(path, READ_CAP) at most once per bound path; any failure yields None, never bytes."""

    def __init__(self, read_bytes):
        self._read_bytes = read_bytes if callable(read_bytes) else None
        self._cache = {}

    def read(self, path):
        if not _norm_abs(path):
            return None
        if path not in self._cache:
            data = None
            if self._read_bytes is not None:
                try:
                    data = self._read_bytes(path, READ_CAP)
                except READ_FAILURES:
                    data = None
                if type(data) is not bytes or len(data) > READ_CAP:
                    data = None
            self._cache[path] = data
        return self._cache[path]

def _exact(left, right):
    """Iterative typed JSON equality with no depth cap; reject cycles but allow repeated acyclic aliases."""
    stack = [(left, right, False)]
    active = set()
    while stack:
        a, b, leave = stack.pop()
        pair = (id(a), id(b))
        if leave:
            active.remove(pair)
            continue
        if type(a) is not type(b):
            return False
        if type(a) in (dict, list):
            if pair in active:
                return False
            active.add(pair)
            stack.append((a, b, True))
            if type(a) is dict:
                if (len(a) != len(b) or any(type(k) is not str for k in b)
                        or any(type(k) is not str or k not in b for k in a)):
                    return False
                stack.extend((a[k], b[k], False) for k in a)
            else:
                if len(a) != len(b):
                    return False
                stack.extend((x, y, False) for x, y in zip(a, b))
        elif type(a) is float:
            if not math.isfinite(a) or not math.isfinite(b) or a != b:
                return False
        elif type(a) in (str, int, bool):
            if a != b:
                return False
        elif a is not None:
            return False
    return True

    return type(record) is dict and all(type(k) is str for k in record) and set(record) == set(keys)
