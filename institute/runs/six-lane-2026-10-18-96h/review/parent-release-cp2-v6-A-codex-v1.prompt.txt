706: def _native(state, stage, pid, function, *args, **kwargs):
707:     """One native acquisition between ORIGINAL checks. Every primary and secondary object retained. The failing
708:     completion check runs INSIDE the primary's handler, so interpreter implicit context links retain it without
709:     calling a foreign exception's __context__ property. No Python-level linking/overwriting.
710:     V6 selection: a non-Exception primary (KeyboardInterrupt/SystemExit/...) outranks an ORDINARY completion

714:     _check(state, stage, pid, 'before')

716:         value = function(*args, **kwargs)

912: def _attempt(api, state, call, attempt):
913:     """Start before acquisition; end once on success or abort. Setup/journal failure is explicitly poisoned, not
914:     claimed to have been journaled. No retries of a failed end-journal operation."""
915:     rows, work = [], None
916:     try:
917:         state.journal('attempt_start', {'call': call, 'attempt': attempt})
918:         work = {'call': call, 'attempt': attempt, 'status': 'ABORTED', 'listed': None, 'identified': 0, 'missing_n': 0,
919:                 'missing': [], 'last_pid': None, 'observer_pid': None, 'observer_ok': False, 'rows_sha256': None,
920:                 'error': None, 'error_chain': []}
921:         me = _native(state, 'getpid', None, os.getpid)
922:         work['observer_pid'] = me
923:         before = _identify(api, me, state)
924:         pids = _all_pids(api, state)
925:         work['listed'] = len(pids)
926:         if len(pids) > MAX_ROWS:
927:             raise CensusRefusal('census process cap')
928:         for pid in pids:
929:             work['last_pid'] = pid
930:             try:
931:                 rows.append(_identify(api, pid, state))
932:             except _NativeError as exc:
933:                 if state.poisoned:
934:                     raise
935:                 work['missing_n'] += 1
936:                 if len(work['missing']) < MAX_MISSING_SAMPLE:
937:                     work['missing'].append([pid, _why(exc)])
938:             work['identified'] = len(rows)
939:         after = _identify(api, me, state)
940:         mine = [p for p in rows if p['pid'] == me]
941:         ok = (len(mine) == 1 and _sv19_fingerprint(before) == _sv19_fingerprint(after) == _sv19_fingerprint(mine[0]))
942:         complete = work['missing_n'] == 0 and ok
943:         work.update(status='COMPLETE' if complete else 'INCOMPLETE', observer_ok=ok,
944:                     rows_sha256=hashlib.sha256(_canonical(rows)).hexdigest())
945:     except BaseException as exc:
946:         state.poisoned = True
947:         state.keep(exc)
948:         if work is not None:
949:             _abort_end(state, work, rows, exc)
950:         raise
951:     # Successful end journaling outside the handler prevents a second attempt on journal failure.
952:     state.journal('attempt_end', work)
953:     return {'complete': complete, 'observer_ok': ok, 'rows': rows, 'before': before, 'after': after,
954:             'observer_pid': me, 'listed': len(pids), 'missing_n': work['missing_n'], 'missing': list(work['missing']),
955:             'rows_sha256': work['rows_sha256']}

969: def make_census(deadline_check):
970:     """Return an UNBOUND (census, log). CP1 must bind its S1 nonce immediately before invoking PRT. One call only;
971:     every failed pre-seal call, refused bind, reentry or journal failure poisons, even if the caller catches it.
972:     ORIGINAL deadline/callback identities and nonce are checked during evaluation. No helper/process/signal/cache.
973:     """
974:     _require(callable(deadline_check), 'deadline_check must be callable')
975:     state = _Transaction(deadline_check)
976:     state.pid = _native(state, 'getpid', None, os.getpid)
977:     log = CensusLog(_ISSUE_KEY)
978:     _ISSUED[log] = state
979:     bound = []
980: 
981:     def census():
982:         if state.sealed:
983:             state.post_seal += 1
984:             raise CensusRefusal('post-seal census refused; immutable journal unchanged')
985:         if state.active:
986:             state.poisoned = True
987:             state.reentries += 1
988:             error = CensusRefusal('reentry poisons even if caught; no acquisition')
989:             state.keep(error)
990:             raise error
991:         state.active = True
992:         try:
993:             state.calls += 1
994:             if state.token is None or state.poisoned:
995:                 raise CensusRefusal('census unbound or poisoned')
996:             if state.calls > MAX_CALLS:
997:                 raise CensusRefusal('one census call per factory')
998:             _check(state, 'census start')
999:             if not bound:
1000:                 bound.append(_bind_libproc(state))
1001:             chosen, attempt = None, 0
1002:             while attempt < ATTEMPTS_PER_CALL:
1003:                 attempt += 1
1004:                 chosen = _attempt(bound[0], state, state.calls, attempt)
1005:                 if chosen['complete']:
1006:                     break
1007:                 if attempt < ATTEMPTS_PER_CALL:
1008:                     _native(state, 'retry pause', None, time.sleep, RETRY_PAUSE_S)
1009:             observed = _native(state, 'monotonic', None, time.monotonic)
1010:             if not _finite(observed):
1011:                 raise CensusRefusal('observed_at_mono unavailable; never fabricated')
1012:             rows = chosen['rows']
1013:             served = {'complete': chosen['complete'],
1014:                       'processes': [{'pid': p['pid'], 'ppid': p['ppid'], 'pgid': p['pgid'], 'uid': p['euid'],
1015:                                      'start_fingerprint': start_fingerprint(p['pid'], p['uniqueid'], p['idversion'])}
1016:                                     for p in rows], 'observed_at_mono': observed}
1017:             record = {'call': state.calls, 'attempts': attempt, 'complete': chosen['complete'],
1018:                       'observer_ok': chosen['observer_ok'], 'observer_pid': chosen['observer_pid'],
1019:                       'factory_pid': state.pid, 'observer_before': chosen['before'], 'observer_after': chosen['after'],
1020:                       'listed': chosen['listed'], 'missing_n': chosen['missing_n'], 'missing': chosen['missing'],
1021:                       'process_count': len(rows), 'rows': rows,
1022:                       'row_sha256': [hashlib.sha256(_canonical(p)).hexdigest() for p in rows],
1023:                       'rows_sha256': chosen['rows_sha256'], 'served_sha256': hashlib.sha256(_canonical(served)).hexdigest(),
1024:                       'observed_at_mono': observed, 'release_token': state.token}

1030:             _refusal(state, exc)

1032:         finally:

1040: def _request(request):
1041:     """Bounded detach (no aliasing, no arbitrary objects/cycles) and exact request validation."""
1042:     stack, nodes = [(request, 0)], 0
1043:     while stack:
1044:         value, depth = stack.pop()
1045:         nodes += 1
1046:         _require(nodes <= REQUEST_NODES and depth <= 8, 'request node/depth cap')
1047:         if type(value) is dict:
1048:             _require(len(value) <= REQUEST_NODES, 'request object size')
1049:             for key, item in list(value.items()):
1050:                 _require(type(key) is str and len(key) <= 64, 'request key')
1051:                 stack.append((item, depth + 1))
1052:         elif type(value) is list:
1053:             _require(len(value) <= REQUEST_NODES, 'request list size')
1054:             stack.extend((item, depth + 1) for item in value)
1055:         elif type(value) is str:
1056:             _require(len(value) <= 4096, 'request string length')
1057:         else:
1058:             _require(value is None or type(value) is bool or (type(value) is int and abs(value) <= UINT64),
1059:                      'request scalar type')
1060:     data = _canonical(request)
1061:     _require(len(data) <= REQUEST_CAP, 'request byte cap')
1062:     request = _json(data)
1063:     _keys(request, ('format', 'item', 'histories'))
1064:     _require(request['format'] == REQUEST_FORMAT, 'request format')
1065:     _text(request['item'])
1066:     histories = request['histories']
1067:     _require(type(histories) is list and 0 < len(histories) <= MAX_HISTORIES, 'histories list 1..16')
1068:     for h in histories:
1069:         _keys(h, HISTORY_REQUEST_KEYS)
1070:         _require(h['family'] in ('main', 'oracle'), 'family main|oracle')
1071:         _ref(h['producer'])
1072:         _ref(h['wait_record'])
1073:         for key in ('launch_record', 'receipt', 'settlement'):
1074:             if h[key] is not None:
1075:                 _ref(h[key])
1076:         _capture_record(h['stdout'])
1077:     return request

1334: def _baseline_scan(recon, events, baseline):
1335:     """M3 witness from RECORDED sv19 evidence only (no inferred time): the leading ADMISSION identity scans are the
1336:     establish_baseline loop; every one but the last carries a coverage stamp in its window, the last none; no
1337:     observation straddles the last window, none inside it is flagged unproved, and the uniqueids observed inside it
1338:     EQUAL the recorded baseline set, one observation each. Returns that scan window."""
1339:     attempts, coverage = recon['observation_attempts'], recon['coverage_history']
1340:     _require(type(attempts) is list and type(coverage) is list, 'baseline witness: attempt/coverage lists')
1341:     lead = []
1342:     for attempt in attempts:
1343:         if not (type(attempt) is dict and attempt.get('kind') == 'identity_scan'
1344:                 and attempt.get('phase') == 'ADMISSION' and attempt.get('inside_helper') is False):
1345:             break
1346:         lead.append(attempt)
1347:     _require(1 <= len(lead) <= 3, 'baseline witness: 1..3 leading ADMISSION identity scans required')
1348:     stamps = [entry.get('mono') if type(entry) is dict else None for entry in coverage]
1349:     previous = None
1350:     for index, attempt in enumerate(lead):
1351:         start, end = attempt.get('mono_start'), attempt.get('mono_end')
1352:         _require(attempt.get('aborted') is False and attempt.get('error') is None and _finite(start)
1353:                  and _finite(end) and start <= end and (previous is None or previous <= start),
1354:                  'baseline witness: scan attempt window')
1355:         previous = end
1356:         gapped = any(not _finite(mono) or start <= mono <= end for mono in stamps)
1357:         _require(gapped == (index < len(lead) - 1), 'baseline witness: only the last leading scan is gap-free')
1358:     start, end = lead[-1]['mono_start'], lead[-1]['mono_end']
1359:     inside = []
1360:     for uid, _variant, _status, begin, finish, unproved in events:
1361:         if finish < start or begin > end:
1362:             continue
1363:         _require(start <= begin and finish <= end, 'baseline witness: observation straddles the scan window')
1364:         _require(not unproved, 'baseline witness: unproved observation inside the baseline scan')
1365:         inside.append(uid)
1366:     count = lead[-1].get('identified_count')
1367:     _require(set(inside) == baseline and len(inside) == len(baseline) and _int(count) and count == len(baseline),
1368:              'baseline witness: scan window uniqueids != recorded baseline set (equality required)')
1369:     return start, end

1941: def _build_union(request, read_bytes, deadline_check, retain=None):
1942:     """One pinned builder for initial union AND every evaluation rebuild. Ordinary errors inside setup, evaluation,
1943:     assembly and finalization return HELD/MINIMAL/PREBUILT; interpreter exception-delivery boundary in module docstring.
1944:     retain (optional exact list) keeps actual exception objects32; partial attachment also keeps these on propagation."""
1945:     st, io = None, None
1946:     try:
1947:         st, io = _State(U_CAUSES), _Io(read_bytes, deadline_check, retain)
1948:         item, analyses, members, unknowns, rows = None, [], [], [], None
1949:         try:
1950:             if type(retain) is not list or not callable(read_bytes) or not callable(deadline_check):
1951:                 st.fire('U_CALLBACKS_INVALID', 'exact retain list and callable reader/ORIGINAL deadline required')
1952:             elif not io.tick('union start'):
1953:                 st.fire('U_DEADLINE', 'ORIGINAL deadline unavailable at union start')

1955:                 detached = _guard(io, st, 'U_REQUEST_INVALID', 'request', None, _request, request)

## V6 change: native exception priority (the only source delta)

v5 `_native` re-raised the completion-check secondary whenever one existed. If the native call raised a non-Exception primary P (KeyboardInterrupt, SystemExit or any other BaseException that is not an Exception) and the ORIGINAL completion check raised an ordinary secondary S, PRT could catch S as an ordinary failure and absorb P (**H09**, recorded as the v5 SELF-HOLD). v6 `_native` (source lines 706-728):

```
    except BaseException as primary:
        try:
            _check(state, stage, pid, 'after failed')
        except BaseException as secondary:
            state.keep(primary)
            state.keep(secondary)
            if _ordinary(primary) or not _ordinary(secondary):
                raise  # secondary selected; interpreter context keeps the primary
        state.keep(primary)  # identity-deduplicated: completion passed or the primary outranks an ordinary secondary
        raise
```

| native primary P | after-failed completion check | propagates |
|---|---|---|
| any | passes | P (as v5) |
| Exception | raises ordinary S | S, with S.__context__ = P (as v5) |
| Exception | raises non-Exception S | S (as v5) |
| non-Exception | raises ordinary S | **P (v6; v5 propagated S)** |
| non-Exception | raises non-Exception S | S (as v5) |

- Both actual objects stay in the transaction's retained list in every branch (`log.exceptions()`). `_retain` is identity-deduplicated, so the repeated `keep(primary)` stores P once.
- P is selected by a bare `raise` after the inner handler ends. That re-raises the currently handled P without interpreter context chaining. A literal `raise primary` inside the secondary handler (the H09 record's repair wording) would make the interpreter set `P.__context__ = S` and clear `S.__context__` to break the cycle; v6 rewrites neither. S keeps `S.__context__ = P`.
- Every ordinary `_check` failure latches `state.poisoned = True` before raising (lines 689-703), so selecting P loses no fail-closed latch. `_attempt` (line 945) and the census wrapper's `_refusal` (line 958) also poison.
- `_ordinary` is `issubclass(type(exc), Exception)`: no foreign method, property or override is consulted.
- The reader path (`_Io.tick` 589-598, `_Io.settle_read` 600-614) already had this priority and is unchanged: a non-Exception S propagates, else P, and an ordinary S latches `crossed`. When both are ordinary the reader propagates P and `_native` propagates S; both outcomes are nonpositive.
- Controls: H09 and K11 now state the v6 behaviour; H11-H16 isolate every branch of the table; K29 documents the availability consequence below.

## Exception-priority sites audited (unchanged in v6)

- `_check` (689-703), `CensusLog.bind` (861-876) and `_Transaction.journal` (829-843) each handle one exception: keep it, poison, bare re-raise.
- `_abort_end` (897-909) and `_refusal` (958-966) catch journal failures only as `Exception`, keep them and never replace the propagating exception. They skip journaling once retention has failed.
- The `_attempt` per-pid handler (930-937) catches only `_NativeError` and re-raises it when the transaction is poisoned.
- The census wrapper (1029-1033) calls `_refusal` and then re-raises bare; its `finally` only clears `active`.
- Evaluator reader and diagnostic paths keep the v5 M3/SR6 rules (controls C43, E26, E41, E42, H06-H08).

## Limitations carried or newly documented (no code change)

- When P outranks S, the attempt_end and refusal journal entries describe P's chain only. S survives as an actual retained object, and in the retained count, but not as a journal description. Outside minimal scope; the transaction is nonpositive either way.
- **K29.** Every ordinary identification failure (a pid that vanished between listing and identification) is a retained object. Thirty in one transaction, across up to 3 attempts, stop admission: the census refuses (HELD), and that attempt gets no attempt_end and no refusal entry, even where a later attempt might have completed. This fail-closed availability cost comes from the approved cap32/stop30 interface and is unchanged since v5.
- SR4 (dependent row-conflict noise) and SR5 (no structured request binding inside the union) remain out of scope. There is no forgery protection; parent code and observation provenance stay prerequisites.

## D8 consumed-snapshot join (no CP2 interface change)

D8 (79540a18) is now REQUIRED for CP1. CP2 already provides what it needs:

- `census()` returns the exact served dict `{'complete': bool, 'processes': [{'pid', 'ppid', 'pgid', 'uid': int, 'start_fingerprint': str}, ...], 'observed_at_mono': finite float}`.
- `log.served_sha256()` and `summary()['served_sha256']` hold `sha256(_canonical(served))`, where `_canonical(v) = (json.dumps(v, sort_keys=True, ensure_ascii=True, allow_nan=False, separators=(',', ':')) + '\n').encode('ascii')`. The digest is set when the served entry is journaled, before the final ORIGINAL check, so it can exist on a refused call: it is NOT success.
- The evaluator's `census.served_sha256` must equal the digest recomputed from the journaled rows and the transaction's served digest (`_census_evidence`).
- CP2's row cap MAX_ROWS = 200000 equals PRT's row limit; more listed pids raise CensusRefusal('census process cap').
- The evaluator must receive the ORIGINAL inner census, the ORIGINAL deadline and the bound nonce, never CP1's wrapper; otherwise E_CENSUS_BINDING.

## Approved CP1/census interface

Main approved this SOURCE interface; CP1 accepted its call ordering, retention and emergency rules before code. Not runtime or pin acceptance. v6 keeps it unchanged; only the module and design-freeze values in item 6 change.
