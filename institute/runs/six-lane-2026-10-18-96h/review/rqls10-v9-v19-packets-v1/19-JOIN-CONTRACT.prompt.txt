484|def ray_count(Q, v):
485|    """(number of rays after transitive merging at 1 - |<q,q'>|^2 < 1e-7, minsep_nonclose_pair = minimum separation
486|    over pairs that are NOT below the closeness threshold; such a pair may still lie in one merged component)."""
487|    V = Q.reshape(v * v, v)
488|    V = V / np.linalg.norm(V, axis=1, keepdims=True)
489|    O = np.abs(np.conj(V) @ V.T) ** 2
490|    n = v * v
491|    parent = list(range(n))
492|
493|    def find(a):
494|        while parent[a] != a:
495|            parent[a] = parent[parent[a]]
496|            a = parent[a]
497|        return a
498|    iu = np.triu_indices(n, 1)
499|    sep = 1.0 - O[iu]
500|    close = sep < 1e-7
501|    for a, b in zip(iu[0][close], iu[1][close]):
502|        parent[find(int(a))] = find(int(b))
503|    nonclose = sep[~close]
504|    return len({find(a) for a in range(n)}), (fnum(nonclose.min()) if nonclose.size else 1.0)

648|def dihedral5():

650|    idx = {g: t for t, g in enumerate(el)}

670|def jac_selftest(Pr, rng):
671|    U = rand_U(Pr.v, rng)
672|    th = rng.standard_normal(Pr.P)
673|    JTJ, JTr, res, Jr = Pr.normal(U)
674|    T = th.reshape(Pr.v - 1, Pr.nb)
675|    loc = np.zeros((Pr.npairs, 2 * Pr.nb))
676|    for k, (i, ip) in enumerate(Pr.pairs):
677|        if i >= 1:
678|            loc[k, :Pr.nb] = T[i - 1]
679|        loc[k, Pr.nb:] = T[ip - 1]
680|    Jt = np.einsum("nra,na->nr", Jr, loc)
681|    eps = 1e-4
682|    A = Pr.A_of(th)
683|
684|    def pert(s):
685|        U2 = U.copy()
686|        U2[1:] = U[1:] @ (np.eye(Pr.v) + s * eps * A)
687|        rc = Pr.resid(U2)
688|        return np.concatenate([rc.real, rc.imag], axis=1)
689|    fd = (pert(1.0) - pert(-1.0)) / (2 * eps)
690|    e_fd = float(np.max(np.abs(Jt - fd)) / max(1.0, float(np.max(np.abs(Jt)))))
691|    e_ne = None
692|    if Pr.v <= 7:
693|        Jd = np.zeros((Pr.npairs * 4 * Pr.v, Pr.P))
694|        for k, (i, ip) in enumerate(Pr.pairs):
695|            rows = slice(k * 4 * Pr.v, (k + 1) * 4 * Pr.v)
696|            if i >= 1:
697|                Jd[rows, (i - 1) * Pr.nb:i * Pr.nb] = Jr[k, :, :Pr.nb]
698|            Jd[rows, (ip - 1) * Pr.nb:ip * Pr.nb] = Jr[k, :, Pr.nb:]
699|        rr = np.concatenate([res.real, res.imag], axis=1).reshape(-1)
700|        e_ne = max(float(np.max(np.abs(Jd.T @ Jd - JTJ)) / max(1.0, float(np.max(np.abs(JTJ))))),
701|                   float(np.max(np.abs(Jd.T @ rr - JTr)) / max(1.0, float(np.max(np.abs(JTr))))))
702|    U3 = Pr.retract(U, 0.3 * th / max(1.0, float(np.linalg.norm(th))))
703|    e_un = float(np.max(np.abs(np.conj(U3).transpose(0, 2, 1) @ U3 - np.eye(Pr.v))))
704|    return {"fd": fnum(e_fd), "normal_eq": None if e_ne is None else fnum(e_ne), "retract_unitarity": fnum(e_un)}
