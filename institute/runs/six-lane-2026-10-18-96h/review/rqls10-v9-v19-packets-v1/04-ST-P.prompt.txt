201|def snapshot(fdroot, name, budget, cap=MAX_BYTES):
202|    budget.check()
203|    before = os.stat(name, dir_fd=fdroot, follow_symlinks=False)
204|    if not stat.S_ISREG(before.st_mode) or not 0 <= before.st_size <= cap or before.st_nlink != 1:
205|        raise Invalid("not a bounded one-link regular file")
206|    fd = os.open(name, os.O_RDONLY | os.O_NOFOLLOW | os.O_NONBLOCK, dir_fd=fdroot)
207|    try:
208|        if signature(os.fstat(fd)) != signature(before):
209|            raise Invalid("changed open")
210|        data = bytearray()
211|        while len(data) < before.st_size:
212|            budget.check()
213|            part = os.read(fd, min(65536, before.st_size - len(data)))
214|            if not part:
215|                raise Invalid("short read")
216|            data.extend(part)
217|        if os.read(fd, 1) or signature(os.fstat(fd)) != signature(before) or \
218|                signature(os.stat(name, dir_fd=fdroot, follow_symlinks=False)) != signature(before):
219|            raise Invalid("changed read")
220|    finally:
221|        os.close(fd)
222|    budget.check()
223|    raw = bytes(data)
224|    return {"raw": raw, "sha256": hashlib.sha256(raw).hexdigest(), "bytes": len(raw),
225|            "inode": [before.st_dev, before.st_ino], "nlink": before.st_nlink,
226|            "mode": stat.S_IMODE(before.st_mode)}

349|    def reading(self, clock):
350|        self.model.guard()

352|        self.model.checkpoint(stack)
353|        value = self.mono if clock == "monotonic" else self.epoch_origin + self.mono - MONO + self.civil_shift
354|        for rule in self.rules:
355|            if rule["fired"] or rule["clock"] != clock or not self.model.matches(stack, rule):
356|                continue
357|            rule["seen"] += 1
358|            if rule["seen"] != rule["occurrence"]:
359|                continue
360|            rule["fired"] = True
361|            rule["mono_before"] = self.mono
362|            if rule["action"] == "set":

365|            elif rule["action"] == "epoch":
366|                value = rule["set_to"]
367|                self.civil_shift = value - (self.epoch_origin + self.mono - MONO)
368|            elif rule["action"] == "raise":
