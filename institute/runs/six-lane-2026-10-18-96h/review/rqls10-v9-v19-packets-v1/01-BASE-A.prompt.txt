510|def snapshot_fd(fd, path, cap, deadline):
511|    check_deadline(deadline, "before regular snapshot")
512|    before = os.fstat(fd)
513|    if not stat.S_ISREG(before.st_mode) or before.st_nlink != 1 or not 0 <= before.st_size <= cap:
514|        raise ValueError("not a bounded single-link regular file")
515|    raw = bytearray()
516|    while len(raw) < before.st_size:
517|        check_deadline(deadline, "regular snapshot read")
518|        part = os.pread(fd, min(65536, before.st_size - len(raw)), len(raw))
519|        if not part:
520|            raise OSError("short regular snapshot")
521|        raw.extend(part)
522|    if os.pread(fd, 1, before.st_size) or stat_signature(os.fstat(fd)) != stat_signature(before):
523|        raise ValueError("regular file changed during snapshot")
524|    check_deadline(deadline, "after regular snapshot")
525|    return {"path": path, "inode": [before.st_dev, before.st_ino], "sha256": hashlib.sha256(raw).hexdigest(),
526|            "bytes": len(raw), "nlink": before.st_nlink}, bytes(raw)

380|def read_bounded(dir_fd, name, cap, budget, keep=False):
381|    """Bounded no-follow regular-file read relative to a retained directory descriptor; any change refuses."""
382|    st = os.stat(name, dir_fd=dir_fd, follow_symlinks=False)
383|    if not stat.S_ISREG(st.st_mode) or not 0 <= st.st_size <= cap:
384|        raise OSError("not a bounded regular file " + name)
385|    fd = os.open(name, os.O_RDONLY | os.O_NOFOLLOW | os.O_NONBLOCK, dir_fd=dir_fd)
386|    h, parts, got = hashlib.sha256(), [], 0
387|    try:
388|        fs = os.fstat(fd)
389|        if (fs.st_dev, fs.st_ino, fs.st_size) != (st.st_dev, st.st_ino, st.st_size):
390|            raise OSError("file changed at open " + name)
391|        while got < st.st_size:
392|            b = os.read(fd, min(1 << 20, st.st_size - got))
393|            if not b:
394|                raise OSError("short read " + name)
395|            h.update(b)
396|            got += len(b)
397|            if keep:
398|                parts.append(b)
399|            budget.check("read " + name)
400|        tail = os.read(fd, 1)
401|        end = os.fstat(fd)
402|        if tail or (end.st_size, end.st_mtime_ns) != (st.st_size, st.st_mtime_ns):
403|            raise OSError("file changed during read " + name)
404|    finally:
405|        os.close(fd)
406|    facts = {"sha256": h.hexdigest(), "inode": [st.st_dev, st.st_ino], "bytes": got, "nlink": st.st_nlink}
407|    return facts, (b"".join(parts) if keep else None)

425|def create_exclusive(dir_fd, name, data, mode, budget):
426|    """Exclusive no-follow creation. Recorded inode/size/nlink are fstat of the CREATED descriptor BEFORE close and
427|    the content is read back on that descriptor; a later no-follow read must still find exactly that file."""
428|    fd = os.open(name, os.O_RDWR | os.O_CREAT | os.O_EXCL | os.O_NOFOLLOW, mode, dir_fd=dir_fd)
429|    try:
430|        view = memoryview(data)
431|        while view:
432|            n = os.write(fd, view)
433|            if n <= 0:
434|                raise OSError("zero-progress write " + name)
435|            view = view[n:]
436|            budget.check("write " + name)
437|        os.fsync(fd)
438|        st = os.fstat(fd)
439|        if not stat.S_ISREG(st.st_mode) or st.st_size != len(data) or st.st_nlink != 1 or \
440|                os.pread(fd, len(data) + 1, 0) != data:
441|            raise OSError("created descriptor facts differ " + name)
442|        created = {"sha256": hashlib.sha256(data).hexdigest(), "inode": [st.st_dev, st.st_ino],
443|                   "bytes": st.st_size, "nlink": st.st_nlink}
444|    finally:
445|        os.close(fd)
446|    if not typed_equal(read_bounded(dir_fd, name, len(data), budget)[0], created):
447|        raise OSError("created file changed after close " + name)
448|    return created
