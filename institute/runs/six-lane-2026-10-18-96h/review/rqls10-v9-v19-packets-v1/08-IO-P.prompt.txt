282|    """Conservative distinct-inode RUN bound, including outgoing parent captures.
283|

285|    This is a cooperative fixture limit, not a reservation-release accountant.
286|    """
287|    if RUN is None:
288|        return

290|    def walk(path, depth=0):
291|        if depth > 16:
292|            raise OSError("FIXTURE_OUTPUT_DEPTH")
293|        fd = directory(path)
294|        total = 0
295|        try:
296|            for name in os.listdir(fd):
297|                check("bounded entire current-run output census")
298|                entries[0] += 1

834|
835|    An EBADF stimulus retains OUTPUT_LOCK then FD_LOCK until the source's last
836|    close of that number. Source sink threads never allocate except through this
837|    locked proxy/output census. Every source operation on the closed number
