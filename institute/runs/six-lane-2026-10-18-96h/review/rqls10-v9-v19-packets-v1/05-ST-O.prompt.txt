151|def snapshot(fdroot, name, budget, cap=LIMIT):
152|    budget.check()
153|    st = os.stat(name, dir_fd=fdroot, follow_symlinks=False)
154|    if not stat.S_ISREG(st.st_mode) or not 0 <= st.st_size <= cap or st.st_nlink != 1:

157|    try:
