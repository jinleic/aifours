def fifo_endpoint_refusals(rec):
    """I1-1/IP1-1 consumer schema shared by the RAM collector and the file sink: exact retained FIFO identity; the
    reader endpoint acquired and observed as that FIFO; the unblock endpoint either acquired, observed as that FIFO
    and recorded as the runner unblock, or explicitly never acquired (no number, no observation, no unblock).
    Missing, unavailable, numeric, truthy or contradictory evidence refuses."""
    fifo = rec.get("fifo_identity")
    if type(fifo) is not list or len(fifo) != 2 or any(type(x) is not int or x < 0 for x in fifo):
        return ["retained FIFO identity missing/malformed"]
    expected = {"dev": fifo[0], "ino": fifo[1], "kind": "fifo"}
    why = []
    if rec.get("reader_fd_acquired") is not True or not _observed(rec.get("reader_fd_observed_identity"), "fifo") \
            or rec["reader_fd_observed_identity"] != expected:
        why.append("reader endpoint acquisition/observed FIFO identity unproven")
    unblock = rec.get("unblock_fd_acquired")
    if unblock is True:
        if not _observed(rec.get("unblock_fd_observed_identity"), "fifo") \
                or rec["unblock_fd_observed_identity"] != expected or rec.get("unblocked_by_runner") is not True:
            why.append("acquired unblock endpoint lacks its observed FIFO identity/unblock record")
    elif unblock is False:
        if "unblock_fd_observed_identity" not in rec or rec["unblock_fd_observed_identity"] is not None \
                or "unblock_fd" not in rec or rec["unblock_fd"] is not None or rec.get("unblocked_by_runner") is not False:
            why.append("never-acquired unblock endpoint carries contradictory evidence")
    else:
        why.append("unblock endpoint acquisition state missing/not an exact boolean")
    return why

    started before started) >= C6_MIN_RUN_S; runtime upper bound (exit_observed - popen_call) <= C6_MAX_RUN_S.
