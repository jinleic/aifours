def _tick(check):

def _hex(value):
    return type(value) is str and len(value) == 64 and all(c in '0123456789abcdef' for c in value)


def _exact(left, right):
    if type(left) is not type(right):
        return False
    if type(left) is dict:
        return set(left) == set(right) and all(_exact(left[k], right[k]) for k in left)
    if type(left) is list:
        return len(left) == len(right) and all(_exact(a, b) for a, b in zip(left, right))
    return left == right


def _json(data):
    def pairs(rows):
        value = {}
        for key, entry in rows:
            if key in value:
                raise SnapshotRefusal('duplicate JSON key')
            value[key] = entry
        return value
    def bad(_):
        raise SnapshotRefusal('nonfinite JSON')
    def number(text):
        value = float(text)
        if not math.isfinite(value):
            bad(text)
        return value
    if type(data) is not bytes or len(data) > CAP:
        raise SnapshotRefusal('JSON type/cap')
    return json.loads(data.decode('utf-8'), object_pairs_hook=pairs,
                      parse_constant=bad, parse_float=number)

class _Reader:
    """Calls read_bytes(path, READ_CAP) at most once per bound path; any failure yields None, never bytes."""

    def __init__(self, read_bytes):
        self._read_bytes = read_bytes if callable(read_bytes) else None
        self._cache = {}

    def read(self, path):
        if not _norm_abs(path):
            return None
        if path not in self._cache:
            data = None
            if self._read_bytes is not None:
                try:
                    data = self._read_bytes(path, READ_CAP)
                except READ_FAILURES:
                    data = None
                if type(data) is not bytes or len(data) > READ_CAP:
                    data = None
            self._cache[path] = data
        return self._cache[path]

def _exact(left, right):
    """Iterative typed JSON equality with no depth cap; reject cycles but allow repeated acyclic aliases."""
    stack = [(left, right, False)]
    active = set()
    while stack:
        a, b, leave = stack.pop()
        pair = (id(a), id(b))
        if leave:
            active.remove(pair)
            continue
        if type(a) is not type(b):
            return False
        if type(a) in (dict, list):
            if pair in active:
                return False
            active.add(pair)
            stack.append((a, b, True))
            if type(a) is dict:
                if (len(a) != len(b) or any(type(k) is not str for k in b)
                        or any(type(k) is not str or k not in b for k in a)):
                    return False
                stack.extend((a[k], b[k], False) for k in a)
            else:
                if len(a) != len(b):
                    return False
                stack.extend((x, y, False) for x, y in zip(a, b))
        elif type(a) is float:
            if not math.isfinite(a) or not math.isfinite(b) or a != b:
                return False
        elif type(a) in (str, int, bool):
            if a != b:
                return False
        elif a is not None:
            return False
    return True

def _capture(expectation, observed, stream, reader):
    """(MISSING, None), (INCOMPLETE, None) or (COMPLETE, verified actual capture bytes)."""
    captures = _get(observed, "captures")
    record = _get(captures, stream)
    if record is _ABSENT or record is None or (type(record) is dict and _get(record, "present") is False):
        return _MISSING, None

        return _INCOMPLETE, None
    path = _bound_capture_path(expectation, stream)
    if path is None or not _keys(record, CAPTURE_KEYS) or record["present"] is not True:
        return _INCOMPLETE, None

    if (not _same_str(record["path"], path) or not _hex64(digest) or not _count(seen) or not _count(stored)
            or seen != stored or stored > READ_CAP or record["eof"] is not True
            or record["overflow"] is not False or record["error"] is not None):
        return _INCOMPLETE, None
    data = reader.read(path)
    if data is None or len(data) != stored or _sha256(data) != digest:
        return _INCOMPLETE, None
    return _COMPLETE, data

class EvidenceReader:
    """One parent-owned audit snapshot, with durable in-memory acquisition and close history.

    Ledger entries are retained before the operation and never removed. NOT_ACQUIRED is reserved
    for an os.open/scandir OSError return; an unexpected acquisition exception stays UNKNOWN.
    A close exception is UNKNOWN even when the number later disappears or is reused. No retry.
    This is not a process supervisor. Passing bytes into consumers remains a separate parent act.
    """
    def __init__(self, deadline_check):
        if not callable(deadline_check):
            raise ValueError('deadline_check must be callable')
        self._check = deadline_check
        self._resources = []
        self._observations = []
        self._last = {}
        self._calls = 0

    def _tick(self):
        if self._check() is not None:
            raise EvidenceRefusal('deadline_check must return None or raise')

    def _begin(self, kind, path):
        if len(self._resources) >= MAX_RESOURCES:
            raise EvidenceRefusal('retained resource ledger cap')
        rec = {'id': len(self._resources) + 1, 'kind': kind, 'path': path,
               'state': 'ACQUIRING', 'fd': None, 'close_attempted': False,
               'acquisition_error': None, 'close_error': None}
        self._resources.append(rec)
        return rec

    def _open(self, path, flags, parent=None, display=None):
        rec = self._begin('fd', display if display is not None else path)
        try:
            fd = os.open(path, flags) if parent is None else os.open(path, flags, dir_fd=parent)
        except OSError as exc:
            rec['state'], rec['acquisition_error'] = 'NOT_ACQUIRED', repr(exc)
            raise
        except BaseException as exc:
            rec['state'], rec['acquisition_error'] = 'UNKNOWN', repr(exc)
            raise
        rec['fd'], rec['state'] = fd, 'OWNED'  # no observation or clock may precede this retention
        return rec

    def _close(self, rec):
        if rec['state'] != 'OWNED' or rec['close_attempted']:
            return
        rec['close_attempted'] = True
        try:
            if rec['kind'] == 'fd':
                os.close(rec['fd'])
            else:
                rec['handle'].close()
        except BaseException as exc:
            rec['state'], rec['close_error'] = 'UNKNOWN', repr(exc)
        else:
            rec['state'] = 'CLOSED'  # destructive close result is retained before any next operation

    def _finish(self, owned):
        for rec in reversed(owned):
            self._close(rec)
        if any(r['state'] not in ('CLOSED', 'NOT_ACQUIRED') for r in owned):
            raise EvidenceRefusal('unresolved acquisition/close; inspect reader report')
        self._tick()

        self._calls += 1
        if self._calls > MAX_CALLS:
            raise EvidenceRefusal('snapshot call limit')
        self._tick()
