728: def _native(state, stage, pid, function, *args, **kwargs):
729:     """One native acquisition between ORIGINAL checks. Every primary and secondary object retained. The failing
730:     completion check runs INSIDE the primary's handler, so interpreter implicit context links retain it without
731:     calling a foreign exception's __context__ property. No Python-level linking/overwriting.
732:     V6 selection: a non-Exception primary (KeyboardInterrupt/SystemExit/...) outranks an ORDINARY completion
733:     secondary and propagates by bare re-raise after the secondary handler ends, so its own context is not rewritten
734:     and no caller can absorb it as an ordinary failure; otherwise the secondary propagates. Both objects stay in
735:     the transaction's retained exceptions (CensusLog.exceptions) either way."""
736:     _check(state, stage, pid, 'before')
737:     try:
738:         value = function(*args, **kwargs)
739:     except BaseException as primary:
740:         try:
741:             _check(state, stage, pid, 'after failed')
742:         except BaseException as secondary:
743:             state.keep(primary)
744:             state.keep(secondary)
745:             if _ordinary(primary) or not _ordinary(secondary):
746:                 raise  # secondary selected; interpreter context keeps the primary
747:         state.keep(primary)  # identity-deduplicated: completion passed or the primary outranks an ordinary secondary
748:         raise
749:     _check(state, stage, pid, 'after')

934: def _attempt(api, state, call, attempt):
935:     """Start before acquisition; end once on success or abort. Setup/journal failure is explicitly poisoned, not
936:     claimed to have been journaled. No retries of a failed end-journal operation."""
937:     rows, work = [], None
938:     try:
939:         state.journal('attempt_start', {'call': call, 'attempt': attempt})
940:         work = {'call': call, 'attempt': attempt, 'status': 'ABORTED', 'listed': None, 'identified': 0, 'missing_n': 0,
941:                 'missing': [], 'last_pid': None, 'observer_pid': None, 'observer_ok': False, 'rows_sha256': None,
942:                 'error': None, 'error_chain': []}
943:         me = _native(state, 'getpid', None, os.getpid)
944:         work['observer_pid'] = me
945:         before = _identify(api, me, state)
946:         pids = _all_pids(api, state)

948:         if len(pids) > MAX_ROWS:
949:             raise CensusRefusal('census process cap')

951:             work['last_pid'] = pid
952:             try:

954:             except _NativeError as exc:

956:                 if state.poisoned:
957:                     raise

960:                     work['missing'].append([pid, _why(exc)])
961:             work['identified'] = len(rows)
962:         after = _identify(api, me, state)
963:         mine = [p for p in rows if p['pid'] == me]
964:         ok = (len(mine) == 1 and _sv19_fingerprint(before) == _sv19_fingerprint(after) == _sv19_fingerprint(mine[0]))
965:         complete = work['missing_n'] == 0 and ok

1063: def _request(request):
1064:     """Bounded detach (no aliasing, no arbitrary objects/cycles) and exact request validation."""
1065:     stack, nodes = [(request, 0)], 0
1066:     while stack:
1067:         value, depth = stack.pop()
1068:         nodes += 1
1069:         _require(nodes <= REQUEST_NODES and depth <= 8, 'request node/depth cap')
1070:         if type(value) is dict:
1071:             _require(len(value) <= REQUEST_NODES, 'request object size')
1072:             for key, item in list(value.items()):
1073:                 _require(type(key) is str and len(key) <= 64, 'request key')
1074:                 stack.append((item, depth + 1))
1075:         elif type(value) is list:
1076:             _require(len(value) <= REQUEST_NODES, 'request list size')
1077:             stack.extend((item, depth + 1) for item in value)
1078:         elif type(value) is str:
1079:             _require(len(value) <= 4096, 'request string length')
1080:         else:
1081:             _require(value is None or type(value) is bool or (type(value) is int and abs(value) <= UINT64),
1082:                      'request scalar type')
1083:     data = _canonical(request)
1084:     _require(len(data) <= REQUEST_CAP, 'request byte cap')
1085:     request = _json(data)
1086:     _keys(request, ('format', 'item', 'histories'))
1087:     _require(request['format'] == REQUEST_FORMAT, 'request format')
1088:     _text(request['item'])
1089:     histories = request['histories']
1090:     _require(type(histories) is list and 0 < len(histories) <= MAX_HISTORIES, 'histories list 1..16')
1091:     for h in histories:

1357: def _baseline_scan(recon, events, baseline):
1358:     """M3 witness from RECORDED sv19 evidence only (no inferred time): the leading ADMISSION identity scans are the
1359:     establish_baseline loop; every one but the last carries a coverage stamp in its window, the last none; no
1360:     observation straddles the last window, none inside it is flagged unproved, and the uniqueids observed inside it
1361:     EQUAL the recorded baseline set, one observation each. Returns that scan window."""
1362:     attempts, coverage = recon['observation_attempts'], recon['coverage_history']
1363:     _require(type(attempts) is list and type(coverage) is list, 'baseline witness: attempt/coverage lists')
1364:     lead = []
1365:     for attempt in attempts:
1366:         if not (type(attempt) is dict and attempt.get('kind') == 'identity_scan'
1367:                 and attempt.get('phase') == 'ADMISSION' and attempt.get('inside_helper') is False):
1368:             break
1369:         lead.append(attempt)
1370:     _require(1 <= len(lead) <= 3, 'baseline witness: 1..3 leading ADMISSION identity scans required')
1371:     stamps = [entry.get('mono') if type(entry) is dict else None for entry in coverage]
1372:     previous = None
1373:     for index, attempt in enumerate(lead):
1374:         start, end = attempt.get('mono_start'), attempt.get('mono_end')
1375:         _require(attempt.get('aborted') is False and attempt.get('error') is None and _finite(start)
1376:                  and _finite(end) and start <= end and (previous is None or previous <= start),
1377:                  'baseline witness: scan attempt window')
1378:         previous = end
1379:         gapped = any(not _finite(mono) or start <= mono <= end for mono in stamps)
1380:         _require(gapped == (index < len(lead) - 1), 'baseline witness: only the last leading scan is gap-free')
1381:     start, end = lead[-1]['mono_start'], lead[-1]['mono_end']
1382:     inside = []
1383:     for uid, _variant, _status, begin, finish, unproved in events:
1384:         if finish < start or begin > end:
1385:             continue
1386:         _require(start <= begin and finish <= end, 'baseline witness: observation straddles the scan window')
1387:         _require(not unproved, 'baseline witness: unproved observation inside the baseline scan')
1388:         inside.append(uid)
1389:     count = lead[-1].get('identified_count')
1390:     _require(set(inside) == baseline and len(inside) == len(baseline) and _int(count) and count == len(baseline),
1391:              'baseline witness: scan window uniqueids != recorded baseline set (equality required)')
1392:     return start, end

1969: def _build_union(request, read_bytes, deadline_check, retain=None):
1970:     """One pinned builder for initial union AND every evaluation rebuild. Ordinary errors inside setup, evaluation,
1971:     assembly and finalization return HELD/MINIMAL/PREBUILT; interpreter exception-delivery boundary in module docstring.
1972:     retain (optional exact list) keeps actual exception objects32; partial attachment also keeps these on propagation."""
1973:     st, io = None, None

1976:         item, analyses, members, unknowns, rows = None, [], [], [], None
1977:         # V7 M1: ONE explicit callback-admission predicate gates every reader/deadline callback incl. the final check.
1978:         admitted = type(retain) is list and callable(read_bytes) and callable(deadline_check)
1979:         try:
1980:             if not admitted:
1981:                 st.fire('U_CALLBACKS_INVALID', 'exact retain list and callable reader/ORIGINAL deadline required')
1982:             elif not io.tick('union start'):

1985:                 detached = _guard(io, st, 'U_REQUEST_INVALID', 'request', None, _request, request)
1986:                 if detached is not _FAILED:

1994:                     built = _guard(io, st, 'U_ROW_CONFLICT', 'rows', None, _rows, members)

## V6 change: native exception priority (retained unchanged in v7)

v5 `_native` re-raised the completion-check secondary whenever one existed. If the native call raised a non-Exception primary P (KeyboardInterrupt, SystemExit or any other BaseException that is not an Exception) and the ORIGINAL completion check raised an ordinary secondary S, PRT could catch S as an ordinary failure and absorb P (**H09**, recorded as the v5 SELF-HOLD). v6/v7 `_native` (v7 source lines 728-750):

```
    except BaseException as primary:
        try:
            _check(state, stage, pid, 'after failed')
        except BaseException as secondary:
            state.keep(primary)
            state.keep(secondary)
            if _ordinary(primary) or not _ordinary(secondary):
                raise  # secondary selected; interpreter context keeps the primary
        state.keep(primary)  # identity-deduplicated: completion passed or the primary outranks an ordinary secondary
        raise
```

| native primary P | after-failed completion check | propagates |
|---|---|---|
| any | passes | P (as v5) |
| Exception | raises ordinary S | S, with S.__context__ = P (as v5) |
| Exception | raises non-Exception S | S (as v5) |
| non-Exception | raises ordinary S | **P (v6; v5 propagated S)** |
| non-Exception | raises non-Exception S | S (as v5) |

- Both actual objects stay in the transaction's retained list in every branch (`log.exceptions()`). `_retain` is identity-deduplicated, so the repeated `keep(primary)` stores P once.
- P is selected by a bare `raise` after the inner handler ends. That re-raises the currently handled P without interpreter context chaining. A literal `raise primary` inside the secondary handler (the H09 record's repair wording) would make the interpreter set `P.__context__ = S` and clear `S.__context__` to break the cycle; v6 rewrites neither. S keeps `S.__context__ = P`.
- Every ordinary `_check` failure latches `state.poisoned = True` before raising (v7 lines 711-725), so selecting P loses no fail-closed latch. `_attempt` (v7 line 968) and the census wrapper's `_refusal` (v7 line 981) also poison.
- `_ordinary` is `issubclass(type(exc), Exception)`: no foreign method, property or override is consulted.
- The reader path (`_Io.tick` 604-613, `_Io.settle_read` 615-631) already had this priority and is unchanged: a non-Exception S propagates, else P, and an ordinary S latches `crossed`. When both are ordinary the reader propagates P and `_native` propagates S; both outcomes are nonpositive.
- Controls: H09 and K11 now state the v6 behaviour; H11-H16 isolate every branch of the table; K29 documents the availability consequence below.
