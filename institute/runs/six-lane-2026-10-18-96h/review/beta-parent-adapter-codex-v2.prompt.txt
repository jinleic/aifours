def _tick(check):

def _hex(value):
    return type(value) is str and len(value) == 64 and all(c in '0123456789abcdef' for c in value)


def _exact(left, right):
    if type(left) is not type(right):
        return False
    if type(left) is dict:
        return set(left) == set(right) and all(_exact(left[k], right[k]) for k in left)
    if type(left) is list:
        return len(left) == len(right) and all(_exact(a, b) for a, b in zip(left, right))
    return left == right


def _json(data):
    def pairs(rows):
        value = {}
        for key, entry in rows:
            if key in value:
                raise SnapshotRefusal('duplicate JSON key')
            value[key] = entry
        return value
    def bad(_):
        raise SnapshotRefusal('nonfinite JSON')
    def number(text):
        value = float(text)
        if not math.isfinite(value):
            bad(text)
        return value
    if type(data) is not bytes or len(data) > CAP:
        raise SnapshotRefusal('JSON type/cap')
    return json.loads(data.decode('utf-8'), object_pairs_hook=pairs,
                      parse_constant=bad, parse_float=number)


def _pinned(ref, reader, check, cap=CAP):

            raise SnapshotRefusal('bundle path/cap')

        entry = self.entries[path]

"""SOURCE ONLY, UNTESTED. Parent beta descriptor-anchored evidence reader.

Scope: beta-parent-reader-scope-v1.json. No file writes, process calls, or clock helper.
The supplied deadline_check returns None or raises; its original deadline must never renew.
A synchronous kernel call may stall. A late result is retained and refused, not certified timely.
Every caller must inspect report()['unresolved']; a read error never implies resource closure.
No receipt, capture, filesystem snapshot or empty report establishes process endedness or release.
"""
import hashlib
import os
import stat

MAX_CAP = 16 << 20
MAX_CALLS = 1024
MAX_RESOURCES = 32768
MAX_COMPONENTS = 128


class EvidenceRefusal(OSError):
    pass


def _path_parts(path):
    if type(path) is not str or not path.startswith('/') or '\x00' in path:
        raise EvidenceRefusal('normalized absolute path required')
    parts = path.split('/')[1:]
    if not parts or len(parts) > MAX_COMPONENTS or any(p in ('', '.', '..') for p in parts):
        raise EvidenceRefusal('normalized path/component limit')
    return parts


def _metadata(info):
    return {'dev': info.st_dev, 'ino': info.st_ino, 'mode': info.st_mode, 'nlink': info.st_nlink,
            'size': info.st_size, 'mtime_ns': info.st_mtime_ns, 'ctime_ns': info.st_ctime_ns}


def _same_file(before, after):
    return all(before[k] == after[k] for k in ('dev', 'ino', 'mode', 'nlink', 'size', 'mtime_ns', 'ctime_ns'))


class EvidenceReader:
    """One parent-owned audit snapshot, with durable in-memory acquisition and close history.

    Ledger entries are retained before the operation and never removed. NOT_ACQUIRED is reserved
    for an os.open/scandir OSError return; an unexpected acquisition exception stays UNKNOWN.
    A close exception is UNKNOWN even when the number later disappears or is reused. No retry.
    This is not a process supervisor. Passing bytes into consumers remains a separate parent act.
    """
    def __init__(self, deadline_check):
        if not callable(deadline_check):
            raise ValueError('deadline_check must be callable')
        self._check = deadline_check
        self._resources = []
        self._observations = []
        self._last = {}
        self._calls = 0

    def _tick(self):
        if self._check() is not None:
            raise EvidenceRefusal('deadline_check must return None or raise')

    def _begin(self, kind, path):
        if len(self._resources) >= MAX_RESOURCES:
            raise EvidenceRefusal('retained resource ledger cap')
        rec = {'id': len(self._resources) + 1, 'kind': kind, 'path': path,
               'state': 'ACQUIRING', 'fd': None, 'close_attempted': False,
               'acquisition_error': None, 'close_error': None}
        self._resources.append(rec)
        return rec

    def _open(self, path, flags, parent=None, display=None):
        rec = self._begin('fd', display if display is not None else path)
        try:
            fd = os.open(path, flags) if parent is None else os.open(path, flags, dir_fd=parent)
        except OSError as exc:
            rec['state'], rec['acquisition_error'] = 'NOT_ACQUIRED', repr(exc)
            raise
        except BaseException as exc:
            rec['state'], rec['acquisition_error'] = 'UNKNOWN', repr(exc)
            raise
        rec['fd'], rec['state'] = fd, 'OWNED'  # no observation or clock may precede this retention
        return rec

    def _close(self, rec):
        if rec['state'] != 'OWNED' or rec['close_attempted']:
            return
        rec['close_attempted'] = True
        try:
            if rec['kind'] == 'fd':
                os.close(rec['fd'])
            else:
                rec['handle'].close()
        except BaseException as exc:
            rec['state'], rec['close_error'] = 'UNKNOWN', repr(exc)
        else:
            rec['state'] = 'CLOSED'  # destructive close result is retained before any next operation

    def _finish(self, owned):
        for rec in reversed(owned):
            self._close(rec)
        if any(r['state'] not in ('CLOSED', 'NOT_ACQUIRED') for r in owned):
            raise EvidenceRefusal('unresolved acquisition/close; inspect reader report')
        self._tick()

    def _start(self):
        self._calls += 1
        if self._calls > MAX_CALLS:
            raise EvidenceRefusal('snapshot call limit')
        self._tick()

    def _directory(self, parts, owned):
        root = self._open('/', os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW)
        owned.append(root)
        current = root['fd']
        location = ''
        for part in parts:
            self._tick()
            location += '/' + part
            rec = self._open(part, os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW,
                             current, location)
            owned.append(rec)
            current = rec['fd']
            self._tick()
        return current

    def read_bytes(self, path, cap):
        parts = _path_parts(path)
        if type(cap) is not int or not 0 <= cap <= MAX_CAP:
            raise EvidenceRefusal('exact integer cap in [0,16MiB] required')
        self._start()
        observation = {'path': path, 'kind': 'file', 'cap': cap, 'bytes_seen': 0,
                       'eof': False, 'before': None, 'after': None, 'sha256': None,
                       'completed': False, 'error': None, 'resource_ids': []}
        self._observations.append(observation)
        owned = []
        try:
            parent = self._directory(parts[:-1], owned)
            rec = self._open(parts[-1], os.O_RDONLY | os.O_NONBLOCK | os.O_NOFOLLOW,
                             parent, path)
            owned.append(rec)
            before = os.fstat(rec['fd'])
            observation['before'] = _metadata(before)
            self._tick()
            if not stat.S_ISREG(before.st_mode) or before.st_nlink != 1 or before.st_size > cap:
                raise EvidenceRefusal('regular one-link file within cap required')
            chunks = []
            while True:
                self._tick()
                block = os.read(rec['fd'], min(65536, cap + 1 - observation['bytes_seen']))
                observation['bytes_seen'] += len(block)
                if not block:
                    observation['eof'] = True
                else:
                    chunks.append(block)
                self._tick()
                if observation['bytes_seen'] > cap:
                    raise EvidenceRefusal('actual byte cap exceeded')
                if not block:
                    break
            after = os.fstat(rec['fd'])
            observation['after'] = _metadata(after)
            self._tick()
            if not _same_file(observation['before'], observation['after']) or after.st_size != observation['bytes_seen']:
                raise EvidenceRefusal('file identity/link/size/time changed while read')
            data = b''.join(chunks)
            observation['sha256'] = hashlib.sha256(data).hexdigest()
            self._tick()
        except BaseException as exc:
            observation['error'] = repr(exc)
            raise
        finally:
            observation['resource_ids'] = [r['id'] for r in owned]
            self._finish(owned)
        observation['completed'] = True
        self._last[path] = observation
        return data

    def directory_names(self, path, cap=256):
        parts = _path_parts(path)
        if type(cap) is not int or not 0 <= cap <= 256:
            raise EvidenceRefusal('exact directory entry cap in [0,256] required')
        self._start()
        observation = {'path': path, 'kind': 'directory', 'cap': cap, 'names': [],
                       'before': None, 'after': None, 'completed': False, 'error': None,
                       'resource_ids': []}
        self._observations.append(observation)
        owned = []
        try:
            fd = self._directory(parts, owned)
            before = os.fstat(fd)
            observation['before'] = _metadata(before)
            self._tick()
            rec = self._begin('scandir', path)
            owned.append(rec)
            try:
                iterator = os.scandir(fd)
            except OSError as exc:
                rec['state'], rec['acquisition_error'] = 'NOT_ACQUIRED', repr(exc)
                raise
            except BaseException as exc:
                rec['state'], rec['acquisition_error'] = 'UNKNOWN', repr(exc)
                raise
            rec['handle'], rec['state'] = iterator, 'OWNED'
            self._tick()
            for entry in iterator:
                observation['names'].append(entry.name)
                self._tick()
                if len(observation['names']) > cap:
                    raise EvidenceRefusal('directory entry cap exceeded')
            after = os.fstat(fd)
            observation['after'] = _metadata(after)
            self._tick()
            if not _same_file(observation['before'], observation['after']):
                raise EvidenceRefusal('directory changed during census')
        except BaseException as exc:
            observation['error'] = repr(exc)
            raise
        finally:
            observation['resource_ids'] = [r['id'] for r in owned]
            self._finish(owned)
        observation['completed'] = True
        return sorted(observation['names'])

    def last_file(self, path):
        """Return a copy of the latest successful actual metadata; never a failed observation or guessed inode."""
        import copy
        return copy.deepcopy(self._last.get(path))

    def report(self):
        """JSON-shaped snapshot; handles stay owned in this instance, never serialized as success evidence."""
        import copy
        rows = [{k: v for k, v in r.items() if k != 'handle'} for r in self._resources]
        unresolved = [r['id'] for r in rows if r['state'] not in ('CLOSED', 'NOT_ACQUIRED')]
        return copy.deepcopy({'resources': rows, 'observations': self._observations,
                              'unresolved': unresolved, 'calls': self._calls})

class _Reader:
    """Calls read_bytes(path, READ_CAP) at most once per bound path; any failure yields None, never bytes."""

    def __init__(self, read_bytes):
        self._read_bytes = read_bytes if callable(read_bytes) else None
        self._cache = {}

    def read(self, path):
        if not _norm_abs(path):
            return None
        if path not in self._cache:
            data = None
            if self._read_bytes is not None:
                try:
                    data = self._read_bytes(path, READ_CAP)
                except READ_FAILURES:
                    data = None
                if type(data) is not bytes or len(data) > READ_CAP:
                    data = None
            self._cache[path] = data
        return self._cache[path]


def _get(record, key):

def _sha256(data):
    return hashlib.sha256(data).hexdigest()


def _hex64(value):
    return type(value) is str and len(value) == 64 and set(value) <= HEX_DIGITS


def _count(value):
    return type(value) is int and value >= 0


def _same_str(value, want):

def _norm_abs(path):
    """Absolute normalized path as in ext4_gate anchored_parent: leading '/', no '', '.', '..' component, no NUL."""
    if type(path) is not str or not path.startswith("/") or "\x00" in path:
        return False
    return all(part not in ("", ".", "..") for part in path.split("/")[1:])


def _plain_name(name):
    return type(name) is str and name not in ("", ".", "..") and "/" not in name and "\x00" not in name


def _names(value):
    """Exact nonempty list of distinct plain names, else None; no uncontracted count limit."""
    if type(value) is not list or len(value) == 0:
        return None
    if not all(_plain_name(name) for name in value) or len(set(value)) != len(value):
        return None
    return value


def _exact(left, right):
    """Iterative typed JSON equality with no depth cap; reject cycles but allow repeated acyclic aliases."""
    stack = [(left, right, False)]
    active = set()
    while stack:
        a, b, leave = stack.pop()
        pair = (id(a), id(b))
        if leave:
            active.remove(pair)
            continue
        if type(a) is not type(b):
            return False
        if type(a) in (dict, list):
            if pair in active:
                return False
            active.add(pair)
            stack.append((a, b, True))
            if type(a) is dict:
                if (len(a) != len(b) or any(type(k) is not str for k in b)
                        or any(type(k) is not str or k not in b for k in a)):
                    return False
                stack.extend((a[k], b[k], False) for k in a)
            else:
                if len(a) != len(b):
                    return False
                stack.extend((x, y, False) for x, y in zip(a, b))
        elif type(a) is float:
            if not math.isfinite(a) or not math.isfinite(b) or a != b:
                return False
        elif type(a) in (str, int, bool):
            if a != b:
                return False
        elif a is not None:
            return False
    return True

def _keys(record, keys):
    """Exact built-in JSON string keys on every schema boundary; str subclasses never alias authority."""
    return type(record) is dict and all(type(k) is str for k in record) and set(record) == set(keys)

def _capture(expectation, observed, stream, reader):
    """(MISSING, None), (INCOMPLETE, None) or (COMPLETE, verified actual capture bytes)."""
    captures = _get(observed, "captures")
    record = _get(captures, stream)
    if record is _ABSENT or record is None or (type(record) is dict and _get(record, "present") is False):
        return _MISSING, None
    if type(captures) is not dict or any(type(key) is not str or key not in STREAM_KEYS for key in captures):
        return _INCOMPLETE, None
    path = _bound_capture_path(expectation, stream)
    if path is None or not _keys(record, CAPTURE_KEYS) or record["present"] is not True:
        return _INCOMPLETE, None
    seen, stored, digest = record["bytes_seen"], record["bytes_stored"], record["sha256"]
    if (not _same_str(record["path"], path) or not _hex64(digest) or not _count(seen) or not _count(stored)
            or seen != stored or stored > READ_CAP or record["eof"] is not True
            or record["overflow"] is not False or record["error"] is not None):
        return _INCOMPLETE, None
    data = reader.read(path)
    if data is None or len(data) != stored or _sha256(data) != digest:
        return _INCOMPLETE, None
    return _COMPLETE, data

def _artifact_ok(record, path, reader):
    if not _keys(record, ARTIFACT_KEYS):
        return False
    size, digest, ino, nlink = record["bytes"], record["sha256"], record["ino"], record["nlink"]
    if (not _same_str(record["path"], path) or not _hex64(digest) or not _count(size) or size > READ_CAP
            or type(ino) is not int or ino <= 0 or type(nlink) is not int or nlink != 1):
        return False
    data = reader.read(path)
    return data is not None and len(data) == size and _sha256(data) == digest

    settlement = _get(observed, "settlement")
    if not _keys(settlement, SETTLEMENT_KEYS):

            and settlement["census_complete"] is True and type(unresolved) is list and len(unresolved) == 0
            and type(signals) is list and len(signals) == 0)


def _provenance_ok(expectation, observed, reader):
    ok = _record_ok(expectation, EXPECTATION_KEYS, EXPECTATION_FORMAT)
    if not _record_ok(observed, OBSERVATION_KEYS, OBSERVATION_FORMAT):
        ok = False

        ok = False
    item = _get(expectation, "item")
    if type(item) is not str or not _same_str(_get(observed, "item"), item):
        ok = False
    plan = _get(expectation, "plan_sha256")
    if not _same_str(plan, AUTHORITY_PLAN_SHA256) or not _same_str(_get(observed, "plan_sha256"), plan):
        ok = False

        ok = False

    if (type(argv) is not list or not all(type(arg) is str for arg in argv)

        ok = False
    go = _get(expectation, "go_sha256")
    if not _hex64(go) or not _same_str(_get(observed, "go_sha256"), go):
        ok = False

        if not _pins_bound(_get(expectation, key), _get(observed, key), reader):
            ok = False
    for key in ("date_start", "date_end"):
        record = _get(expectation, key)
        if type(record) is not dict or len(record) == 0 or not _exact(record, _get(observed, key)):
            ok = False
    return ok


def _record_ok(record, keys, form):
    return _keys(record, keys) and _same_str(record["format"], form)


def _receipt_ok(expectation, observed, reader):
    path, digest = _get(expectation, "receipt_path"), _get(expectation, "receipt_sha256")
    if not _norm_abs(path) or not _hex64(digest):
        return False
    receipt = _get(observed, "receipt")
    ok = (_keys(receipt, RECEIPT_KEYS) and _same_str(receipt["path"], path)
          and _same_str(receipt["sha256"], digest))
    data = reader.read(path)
    return ok and data is not None and _sha256(data) == digest


def _pins_bound(pins, observed_pins, reader):
    if (type(pins) is not dict
            or not all(_norm_abs(path) and _hex64(digest) for path, digest in pins.items())):
        return False
    ok = _exact(pins, observed_pins)
    for path in sorted(pins):
        data = reader.read(path)
        if data is None or _sha256(data) != pins[path]:
            ok = False
    return ok

    if not hex64(d['input_sha256']) or d['input_sha256']!=digest(raw):raise Failure('actual raw hash mismatch')
    if not hex64(d['program_sha256']):raise Failure('informational program hash schema')
    if not hex64(d['solutions_sha256']):raise Failure('solution digest schema')
    for key in ('solutions','chunks'):
        if type(d[key]) is not list:raise Failure('list schema '+key)
    # Site-specific metadata: beta pin check precedes all published shape/args metadata;
    # joint argument fields are established before its pin check; neither begins any chunk on rejection.
    D=rawrows(raw);n=len(D[0]);k=len(D)
    expected={'n':n,'q':n//3,'k':k,'m':control['m'],'chunk_lo':control['lo'],'chunk_hi':control['hi'],'mode':'search'}
    if d['status']=='INPUT_PIN_MISMATCH':
        expected.update(n=None,q=None,k=None)
        if job['program']=='beta':expected.update(m=None,chunk_lo=None,chunk_hi=None,mode=None)
    for key,value in expected.items():
        if not exact(d[key],value):raise Failure('per-program/per-status field '+key)
    if d['status'] in ('NOT_DM','INPUT_PIN_MISMATCH'):
        if not exact(d['solutions'],[]) or not exact(d['chunks'],[]) or not exact(d['extra'],{}):raise Failure('early error must have empty payload/chunks/extra')
        if d['solutions_sha256']!=digest(b'[]'):raise Failure('empty solution digest')
        return d
    if d['status']!='COMPLETE':raise Failure('only COMPLETE controls qualify')
    if job['program']=='beta':
        if type(d['extra']) is not dict or set(d['extra'])!={'stage_b_leaves','stage_h_nodes'} or any(type(v) is not int or v<0 for v in d['extra'].values()):raise Failure('beta extra schema')
    elif not exact(d['extra'],{}):raise Failure('joint extra schema')
    sols=d['solutions']
    if len(sols)>4096 or len(d['chunks'])>4096:raise Failure('retention schema')
    counts={}
    for f in sols:
        if type(f) is not list or len(f)!=n or any(type(v) is not int or not 0<=v<n for v in f) or f[0]!=0:raise Failure('solution vector')
        idx=0
        for c in range(1,control['m']+1):idx=idx*n+f[c]
        if not control['lo']<=idx<control['hi']:raise Failure('solution outside chunk')
        if any(len({(row[c]-f[c])%n for c in range(n)})!=n for row in D):raise Failure('direct extension failure')
        counts[idx]=counts.get(idx,0)+1
    if sols!=sorted(sols) or len({tuple(f) for f in sols})!=len(sols):raise Failure('solution ordering/duplicates')
    if digest(json.dumps(sols,separators=(',',':')).encode('ascii'))!=d['solutions_sha256']:raise Failure('actual solution digest')
    if len(d['chunks'])!=control['hi']-control['lo']:raise Failure('chunk coverage')
    for idx,e in zip(range(control['lo'],control['hi']),d['chunks']):
        if type(e) is not dict or set(e)!={'chunk','status','nodes','solutions'}:raise Failure('chunk schema')
        if not exact(e['chunk'],idx) or not exact(e['status'],'COMPLETE') or type(e['nodes']) is not int or e['nodes']<0 or not exact(e['solutions'],counts.get(idx,0)):raise Failure('chunk contents')
    return d


def main(argv):
    global ACTIVE_DEADLINE
    ACTIVE_DEADLINE=None
    t0=time.monotonic();out={'status':'ERROR','reason':None,'self':None,'plan_sha256':None,'prefix':[],
        'all_as_expected':False,'qualified':False,'target_claim':'UNEVALUATED','items':[],'wall_s':0.0}
    rc=2;opened=[]
    try:
        if len(argv)!=8 or argv[0]!='--plan' or argv[2]!='--plan-sha256' or argv[4]!='--self' or argv[6]!='--budget-s':raise Failure('arguments')
        path,pin,selfname,S=argv[1],argv[3],argv[5],float(argv[7])
        if not math.isfinite(S) or not 0<S<=280:raise Failure('budget')
        ACTIVE_DEADLINE=t0+S
        def tick():
            if time.monotonic()-t0>S:raise TimeoutError('budget')
        pb=anchored_read_bytes(path,CAP)
        if len(pb)>CAP or digest(pb)!=pin:raise Failure('plan pin')
        plan=parse(pb);jobs={j['item']:j for j in plan['jobs']};order=plan['order']
        if selfname not in jobs or jobs[selfname]['kind']!='gate' or order!=[j['item'] for j in plan['jobs']]:raise Failure('self/order')
        prefix=order[:order.index(selfname)];out.update(self=selfname,plan_sha256=pin,prefix=prefix)
        base=dirfd(plan['base_dir']);audit=dirfd(plan['audit_dir']);opened += [base,audit]
        mine=os.open(selfname,os.O_RDONLY|os.O_DIRECTORY|os.O_NOFOLLOW,dir_fd=base)
        opened.append(mine)
        a,b=os.stat('.'),os.fstat(mine)
        if (a.st_dev,a.st_ino)!=(b.st_dev,b.st_ino):raise Failure('wrong working root')
        for p in plan['pins']:
            tick();data=anchored_read_bytes(p['path'],CAP)
            if len(data)>CAP or digest(data)!=p['sha256']:raise Failure('source/input pin '+p['path'])
        states={};ok={};controls=None;datafiles=None
        for item in prefix:
            tick();j=jobs[item];rec={'item':item,'problems':[],'artifacts':{}}
            try:
                fd=os.open(item,os.O_RDONLY|os.O_DIRECTORY|os.O_NOFOLLOW,dir_fd=base)
                try:files={nm:readat(fd,nm) for nm in j['outputs']}
                finally:os.close(fd)
                hashes={nm:digest(bs) for nm,bs in files.items()};rec['artifacts']=hashes
                aud=canonical(readat(audit,item+'.audit.json',1<<20),AKEYS)
                if type(aud) is not dict or set(aud)!=AKEYS:raise Failure('audit keys')
                wanted={'format':'ext-search-job-audit-v1','item':item,'plan_sha256':pin,'verdict':'AUDITED_PASS','settled':True,'exit_code':j['exit']}
                for key,val in wanted.items():
                    if not exact(aud[key],val):raise Failure('audit '+key)
                if not exact(aud['declared_output_sha256'],hashes):raise Failure('actual artifact/audit hashes')
                if not hex64(aud['receipt_sha256']):raise Failure('audit receipt digest must be lowercase SHA256')
                if any(type(aud[k]) is not str for k in ('auditor','at','notes')) or not aud['auditor'] or not aud['at']:raise Failure('audit string fields')
                if j['kind']=='data':
                    maker=canonical(files['result.json'],('status','reason','files','manifest_sha256','wall_s'))
                    if type(maker['wall_s']) is not float or not math.isfinite(maker['wall_s']) or maker['wall_s']<0:raise Failure('maker wall field')
                    if maker['status']!='OK' or maker['reason'] is not None or maker['files']!=list(MFILES) or maker['manifest_sha256']!=digest(files['manifest.json']):raise Failure('maker result')
                    man=canonical(files['manifest.json'],('format','dm21_input_sha256','files','controls'))
                    if man['format']!='ext-search-controls-v1' or set(man['files'])!=set(MFILES):raise Failure('control manifest')
                    if man['dm21_input_sha256']!=plan['dm21_sha256']:raise Failure('DM21 provenance pin')
                    for nm in MFILES:
                        if not exact(man['files'][nm],{'sha256':digest(files[nm]),'bytes':len(files[nm])}):raise Failure('control data hash')
                    if [c['name'] for c in man['controls']]!=['N6','N12','P21','P57','BAD_DM','BAD_PIN']:raise Failure('control names')
                    controls=verify_controls(man,files,plan,tick);datafiles=files
                elif j['kind']=='search':
                    if not ok.get('DATA') or controls is None:raise Failure('unaudited DATA dependency')
                    ctl=controls[j['control']];raw=datafiles[ctl['raw']]
                    res=validate_result(files['result.json'],j,raw,ctl)
                    if ctl['expected']=='COMPLETE_ZERO' and res['solutions']:raise Failure('parity control notzero')
                    if ctl['expected']=='COMPLETE_CONTAINS':
                        known=rawrows(datafiles[ctl['known']])[0]
                        if known not in res['solutions']:raise Failure('known deleted row missing')
                    states[item]=res
                    peer=j.get('compare_to')
                    if peer is not None:
                        if not ok.get(peer) or states[peer]['solutions']!=res['solutions'] or states[peer]['solutions_sha256']!=res['solutions_sha256']:raise Failure('independent COMPLETE sets disagree')
                elif j['kind']=='gate':
                    g=canonical(files['result.json'],GKEYS)
                    expected_gate={'status':'OK','reason':None,'self':item,'plan_sha256':pin,
                        'prefix':order[:order.index(item)],'all_as_expected':True,'qualified':False,'target_claim':'UNEVALUATED'}
                    for key,value in expected_gate.items():
                        if not exact(g[key],value):raise Failure('prior gate field '+key)
                    if not finite_wall(g['wall_s']) or type(g['items']) is not list:raise Failure('prior gate wall/items schema')
                    # Current records were recomputed from actual prefix artifacts and canonical parent audits.
                    # Earlier gate must report exactly those item records, in the same prefix order.
                    if not exact(g['items'],out['items']):raise Failure('prior gate item records differ from actual audited prefix')
                else:raise Failure('unknown jobkind')
            except (Failure,OSError,ValueError,KeyError,TypeError,IndexError) as e:rec['problems'].append(str(e))
            ok[item]=not rec['problems'];out['items'].append(rec)
        tick();good=all(ok.values())
        out.update(status='OK',all_as_expected=good,qualified=(good and selfname==plan['final_gate']))
        rc=0 if good else 1
    except TimeoutError:out.update(status='BUDGET_EXCEEDED',reason='budget');rc=5
    except Exception as e:out.update(status='ERROR',reason=str(e));rc=2
    finally:
        for f in reversed(opened):os.close(f)
    def expired():
        return ('S' in local_budget and math.isfinite(local_budget['S']) and 0<local_budget['S']<=280
                and time.monotonic()-t0>local_budget['S'])
    local_budget={'S':S} if 'S' in locals() else {}
    def encode_report():
        nonlocal rc
        if expired():
            out.update(status='BUDGET_EXCEEDED',reason='publication deadline',all_as_expected=False,qualified=False);rc=5
        out['wall_s']=round(time.monotonic()-t0,3)
        payload=(json.dumps(out,sort_keys=True,separators=(',',':'))+'\n').encode('ascii')
        if expired():
            out.update(status='BUDGET_EXCEEDED',reason='serialization deadline',all_as_expected=False,qualified=False);rc=5
            payload=(json.dumps(out,sort_keys=True,separators=(',',':'))+'\n').encode('ascii')
        return payload
    data=encode_report()
    try:
        if expired():data=encode_report()
        f=os.open('result.json',os.O_WRONLY|os.O_CREAT|os.O_EXCL|os.O_NOFOLLOW,0o644)
        try:
            if expired():data=encode_report()
            v=memoryview(data)
            while v:
                k=os.write(f,v)
                if k<=0:raise OSError('zero write')
                v=v[k:]
        finally:os.close(f)
    except OSError:return 2
    if expired():data=encode_report()  # existing file immutable; non-success/mismatch is audited
    if expired():data=encode_report()  # immediate pre-stdout completion boundary
    try:print(data.decode('ascii'),end='',flush=True)
    except OSError:return 2
    if expired():return 5
    return rc

if __name__=='__main__':sys.exit(main(sys.argv[1:]))
