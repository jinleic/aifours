def parse_anchor(raw_bytes, budget):
    """Section 1: non-empty lines of whitespace-separated decimal integers."""
    lines = raw_bytes.split(b"\n")

        budget.tick()
        tokens = line.split()           # ASCII whitespace (also drops a CR)
        if not tokens:
            raise CheckFailed(1, "anchor line %d is empty" % number)
        row = []
        for tok in tokens:
            digits = tok[1:] if tok[:1] in (b"+", b"-") else tok
            if not digits.isdigit():    # bytes.isdigit: ASCII 0-9 only, non-empty
                raise CheckFailed(1, "anchor line %d: token is not a decimal integer" % number)
            try:
                row.append(int(tok))
            except ValueError:
                raise CheckFailed(1, "anchor line %d: integer too long" % number) from None
        rows.append(row)
    return rows

def var_index(n, x, c):
    """v(x, c) = (x-1)*(n-1) + (c-1) for x = 1..k-1 and c = 1..n-1."""
    return (x - 1) * (n - 1) + (c - 1)


def entry(A, n, x, c):
    """A[x][c] of a point A in F_3^U read as the k x n array with fixed zeros."""
    if x == 0 or c == 0:
        return 0
    return A[var_index(n, x, c)]


def build_equations(der):
    """e_j for j = 0..J-1 as sorted sparse lists of (v, w) with w in {1, 2}."""
    n = der["n"]
    equations = []
    for x, y, cols in der["fibres"]:
        e = {}
        for c in cols:
            if c >= 1:
                if x >= 1:
                    v = var_index(n, x, c)
                    e[v] = (e.get(v, 0) + 1) % 3
                v = var_index(n, y, c)
                e[v] = (e.get(v, 0) + 2) % 3
        equations.append(sorted((v, w) for v, w in e.items() if w != 0))
    return equations


def point(idx, basis, U):
    """point(z) = sum over r of z[r]*basis[r] (mod 3), z = the dim base-3
    digits of idx with z[0] most significant (idx(z) of section 2)."""
    dim = len(basis)
    z = [0] * dim
    rest = idx
    for r in range(dim - 1, -1, -1):
        z[r] = rest % 3
        rest //= 3
    A = [0] * U
    for r in range(dim):
        zr = z[r]
        if zr:
            A = [a + zr * b for a, b in zip(A, basis[r])]
    return [a % 3 for a in A]


def fibre_values(A, n, fibre):
    """The values (A[x][c] - A[y][c]) mod 3 over the columns of the fibre."""
    x, y, cols = fibre
    return [(entry(A, n, x, c) - entry(A, n, y, c)) % 3 for c in cols]


def pairwise_distinct(values):
    a, b, c = values
    return a != b and a != c and b != c

    if len(rainbow_set) != len(rainbow) or len(failure_set) != len(failure_idx):
        raise CheckFailed(6, "duplicate point index")
    if rainbow_set & failure_set:
        raise CheckFailed(6, "rainbow and failure indices are not disjoint")
    if len(rainbow) + len(failure_idx) != total:
        raise CheckFailed(6, "%d indices listed, 3^dim needed (dim = %d)"
                          % (len(rainbow) + len(failure_idx), len(basis)))
    budget.tick()
    if (rainbow_set | failure_set) != set(range(total)):
        raise CheckFailed(6, "indices are not exactly {0, ..., 3^dim - 1}")
    for idx, j in failures:
        budget.tick()
        A = point(idx, basis, der["U"])
        if pairwise_distinct(fibre_values(A, der["n"], der["fibres"][j])):
            raise CheckFailed(6, "failure [%d, %d]: the fibre carries three distinct values"
                              % (idx, j))

def compute_facts(der, basis, rainbow, budget):
    """Facts of section 4 (reported only when VALID)."""
    n, k, U, T = der["n"], der["k"], der["U"], der["T"]
    alpha_hat = [0] * U
    for x in range(1, k):
        for c in range(1, n):
            alpha_hat[var_index(n, x, c)] = (T[x][c] - T[x][0]) % 3
    minus_alpha_hat = [(3 - a) % 3 for a in alpha_hat]
    plus_listed = False
    minus_listed = False
    for idx in rainbow:
        budget.tick()
        A = point(idx, basis, U)
        plus_listed = plus_listed or A == alpha_hat
        minus_listed = minus_listed or A == minus_alpha_hat
    dim = len(basis)
    count = len(rainbow)
    return {
        "dim": dim,
        "rank": U - dim,
        "rainbow_count": count,
        "alpha_hat_listed": plus_listed,
        "minus_alpha_hat_listed": minus_listed,
        "rainbow_is_pm_alpha_hat": (count == 2 and plus_listed and minus_listed
                                    and alpha_hat != minus_alpha_hat),
    }

def read_capped_input(path, cap, budget):
    """I/O boundary only: nonblocking open rejects FIFOs before any read; same-fd cap+1 read.
    Growth after fstat is bounded by bytes actually consumed, not pathname metadata.
    No snapshot or extra retained file is created. Errors are ERROR, never INVALID.
    """
    budget.tick()
    fd = os.open(path, os.O_RDONLY | os.O_NONBLOCK | os.O_NOFOLLOW)
    try:
        info = os.fstat(fd)
        if not stat.S_ISREG(info.st_mode):
            raise OSError("input is not a regular file")
        if info.st_size > cap:
            raise OSError("input exceeds size cap")
        chunks = []
        total = 0
        while True:
            budget.tick()
            block = os.read(fd, min(65536, cap + 1 - total))
            budget.tick()
            if not block:
                break
            total += len(block)
            if total > cap:
                raise OSError("input grew beyond size cap")
            chunks.append(block)
        budget.tick()
        return b"".join(chunks)
    finally:
        os.close(fd)


def run_checks(args, budget_s, t0, res):

        cert_bytes = read_capped_input(args["cert"], 8 << 20, budget)
        res["cert_sha256"] = hashlib.sha256(cert_bytes).hexdigest()
    except BudgetExceeded:
        set_error(res, "budget")
        return
    except OSError as exc:
        set_error(res, "cannot read bounded input: " + type(exc).__name__)
        return

        cert = load_certificate(cert_bytes)                           # check 0
        hdr = check0_format(cert, budget)                             # check 0
        der = check1_anchor(raw_bytes, res["anchor_sha256"],
                            args["anchor_sha256"], hdr, budget)       # check 1

def finish(res, t0, out_file, deadline=None):
    """Write the object to OUT_JSON (when reserved) and print it as one line."""
    if res["verdict"] not in EXIT_CODE:
        set_error(res, "internal error: no verdict")
    if deadline is not None and time.monotonic() > deadline:
        set_error(res, "budget before publication")
    res["wall_s"] = round(time.monotonic() - t0, 3)
    line = json.dumps(res, separators=(",", ":"))
    if out_file is not None:
        try:
            out_file.write(line + "\n")
            out_file.close()
        except OSError as exc:
            set_error(res, "cannot write output file: " + type(exc).__name__)
            res["wall_s"] = round(time.monotonic() - t0, 3)
            line = json.dumps(res, separators=(",", ":"))
    if deadline is not None and time.monotonic() > deadline:
        # The exclusive artifact is preserved; stdout/exit become ERROR, so parent audit must reject.
        set_error(res, "budget after publication")
        res["wall_s"] = round(time.monotonic() - t0, 3)
        line = json.dumps(res, separators=(",", ":"))

        print(line, flush=True)
    except OSError:
        return EXIT_CODE["ERROR"]
    return EXIT_CODE[res["verdict"]]

    except UsageError as exc:

        return finish(res, t0, None)
    try:
        out_file = open(args["out"], "x", encoding="utf-8")       # reserve OUT_JSON
    except FileExistsError:
        set_error(res, "output file already exists")
        return finish(res, t0, None)
    except OSError as exc:
        set_error(res, "cannot create output file: " + type(exc).__name__)
        return finish(res, t0, None)
    try:
        run_checks(args, budget_s, t0, res)
    except Exception as exc:            # checker defect or resource failure: never VALID
        set_error(res, "internal error: " + type(exc).__name__)
    return finish(res, t0, out_file, deadline=t0 + budget_s)

def strict_loads(data):
    def pairs(items):
        out = {}
        for key, value in items:
            if key in out:
                raise ValueError("duplicate key")
            out[key] = value
        return out

    def bad_constant(name):
        raise ValueError("non-finite constant")

    def finite_float(text):
        value = float(text)
        if not math.isfinite(value):
            raise ValueError("non-finite float")
        return value

    return json.loads(data.decode("utf-8"), object_pairs_hook=pairs, parse_constant=bad_constant,
                      parse_float=finite_float)

def same(a, b):
    """Typed structural equality: bool, int, float, str, None, list and dict never compare across types."""
    if type(a) is not type(b):
        return False
    if type(a) is dict:
        return list(a.keys()) == list(b.keys()) and all(same(a[k], b[k]) for k in a) \
            if list(a.keys()) == list(b.keys()) else (set(a) == set(b) and all(same(a[k], b[k]) for k in a))
    if type(a) is list:
        return len(a) == len(b) and all(same(x, y) for x, y in zip(a, b))
    return a == b


def strict_loads(data):
    def pairs(items):
        out = {}
        for key, value in items:
            if key in out:
                raise ValueError("duplicate key")
            out[key] = value
        return out

    def bad_constant(name):
        raise ValueError("non-finite constant")

    def finite_float(text):
        value = float(text)
        if not math.isfinite(value):
            raise ValueError("non-finite float")
        return value

    return json.loads(data.decode("utf-8"), object_pairs_hook=pairs, parse_constant=bad_constant,
                      parse_float=finite_float)


def parse_report(data, keys, sorted_keys):
    """One-line report in its producer's exact serialisation with exactly `keys`."""
    if not data.endswith(b"\n") or data.count(b"\n") != 1:
        raise Problem("report is not exactly one newline-terminated line")
    try:
        obj = strict_loads(data)
    except (UnicodeDecodeError, ValueError, RecursionError):
        raise Problem("report is not strict JSON") from None
    if type(obj) is not dict:
        raise Problem("report is not an object")
    if sorted_keys:
        if set(obj) != set(keys):
            raise Problem("report key set differs")
        canon = json.dumps(obj, sort_keys=True, separators=(",", ":")) + "\n"
    else:
        if list(obj.keys()) != list(keys):
            raise Problem("report keys differ from the fixed order")
        canon = json.dumps(obj, separators=(",", ":")) + "\n"
    if canon.encode("ascii") != data:
        raise Problem("report bytes are not in the exact serialisation")
    return obj


def parse_canonical(data, keys, what):

def open_dir(path):
    if type(path) is not str or not path.startswith("/") or "\x00" in path:
        raise OSError("not an absolute path")
    parts = path.split("/")[1:]
    if any(p in ("", ".", "..") for p in parts):
        raise OSError("path is not normalised")
    fd = os.open("/", os.O_RDONLY | os.O_DIRECTORY)
    try:
        for p in parts:
            nfd = os.open(p, os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW, dir_fd=fd)
            os.close(fd)
            fd = nfd
    except OSError:
        os.close(fd)
        raise
    return fd


def read_entry(dir_fd, name, cap):
    if not NAME_RE.fullmatch(name) or name in (".", ".."):
        raise Problem("bad file name")
    try:
        fd = os.open(name, os.O_RDONLY | os.O_NOFOLLOW, dir_fd=dir_fd)
    except OSError as exc:
        raise Problem("cannot open %s: %s" % (name, type(exc).__name__)) from None
    try:
        st = os.fstat(fd)
        if not stat.S_ISREG(st.st_mode):
            raise Problem(name + " is not a regular file")
        if st.st_nlink != 1:
            raise Problem(name + " does not have exactly one link")
        if st.st_size > cap:
            raise Problem(name + " exceeds its size cap")
        chunks, total = [], 0
        while True:
            block = os.read(fd, 1 << 20)
            if not block:
                break
            total += len(block)
            if total > cap:
                raise Problem(name + " exceeds its size cap")
            chunks.append(block)
        return b"".join(chunks)
    finally:
        os.close(fd)


def read_pinned(path, cap):
    fd = os.open(path, os.O_RDONLY | os.O_NONBLOCK | os.O_NOFOLLOW)
    try:
        info = os.fstat(fd)
        if not stat.S_ISREG(info.st_mode) or info.st_size > cap:
            raise OSError("pinned input kind or cap")
        chunks, total = [], 0
        while True:
            b = os.read(fd, min(65536, cap + 1 - total))
            if not b:
                break
            total += len(b)
            if total > cap:
                raise OSError("pinned input grew past cap")
            chunks.append(b)
        return b"".join(chunks)
    finally:
        os.close(fd)

def derive_nc6(src, anchor_raw, budget):
    """Independent from producer/generator/checker code: derive ALL candidate fibres from pinned raw,
    reconstruct each source point, and select first eligible failure + smallest passing fibre.
    No manifest position or new_fibre is used to guide the selection.
    """
    n, q, k, U, dim = (src[key] for key in ("n", "q", "k", "U", "dim"))
    if not all(type(v) is int for v in (n, q, k, U, dim)) or not (2 <= k <= 12 and n <= 120
            and n == 3*q and q >= 1 and q % 3 != 0 and 0 <= dim <= 8 and U == (k-1)*(n-1)):
        raise Problem("nc6 source header/work limits")
    lines = anchor_raw.split(b"\n")
    if lines[-1] == b"":
        lines.pop()
    D = []
    for line in lines:
        budget.tick()
        tokens = line.split()
        if not tokens:
            raise Problem("nc6 raw G1 blank row")
        vals = []
        for t in tokens:
            z = t[1:] if t[:1] in (b"+", b"-") else t
            if not z.isdigit():
                raise Problem("nc6 raw G1 token")
            vals.append(int(t) % n)
        D.append(vals)
    if len(D) != k or any(len(r) != n for r in D):
        raise Problem("nc6 raw/source shape")
    fibres = []
    for x in range(k):
        for y in range(x+1, k):
            budget.tick()
            buckets = [[] for _ in range(q)]
            for c in range(n):
                buckets[(D[x][c]-D[y][c]) % q].append(c)
            for t in range(q):
                if len(buckets[t]) != 3:
                    raise Problem("nc6 raw fibre size")
                fibres.append((x, y, buckets[t]))
    if len(fibres) != src["J"]:
        raise Problem("nc6 fibre count")
    basis = src["basis"]
    if type(basis) is not list or len(basis) != dim or any(type(b) is not list or len(b) != U
            or any(type(v) is not int or not 0 <= v <= 2 for v in b) for b in basis):
        raise Problem("nc6 source basis schema")
    def good(A, j):
        x, y, cols = fibres[j]
        values = []
        for c in cols:
            ax = A[(x-1)*(n-1)+(c-1)] if x and c else 0
            ay = A[(y-1)*(n-1)+(c-1)] if y and c else 0
            values.append((ax-ay) % 3)
        return len(set(values)) == 3
    last = -1
    for pos, entry in enumerate(src["failures"]):
        budget.tick()
        if type(entry) is not list or len(entry) != 2:
            raise Problem("nc6 source failure schema")
        idx, old_j = entry
        if type(idx) is not int or not last < idx < 3**dim or type(old_j) is not int or not 0 <= old_j < len(fibres):
            raise Problem("nc6 source failure index")
        last = idx
        digits = [0]*dim
        rem = idx
        for r in range(dim-1, -1, -1):
            digits[r] = rem % 3
            rem //= 3
        A = [sum(digits[r]*basis[r][v] for r in range(dim)) % 3 for v in range(U)]
        if good(A, old_j):
            raise Problem("nc6 claimed source failure actually passes")
        for j in range(len(fibres)):
            budget.tick()
            if good(A, j):
                return {"position": pos, "idx": idx, "old_fibre": old_j, "new_fibre": j}
    raise Problem("nc6 no eligible source failure")


def apply_mutation(src, name, detail, inputs, job, anchor_raw, budget):
    """Independent re-derivation of the intended mutant from the actual source certificate."""
    if type(detail) is not dict or list(sorted(detail)) != sorted(DETAIL_KEYS[name]) \
            or not all(type(v) is int for v in detail.values()):
        raise Problem(name + ": detail keys or types differ")
    m = copy.deepcopy(src)
    rw, fl = src["rank_witness"], src["failures"]
    if name == "nc0_identity":
        pass
    elif name == "nc1_basis_flip":
        p = rw[0]["pivot"]
        old = src["basis"][0][p]
        if not same(detail, {"vector": 0, "position": p, "old": old, "new": (old + 1) % 3}):
            raise Problem(name + ": detail differs from the source")
        m["basis"][0][p] = (old + 1) % 3
    elif name == "nc2_pivot_swap":
        pa, pb = rw[0]["pivot"], rw[1]["pivot"]
        if not same(detail, {"record_a": 0, "record_b": 1, "pivot_a": pa, "pivot_b": pb}):
            raise Problem(name + ": detail differs from the source")
        m["rank_witness"][0]["pivot"], m["rank_witness"][1]["pivot"] = pb, pa
    elif name == "nc3_rainbow_swap":
        r0, f0, jf0 = src["rainbow"][0], fl[0][0], fl[0][1]
        if not same(detail, {"rainbow_idx_moved": r0, "failure_idx_moved": f0, "failure_fibre_of_moved": jf0,
                             "new_failure_fibre": 0}):
            raise Problem(name + ": detail differs from the source")
        m["rainbow"] = sorted([i for i in src["rainbow"] if i != r0] + [f0])
        m["failures"] = sorted([list(e) for e in fl if e[0] != f0] + [[r0, 0]], key=lambda e: e[0])
    elif name == "nc4_failure_delete":
        if not same(detail, {"deleted_idx": fl[0][0], "deleted_fibre": fl[0][1]}):
            raise Problem(name + ": detail differs from the source")
        m["failures"] = [list(e) for e in fl[1:]]
    elif name == "nc5_wrong_anchor_hash":
        m["anchor_sha256"] = inputs[job["other_input"]]["sha256"]
    elif name == "nc6_failure_wrong_fibre":
        derived = derive_nc6(src, anchor_raw, budget)
        if not same(detail, derived):
            raise Problem(name + ": independently derived first eligible failure/smallest passing fibre differs")
        m["failures"][derived["position"]] = [derived["idx"], derived["new_fibre"]]

            record["problems"].append("plan or data shape error: " + type(exc).__name__)

The new findings, ordered by severity, are:

1. **MAJOR — nc6’s independently verified construction is missing.**

   In `check_cert_runs_v3.py::apply_mutation()`, the nc6 branch checks that the supplied position matches a source failure and that `new_fibre` is different and in range. It does **not** derive fibres from the raw anchor, establish that the replacement fibre passes, or verify the prescribed first eligible failure and smallest passing fibre.

   A concrete counterexample follows directly from the producer. Point 0 is the zero vector, so every fibre fails and its first failure entry is `[0,0]`. Supply:

   ```json
   {"position":0,"idx":0,"old_fibre":0,"new_fibre":1}
   ```

   Change only that certificate entry to `[0,1]`, and consistently update manifest, report and audit hashes. `apply_mutation()` accepts it, and EXP-C can report success. Yet fibre 1 also fails for the zero vector: this is still a valid failure certificate, not the registered negative control.

   The unchanged checker would subsequently return VALID for this mutant, so EXP-D would stop. This counterexample does **not** establish a false final mathematical claim; it establishes a false EXP-C acceptance and disproves the claimed independent reconstruction of every mutant.

   **Minimal repair:** independently read the pinned anchor bytes, derive fibres and points, scan source failures in order, find the smallest passing fibre for the first eligible point, and require the manifest detail and mutant to equal that independently derived result.

2. **MAJOR — checker input caps do not bind the bytes actually read.**

   All entrypoints’ `_check_args()` use `os.stat(value)`. The checker later opens the pathname again and performs unbounded reads in `run_checks()`:

   ```python
   raw_bytes = fh.read()
   cert_bytes = fh.read()
   ```

   Counterexample: the certificate is a small regular file during `stat()`, then grows beyond 8 MiB before the checker opens it. The checker reads beyond the advertised cap. Replacement with a FIFO can instead block before its cooperative budget checks begin.

   The external wall/RSS guards remain essential, but they do not make the stated input-cap guarantee true. Output-directory `O_NOFOLLOW` protection does not resolve this input check/use gap.

   **Minimal repair:** enforce caps on the bytes supplied to the checker through a reviewed bounded snapshot or reader. Retaining a descriptor alone is insufficient if its file can grow. Any added snapshot files must be accounted for explicitly; alternatively, re-review and repin a checker change that performs capped reads.

3. **MINOR — publication checks do not match the per-step budget promise.**

   `make_cert_mutants_v3.py::generate()` has one `budget.tick()` before:

   ```python
   for name, data in blobs:
       write_new(name, data)
   write_new("manifest.json", mdata)
   ```

   There is no check between these twelve publications. If the deadline expires during the first write, the remaining mutants and manifest are still published before `main()` notices expiry. Final `result.json` publication also occurs outside a fresh budget check.

   **Minimal repair:** check immediately before each file publication and final report publication, and preserve failure status when expiry is detected. Continue treating supervisor wall enforcement as independent protection against blocking I/O.

4. **MINOR — the final GO read can cross the 120-second cutoff without rejection.**

   `_await_go()` computes `remaining` before `select()`, but does not check the clock after the last read or probe.

   Counterexample: `select()` returns ready at 119.9 seconds; the process resumes at 121 seconds, reads `GO\n`, completes the zero-timeout probe and loads the algorithm. The deadline is never reconsidered.

   **Minimal repair:** refresh the deadline after read/probe completion and immediately before leaving the GO gate. The original unbounded-wait defect is otherwise resolved.
