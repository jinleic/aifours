452|def walk_fd(anchor, dev, dl, seen=None):
453|    """distinct-inode logical bytes; disappeared/retyped entries are incomplete accounting, never zero."""
454|    seen = set() if seen is None else seen

458|        if depth > 64:

462|            for e in it:
463|                dl.check("walk entry")
464|                count[0] += 1

466|                    raise GuardError("WALK_ENTRY_CAP")

479|            try:

542|    """Exact JSON structure and scalar types; bool never aliases an integer authority field."""
543|    if type(left) is not type(right):
544|        return False

634|def spawn_registered(s, argv, env, role, dl, required_s):
635|    """Spawn with an explicitly empty signal mask while cancellation is protected in the parent.
636|    Registration is the FIRST operation after posix_spawn returns a PID; no Popen finalizer or second reaper exists.
637|    Returned pipe descriptors remain owned even when post-spawn close/deadline work fails."""
638|    allocated = []
639|    record = {"pid": None, "role": role, "identity": None, "leader_reaped": False, "rc": None,
640|              "wait_lost": False, "term_attempted": False, "settled": False, "stdout_fd": None,
641|              "stderr_fd": None, "stdin_fd": None, "spawn_error": None, "parent_close_fds": []}
642|    s.keep.append(record)  # Retain the allocation/pipe ledger even before a PID exists.
643|    if role == "helper":
644|        s.helpers.append(record)
645|    try:
646|        out_read, out_write = os.pipe()
647|        allocated.extend((out_read, out_write))
648|        if role == "helper":
649|            null = os.open("/dev/null", os.O_RDWR)
650|            allocated.append(null)
651|            in_read, err_write = null, null
652|            close_parent = {null, out_write}
653|        else:
654|            in_read, in_write = os.pipe()
655|            err_read, err_write = os.pipe()
656|            allocated.extend((in_read, in_write, err_read, err_write))
657|            record["stdin_fd"], record["stderr_fd"] = in_write, err_read
658|            close_parent = {in_read, out_write, err_write}
659|        record["parent_close_fds"] = list(close_parent)
660|        record["stdout_fd"] = out_read
661|        if any(fd < 3 for fd in allocated):
662|            raise GuardError("STANDARD_DESCRIPTORS_MUST_BE_OPEN")
663|        actions = [(os.POSIX_SPAWN_DUP2, in_read, 0), (os.POSIX_SPAWN_DUP2, out_write, 1),
664|                   (os.POSIX_SPAWN_DUP2, err_write, 2)]
665|        actions += [(os.POSIX_SPAWN_CLOSE, fd) for fd in allocated]
666|        with protected():
667|            dl.check("before protected spawn")
668|            if role == "helper" and (s.failures or s.in_helper is False):
669|                raise GuardError("HELPER_LAUNCH_REFUSED")
670|            s.authorization_io(dl)  # LAST preparation I/O before the protected fresh-clock spawn commit.
671|            dl.check("after protected spawn authorization")
672|            record["launch_commit"] = s.time_commit("HELPER" if role == "helper" else "LAUNCH", required_s)
673|            record["mono_spawn"], record["epoch_spawn"] = time.monotonic(), time.time()
674|            if role != "helper":
675|                s.mono_spawn, s.epoch_spawn = record["mono_spawn"], record["epoch_spawn"]
676|                s.rec.update(mono_spawn=s.mono_spawn, epoch_spawn=s.epoch_spawn,
677|                             mono_child_deadline=s.mono_spawn + s.cfg["max_wall_s"])
678|            pid = os.posix_spawn(argv[0], argv, env, file_actions=actions, setsid=True, setsigmask=(),
679|                                 setsigdef=tuple(CANCEL_SIGNALS | {signal.SIGPIPE, signal.SIGCHLD}))
680|            record["pid"] = pid
681|            if role == "helper":
682|                s.helper_calls += 1
683|            if role != "helper":
684|                s.child, s.pid = record, pid
685|            s.aux_clock(record, "mono_pid_registered")
686|        try:
687|            for fd in close_parent:
688|                os.close(fd)
689|                allocated.remove(fd)
690|                record["parent_close_fds"].remove(fd)
691|            dl.check("after spawn/parent pipe closes")
692|        except BaseException as e:
693|            record["spawn_error"] = repr(e)
694|        return record
695|    except BaseException:
696|        if record["pid"] is None:
697|            record["settled"] = True  # No process was created; no leader/descendant can be abandoned.
698|            for fd in allocated:
699|                try:
700|                    os.close(fd)
701|                except OSError:
702|                    pass
703|        raise
