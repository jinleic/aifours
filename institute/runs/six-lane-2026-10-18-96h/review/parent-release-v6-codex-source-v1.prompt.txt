414:     def native(self, stage, function, *args, retain=None, after_ack=False):
415:         """One native blocking operation under the SAME original deadline.
416:
417:         Checked before the call (no new I/O after detection). Returned ownership/progress is
418:         retained by the non-raising `retain` hook BEFORE the completion check. Success and
419:         Exception completions are both checked; primary and secondary failures stay separate.
420:         The primary Exception is re-raised; the crossed latch refuses every later operation.
421:         """
422:         self.must_tick('before ' + stage, after_ack)
423:         try:
424:             value = function(*args)
425:         except Exception as primary:
426:             self.event(stage, 'NATIVE_EXCEPTION', self.why(primary))
427:             self.failed(stage, primary, after_ack)
428:             raise
429:         except BaseException as primary:
430:             self.keep(primary, True)  # origin: the native function raised
431:             self.event(stage, 'NATIVE_BASE_EXCEPTION', self.why(primary))
432:             raise
433:         if retain is not None:
434:             retain(value)
435:         self.must_tick(stage + ' completion', after_ack)
436:         return value

477:     def close(self, handle, cause):
478:         key = _handle_key(handle)
479:         if key not in self.live:
480:             return True
481:         record = self.live.pop(key)  # one destructive close attempt; never blindly retried
482:         try:
483:             if record['kind'] == 'fd':
484:                 os.close(handle)
485:             else:
486:                 handle.close()
487:         except BaseException as exc:
488:             self.keep(exc, True)  # origin: the destructive close raised
489:             self.unresolved_total += 1
490:             if len(self.unresolved) < 64:
491:                 self.unresolved.append({'kind': record['kind'], 'fd': handle if record['kind'] == 'fd' else None,
492:                                         'path': record['path'], 'error': self.why(exc)})
493:             if cause is not None:
494:                 self.fire(cause, 'unresolved close ' + _short(record['path']) + ': ' + self.why(exc))
495:             if not _ordinary(exc):
496:                 raise
497:             self.failed('close ' + _short(record['path']), exc)
498:             return False
499:         self.closed += 1
500:         self.tick('close completion ' + _short(record['path']))  # cleanup always allowed; checked unless crossed
501:         return True

503:     def release(self, owned, cause):
504:         """One close attempt for EVERY owned handle (reverse order), even after a BaseException.
505:
506:         Returns True only when each close completed cleanly. The newest non-Exception
507:         BaseException among these attempts propagates locally after all attempts; every object
508:         stays retained and final selection uses the transaction's newest actual origin. If none
509:         occurred, the newest escaping ordinary Exception propagates. No close is retried.
510:         """
511:         clean, newest, newest_base = True, None, None
512:         for handle in reversed(owned):
513:             try:
514:                 clean = self.close(handle, cause) and clean
515:             except BaseException as exc:
516:                 self.keep(exc)
517:                 clean, newest = False, exc
518:                 if not _ordinary(exc):
519:                     newest_base = exc
520:         if newest_base is not None:
521:             raise newest_base
522:         if newest is not None:
523:             raise newest
524:         return clean

536:         except BaseException as exc:

1460: def _write_all(txn, fd, data, retained, stage):
1461:     view, at, calls = memoryview(data), 0, 0
1462:     def progress(count):  # native write progress retained BEFORE the completion check
1463:         if type(count) is int and 0 < count <= len(view) - at:
1464:             retained['bytes_written'] = at + count
1465:             retained['line_complete'] = at + count == len(view)
1466:     while at < len(view):
1467:         calls += 1
1468:         _require(calls <= 65536, 'write call cap')
1469:         count = txn.native(stage, os.write, fd, view[at:], retain=progress)
1470:         _require(type(count) is int and 0 < count <= len(view) - at, 'zero/invalid write')
1471:         at += count
1472:         retained['bytes_written'] = at
1473:     retained['line_complete'] = True

1523:         return False
1524:     result['ack'] = 'ACK_ATTEMPTED'  # conservative latch BEFORE readback I/O; promoted only on completion
1525:     try:
1526:         back = txn.observe('read_bytes', retained['path'], len(data), after_ack=True)
1527:     except Exception as exc:
1528:         result['ack'] = 'ACK_FAILED'
1529:         txn.fire('P5_ACK_FAILED', txn.why(exc))
1530:         return False

1532:         result['ack'] = 'ACK_UNPROVED_DEADLINE'
1533:         result['uncertainty'] = 'late readback; no further identity check/registry I/O'
1534:         return False

1537:                  and hashlib.sha256(back).hexdigest() == retained['sha256'], 'ACK bytes/hash mismatch')

1541:                  and actual.st_dev == identity['dev'] and actual.st_ino == identity['ino'], 'ACK not original created inode')
1542:     except _Crossed:
1543:         result['ack'] = 'ACK_UNPROVED_DEADLINE'
1544:         return False

1546:         txn.fire('P5_ACK_FAILED', txn.why(exc))
1547:         result['ack'] = 'ACK_FAILED'
1548:         return False
1549:     if not txn.tick('ACK original-inode identity completion', after_ack=True):
1550:         result['ack'] = 'ACK_UNPROVED_DEADLINE'
