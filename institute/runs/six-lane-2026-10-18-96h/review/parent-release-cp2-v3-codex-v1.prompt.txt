598: def make_census(deadline_check):
599:     """Return (census, log). census() is the PRT census callback over a NEW independent full libproc snapshot.
600: 
601:     Up to three fresh full attempts (sv19 full_census reuse, no merging); first complete attempt else last attempt
602:     with complete False. No helper process, no signal, no cache. Raises on deadline, native listing failure or log
603:     cap; BaseException propagates. Every attempt summary and every served snapshot (full raw identities, digest of
604:     the exact served dict) is retained in log for evaluate_endedness and the CP1 receipt.
605:     """
606:     _require(callable(deadline_check), 'deadline_check must be callable')
607:     log = CensusLog()
608:     bound = []
609: 
610:     def census():
611:         log.calls += 1
612:         call = log.calls
613:         try:
614:             _check(deadline_check, 'census %d start' % call)
615:             if len(log.served) >= MAX_SERVED or len(log.attempts) + ATTEMPTS_PER_CALL > MAX_LOG_ATTEMPTS:
616:                 raise CensusRefusal('census log cap reached: refuse rather than discard evidence')
617:             if not bound:
618:                 bound.append(_bind_libproc())
619:             chosen, attempt = None, 0
620:             while attempt < ATTEMPTS_PER_CALL:
621:                 attempt += 1
622:                 chosen = _attempt(bound[0], deadline_check, call, attempt)

624:                 if chosen['complete']:
625:                     break
626:                 if attempt < ATTEMPTS_PER_CALL:
627:                     _check(deadline_check, 'census %d before retry pause' % call)
628:                     time.sleep(RETRY_PAUSE_S)
629:                     _check(deadline_check, 'census %d after retry pause' % call)
630:             observed = time.monotonic()
631:             if type(observed) is not float or not math.isfinite(observed):
632:                 raise CensusRefusal('monotonic clock unavailable; observed_at_mono is never fabricated')

637:                       'observed_at_mono': observed}

639:                       'observer_ok': chosen['observer_ok'], 'process_count': len(chosen['rows']),
640:                       'rows': [dict(p) for p in chosen['rows']],
