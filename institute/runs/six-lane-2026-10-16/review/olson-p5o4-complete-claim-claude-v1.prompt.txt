36: (a) **Runs are multisets.** Let w = A U B, where U is a run of rotations, and let w' = A U' B with U' a rearrangement of
37: U. Fix a value-preserving bijection π from the positions of U to the positions of U'. Map a selection I in w to
38: I' = (I∩A) ∪ π(I∩U) ∪ (I∩B). This map is a bijection between selections and preserves their size. The rotations
39: commute: rho_x rho_x' = (x + x', +1). So the selected part of U multiplies to rho_{Σ x}, and so does its image in U'.
40: The two products (∏A-part)·rho_{Σx}·(∏B-part) are therefore equal, and w and w' have the same set of 10-subword

63:   for (int i = 1; i < 16; i++) FACT[i] = FACT[i - 1] * (u64)i;
64: }

66: // ---------------- exact ordered DP: r[c] = set of products of c selected letters (bit mask over the 10 elements)
67: struct St {
68:   uint16_t r[FULL + 1];
69: };
70: static inline void st_init(St& s) {
71:   memset(s.r, 0, sizeof s.r);
72:   s.r[0] = (uint16_t)(1u << ID);
73: }
74: static inline void st_push(St& s, int g) {
75:   const uint16_t* t = TAB[g];
76:   for (int c = FULL; c >= 1; c--) s.r[c] |= t[s.r[c - 1]];  // descending c: r[c-1] is still the old value
77: }
78: static inline bool st_solved(const St& s) { return (s.r[TARGET] >> ID) & 1; }
79: static bool dp_scratch(const int* w, int n) {
80:   St s;
81:   st_init(s);
82:   for (int p = 0; p < n; p++) st_push(s, w[p]);
83:   return st_solved(s);
84: }
85: 
86: // ---------------- naive check: every k-subset of positions, product left to right with MUL
87: static bool naive_solved(const int* w, int n, int k) {
88:   int id[WLEN];
89:   for (int i = 0; i < k; i++) id[i] = i;
90:   for (;;) {
91:     int g = ID;
92:     for (int i = 0; i < k; i++) g = MUL[g][w[id[i]]];
93:     if (g == ID) return true;
94:     int i = k - 1;
95:     while (i >= 0 && id[i] == n - k + i) i--;
96:     if (i < 0) return false;
97:     id[i]++;
98:     for (int j = i + 1; j < k; j++) id[j] = id[j - 1] + 1;
99:   }
100: }

102: static int build_word(const int c[4][P], const int a[4], const int y[4], int* w) {
103:   int n = 0;
104:   for (int i = 0; i < 4; i++) {
105:     for (int x = 0; x < P; x++)
106:       for (int k = 0; k < c[i][x]; k++) w[n++] = RHO(x);
107:     for (int k = 0; k < 2 * a[i] + 1; k++) w[n++] = SIG(y[i]);
108:   }
109:   return n;
110: }

112: static std::string word_str(const int* w, int n) {
113:   std::string s;
114:   int i = 0;
115:   while (i < n) {
116:     int j = i;
117:     while (j < n && w[j] == w[i]) j++;
118:     char buf[32];
119:     if (j - i > 1)
120:       snprintf(buf, sizeof buf, "%s%c%d^%d", s.empty() ? "" : " ", ge(w[i]) == 1 ? 'r' : 's', gv(w[i]), j - i);
121:     else
122:       snprintf(buf, sizeof buf, "%s%c%d", s.empty() ? "" : " ", ge(w[i]) == 1 ? 'r' : 's', gv(w[i]));
123:     s += buf;
124:     i = j;
125:   }
126:   return s;
127: }

129: static std::string runs_str(const int c[4][P]) {
130:   std::string s;

132:     if (i) s += '/';
133:     int n = 0;

135:       for (int k = 0; k < c[i][x]; k++) {

137:         n++;
138:       }

144: // ---------------- fact (d): j <= H pairs inside blocks + a zero-sum rotation sub-multiset of size 10 - 2j
145: static bool filter_d(int H, const int m[P]) {
146:   int R = 0;
147:   for (int x = 0; x < P; x++) R += m[x];
148:   int pairs = g_parity ? H + 4 : H;  // BUG parity: each odd block credited (l+1)/2 identity pairs
149:   int jmax = pairs < 5 ? pairs : 5;
150:   for (int j = 0; j <= jmax; j++) {
151:     int s = FULL - 2 * j;
152:     if (s > R) continue;
153:     for (int t0 = 0; t0 <= m[0]; t0++)
154:       for (int t1 = 0; t1 <= m[1]; t1++)
155:         for (int t2 = 0; t2 <= m[2]; t2++)
156:           for (int t3 = 0; t3 <= m[3]; t3++) {
157:             int t4 = s - t0 - t1 - t2 - t3;
158:             if (t4 < 0 || t4 > m[4]) continue;
159:             if ((t1 + 2 * t2 + 3 * t3 + 4 * t4) % P == 0) return true;
160:           }
161:   }
162:   return false;
163: }

165: // rigid: every rotation value in one {c, -c}, c != 0 (vacuous for no rotations)
166: static bool rigid(const int m[P]) {
167:   if (m[0] > 0) return false;
168:   bool a = m[1] + m[4] > 0, b = m[2] + m[3] > 0;
169:   return !(a && b);
170: }

172: static u64 binom(int n, int k) {
173:   if (k < 0 || k > n) return 0;
174:   u64 r = 1;
175:   for (int i = 1; i <= k; i++) r = r * (u64)(n - k + i) / (u64)i;
176:   return r;
177: }
178: static u64 pow5(int e) {
179:   u64 r = 1;
180:   while (e-- > 0) r *= 5;
181:   return r;
182: }
183: static inline int mcode(const int m[P]) { return m[0] + 15 * (m[1] + 15 * (m[2] + 15 * (m[3] + 15 * m[4]))); }

215: // l-tuples (a1,a2,a3,a4), sum H, lexicographic in (a1,a2,a3)
216: static int LID[8][8][8];
217: static int setup_ltuples(int H, std::vector<std::array<int, 4>>& L) {
218:   L.clear();
219:   for (int a1 = 0; a1 <= H; a1++)
220:     for (int a2 = 0; a1 + a2 <= H; a2++)
221:       for (int a3 = 0; a1 + a2 + a3 <= H; a3++) {
222:         LID[a1][a2][a3] = (int)L.size();
223:         L.push_back({a1, a2, a3, H - a1 - a2 - a3});
224:       }
225:   return (int)L.size();
226: }

228: // all (l, y) completions of one run distribution c; DP state shared along common prefixes (exact memoisation)
229: template <class F>
230: static void enum_ly(const int c[4][P], int H, F& f) {
231:   St s0;
232:   st_init(s0);
233:   for (int x = 0; x < P; x++)
234:     for (int k = 0; k < c[0][x]; k++) st_push(s0, RHO(x));
235:   for (int a1 = 0; a1 <= H; a1++)
236:     for (int y1 = 0; y1 < P; y1++) {
237:       St s1 = s0;
238:       for (int k = 0; k < 2 * a1 + 1; k++) st_push(s1, SIG(y1));
239:       for (int x = 0; x < P; x++)
240:         for (int k = 0; k < c[1][x]; k++) st_push(s1, RHO(x));
241:       for (int a2 = 0; a1 + a2 <= H; a2++)
242:         for (int y2 = 0; y2 < P; y2++) {
243:           St s2 = s1;
244:           for (int k = 0; k < 2 * a2 + 1; k++) st_push(s2, SIG(y2));
245:           for (int x = 0; x < P; x++)
246:             for (int k = 0; k < c[2][x]; k++) st_push(s2, RHO(x));
247:           for (int a3 = 0; a1 + a2 + a3 <= H; a3++)
248:             for (int y3 = 0; y3 < P; y3++) {
249:               St s3 = s2;
250:               for (int k = 0; k < 2 * a3 + 1; k++) st_push(s3, SIG(y3));
251:               for (int x = 0; x < P; x++)
252:                 for (int k = 0; k < c[3][x]; k++) st_push(s3, RHO(x));
253:               int a4 = H - a1 - a2 - a3;
254:               for (int y4 = 0; y4 < P; y4++) {
255:                 St s4 = s3;
256:                 for (int k = 0; k < 2 * a4 + 1; k++) st_push(s4, SIG(y4));
257:                 f(a1, a2, a3, a4, y1, y2, y3, y4, st_solved(s4));
258:               }
259:             }
260:         }
261:     }
262: }

264: // all 4-tuples of run multisets (20 counts c[run][value]) with total R
265: template <class F>
266: static void rec_dist(int k, int rem, int c[4][P], F& f) {
267:   if (k == 4 * P - 1) {
268:     c[3][P - 1] = rem;
269:     f(c);
270:     c[3][P - 1] = 0;
271:     return;
272:   }
273:   int r = k / P, x = k % P;
274:   for (int t = 0; t <= rem; t++) {
275:     c[r][x] = t;
276:     rec_dist(k + 1, rem - t, c, f);
277:   }
278:   c[r][x] = 0;
279: }

281: template <class F>
282: static void for_multisets(int R, F& f) {
283:   int m[P];
284:   for (m[0] = 0; m[0] <= R; m[0]++)
285:     for (m[1] = 0; m[0] + m[1] <= R; m[1]++)
286:       for (m[2] = 0; m[0] + m[1] + m[2] <= R; m[2]++)
287:         for (m[3] = 0; m[0] + m[1] + m[2] + m[3] <= R; m[3]++) {
288:           m[4] = R - m[0] - m[1] - m[2] - m[3];
289:           f(m);
290:         }
291: }

299:   else if (bug == "parity") g_parity = 1;
300:   else if (bug != "none") {

304:   build_group();

306:   FILE* fav = xopen(out + "/avoider_classes.txt");

308:   FILE* fnr = xopen(out + "/nonrigid_avoider_classes.txt");
309:   FILE* fsv = xopen(out + "/survivor_multisets.txt");
310:   FILE* fsm = xopen(out + "/summary.json");
311:   double t0 = now_s();
312:   std::vector<int8_t> filt(CODES);
313:   int fail = 0;
314:   u64 T_words_formula = 0, T_words_filter = 0, T_words_dp = 0, T_avoid_words = 0, T_cls_all = 0, T_cls_mixed = 0,
315:       T_nr_words = 0, T_mism = 0, T_words_formula_H04 = 0;
316:   fprintf(fsm, "{\n  \"mode\": \"enum\",\n  \"bug\": \"%s\",\n  \"per_H\": [\n", bug.c_str());
317:   for (int H = 0; H <= 7; H++) {
318:     int R = 14 - 2 * H;

320:     int nL = setup_ltuples(H, L);

324:     std::fill(filt.begin(), filt.end(), (int8_t)-1);
325:     auto fm = [&](const int* m) {
326:       bool s = filter_d(H, m);
327:       filt[mcode(m)] = s ? 1 : 0;
328:       ms_total++;
329:       u64 pr = 1;
330:       for (int x = 0; x < P; x++) pr *= binom(m[x] + 3, 3);
331:       sum_prod += pr;
332:       if (s) ms_solved++;
333:       else {
334:         ms_surv++;
335:         if (rigid(m)) ms_surv_rigid++;
336:         surv_dist_formula += pr;
337:         fprintf(fsv, "H=%d|M=%d,%d,%d,%d,%d|rigid=%d|dists=%llu\n", H, m[0], m[1], m[2], m[3], m[4], rigid(m) ? 1 : 0, pr);
338:       }
339:     };
340:     for_multisets(R, fm);
341:     // ---- distribution level: explicit enumeration of all C(R+19,19) run-multiset 4-tuples
342:     u64 d_total = 0, d_solved = 0, d_surv = 0, w_filter = 0, w_dp = 0, w_dp_solved = 0, w_dp_avoid = 0, cls_dp = 0,
343:         cls_all = 0, cls_mixed = 0, cls_none = 0, nr_words = 0, nr_classes = 0, mism = 0, lit_total = 0, badlen = 0;
344:     static u64 mask[NY][2];  // bit b = l-tuple index b (nL <= 120 < 128)
345:     auto leaf = [&](int c[4][P]) {
346:       int m[P];
347:       for (int x = 0; x < P; x++) m[x] = c[0][x] + c[1][x] + c[2][x] + c[3][x];
348:       int8_t fv = filt[mcode(m)];
349:       u64 num = 1, den = 1;
350:       for (int r = 0; r < 4; r++) {
351:         int rr = 0;
352:         for (int x = 0; x < P; x++) {
353:           rr += c[r][x];
354:           den *= FACT[c[r][x]];
355:         }
356:         num *= FACT[rr];
357:       }
358:       lit_total += num / den;
359:       d_total++;
360:       if (fv != 0) {  // (d) holds: all nL*625 completions solved (proved), counted
361:         d_solved++;
362:         w_filter += (u64)nL * NY;
363:         return;
364:       }
365:       d_surv++;
366:       memset(mask, 0, sizeof mask);
367:       bool rg = rigid(m);
368:       auto cb = [&](int a1, int a2, int a3, int a4, int y1, int y2, int y3, int y4, bool solved) {
369:         int a[4] = {a1, a2, a3, a4}, y[4] = {y1, y2, y3, y4}, w[WLEN + 8];
370:         int n = build_word(c, a, y, w);
371:         if (n != WLEN) badlen++;
372:         if (dp_scratch(w, n) != solved) mism++;
373:         w_dp++;
374:         if (solved) w_dp_solved++;
375:         else {
376:           w_dp_avoid++;
377:           int b = LID[a1][a2][a3];
378:           mask[((y1 * P + y2) * P + y3) * P + y4][b >> 6] |= 1ull << (b & 63);
379:         }
380:       };
381:       enum_ly(c, H, cb);
382:       std::string rs = runs_str(c);
383:       for (int yc = 0; yc < NY; yc++) {
384:         cls_dp++;
385:         int k = __builtin_popcountll(mask[yc][0]) + __builtin_popcountll(mask[yc][1]);
386:         if (k == 0) {
387:           cls_none++;
388:           continue;
389:         }
390:         char yb[8];
391:         yfmt(yc, yb);
392:         if (k == nL) {
393:           cls_all++;
394:           fprintf(fav, "H=%d|runs=%s|y=%s|lavoid=%d/%d\n", H, rs.c_str(), yb, k, nL);
395:         } else {
396:           cls_mixed++;
397:           fprintf(fmx, "H=%d|runs=%s|y=%s|lavoid=%d/%d|mask=%llx:%llx\n", H, rs.c_str(), yb, k, nL, mask[yc][1], mask[yc][0]);
398:         }
399:         if (!rg) {
400:           nr_classes++;
401:           nr_words += (u64)k;
402:           fprintf(fnr, "H=%d|runs=%s|y=%s|lavoid=%d/%d|mask=%llx:%llx\n", H, rs.c_str(), yb, k, nL, mask[yc][1], mask[yc][0]);
403:         }
404:       }
405:     };

413:               d_solved + d_surv == d_total && d_surv == surv_dist_formula && w_filter + w_dp == words_formula &&
414:               w_dp == d_surv * (u64)nL * NY && cls_dp == d_surv * NY && lit_total == lit_formula && mism == 0 &&
415:               badlen == 0;
416:     if (!ok) fail = 1;
417:     T_words_formula += words_formula;
418:     if (H <= 4) T_words_formula_H04 += words_formula;
419:     T_words_filter += w_filter;
420:     T_words_dp += w_dp;
421:     T_avoid_words += w_dp_avoid;
422:     T_cls_all += cls_all;
423:     T_cls_mixed += cls_mixed;
424:     T_nr_words += nr_words;
425:     T_mism += mism;
426:     fprintf(fsm,
427:             "    {\"H\": %d, \"R\": %d, \"ltuples_enumerated\": %d, \"ltuples_formula\": %llu, \"ytuples\": %d,\n"
428:             "     \"multisets_enumerated\": %llu, \"multisets_formula\": %llu, \"multisets_filter_solved\": %llu,"
429:             " \"multisets_survivors\": %llu, \"multisets_survivors_rigid\": %llu,\n"

431:             " \"dists_filter_solved\": %llu, \"dists_survivors\": %llu, \"dists_survivors_formula\": %llu,\n"

466: static int run_brute(const std::string& out, int H, int shard, int nshards) {
467:   if (H < 0 || H > 7 || nshards < 1 || shard < 0 || shard >= nshards) {

470:   }

473:   FILE* fav = xopen(out + "/avoider_classes.txt");
474:   FILE* fmx = xopen(out + "/mixed_classes.txt");
475:   FILE* fsm = xopen(out + "/summary.json");
476:   double t0 = now_s();
477:   int R = 14 - 2 * H;
478:   std::vector<std::array<int, 4>> L;
479:   int nL = setup_ltuples(H, L);
480:   std::vector<int8_t> filt(CODES, (int8_t)-1);
481:   auto fm = [&](const int* m) { filt[mcode(m)] = filter_d(H, m) ? 1 : 0; };
482:   for_multisets(R, fm);
483:   u64 didx = 0, d_shard = 0, d_shard_fsolved = 0, words = 0, solved_w = 0, avoid_w = 0, viol = 0, stride_checked = 0,
